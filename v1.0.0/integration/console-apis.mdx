### Step 2: Set Up Authentication

```javascript Console API Authentication Setup
{`class ConsoleAPIClient {
  constructor(baseURL) {
    this.baseURL = baseURL;
    this.accessToken = null;
    this.refreshToken = null;
    this.tokenExpiry = null;
  }
  
  async registerUser(userData) {
    const response = await fetch(\`\${this.baseURL}/auth/m/register\`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify(userData)
    });
    
    if (!response.ok) {
      const error = await response.json();
      throw new Error(\`Registration failed: \${error.error.message}\`);
    }
    
    return await response.json();
  }
  
  async login(username, password) {
    const response = await fetch(\`\${this.baseURL}/auth/login\`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({ username, password })
    });
    
    if (!response.ok) {
      const error = await response.json();
      throw new Error(\`Login failed: \${error.error.message}\`);
    }
    
    const data = await response.json();
    this.accessToken = data.data.access_token;
    this.refreshToken = data.data.refresh_token;
    this.tokenExpiry = Date.now() + (data.data.expires_in * 1000);
    
    return data;
  }
  
  async refreshAccessToken() {
    if (!this.refreshToken) {
      throw new Error('No refresh token available. Please login again.');
    }
    
    const response = await fetch(\`\${this.baseURL}/auth/refresh/token\`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({ refresh_token: this.refreshToken })
    });
    
    if (!response.ok) {
      throw new Error('Token refresh failed. Please login again.');
    }
    
    const data = await response.json();
    this.accessToken = data.data.access_token;
    this.tokenExpiry = Date.now() + (data.data.expires_in * 1000);
    
    return this.accessToken;
  }
  
  async makeRequest(endpoint, options = {}) {
    if (!this.accessToken || Date.now() >= this.tokenExpiry) {
      await this.refreshAccessToken();
    }
    
    return fetch(\`\${this.baseURL}\${endpoint}\`, {
      ...options,
      headers: {
        'Authorization': \`Bearer \${this.accessToken}\`,
        'Content-Type': 'application/json',
        ...options.headers
      }
    });
  }
}`}
```

```javascript Organization Setup
          method: 'POST',
```

```javascript Team Management System
            firstname: member.firstname,
            lastname: member.lastname,
            email: member.email,
            role: member.role || 'Organization_User'

        });
        
        results.push({
          member,
          success: response.ok,
          response: response.ok ? await response.json() : await response.json()
        });
        
      } catch (error) {
        results.push({
          member,
          success: false,
          error: error.message
        });
      }
    }
    
    return results;
  }
  
  async getTeamWithMembers(teamId) {
    // Get team details
    const teamResponse = await this.apiClient.makeRequest(\`/org/team/members/\${teamId}\`);
    
    if (!teamResponse.ok) {
      throw new Error('Failed to get team members');
    }
    
    const teamData = await teamResponse.json();
    
    // Get team information
    const teamsResponse = await this.apiClient.makeRequest('/org/teams');
    const teams = await teamsResponse.json();
    const teamInfo = teams.data.find(t => t.team_id === teamId);
    
    return {
      ...teamInfo,
      members: teamData.data
    };
  }
  
  async updateTeamMembers(teamId, memberUpdates) {
    const results = [];
    
    for (const update of memberUpdates) {
      try {
        // Update member role or permissions

          method: 'PATCH',
          body: JSON.stringify(update.data)
        });
        
        results.push({
          memberId: update.memberId,
          success: response.ok,
          response: response.ok ? await response.json() : await response.json()
        });
        
      } catch (error) {
        results.push({
          memberId: update.memberId,
          success: false,
          error: error.message
        });
      }
    }
    
    return results;
  }
  
  async removeMemberFromTeam(teamId, memberId) {
    const response = await this.apiClient.makeRequest(\`/org/team/member/\${teamId}/\${memberId}\`, {
      method: 'DELETE'
    });

      throw new Error('Failed to remove member from team');
    }
    
    return await response.json();
  }
```

## Error Handling and Retry Logic

### Comprehensive Error Handling

```javascript Console API Error Handling
{`class ConsoleAPIErrorHandler {
  static async handleResponse(response, context = '') {
    if (response.ok) {
      return await response.json();
    }
    
    const errorData = await response.json().catch(() => ({}));
    
    switch (response.status) {
      case 400:
        throw new ConsoleAPIError('Bad Request', {
          status: 400,
          message: errorData.error?.message || 'Invalid request data',
          details: errorData.error?.details,
          context
        });
        
      case 401:
        throw new ConsoleAPIError('Unauthorized', {
          status: 401,
          message: 'Authentication required or token expired',
          context
        });
        
      case 403:
        throw new ConsoleAPIError('Forbidden', {
          status: 403,
          message: 'Insufficient permissions for this operation',
          context
        });
        
      case 404:
        throw new ConsoleAPIError('Not Found', {
          status: 404,
          message: 'Resource not found',
          context
        });
        
      case 409:
        throw new ConsoleAPIError('Conflict', {
          status: 409,
          message: errorData.error?.message || 'Resource already exists',
          context
        });
        
      case 429:
        throw new ConsoleAPIError('Rate Limited', {
          status: 429,
          message: 'Too many requests',
          retryAfter: response.headers.get('Retry-After'),
          context
        });
        
      case 500:
        throw new ConsoleAPIError('Server Error', {
          status: 500,
          message: 'Internal server error',
          context
        });
        
      default:
        throw new ConsoleAPIError('Unknown Error', {
          status: response.status,
          message: errorData.error?.message || 'Unknown error occurred',
          context
        });
    }
  }
}

class ConsoleAPIError extends Error {
  constructor(message, details = {}) {
    super(message);
    this.name = 'ConsoleAPIError';
    this.details = details;
  }
}`}
```

### Retry Logic with Exponential Backoff

```javascript Retry Logic
{`class ConsoleAPIRetryHandler {
  constructor(maxRetries = 3, baseDelay = 1000) {
    this.maxRetries = maxRetries;
    this.baseDelay = baseDelay;
  }
  
  async executeWithRetry(operation, context = '') {
    let lastError;
    
    for (let attempt = 1; attempt <= this.maxRetries; attempt++) {
      try {
        const response = await operation();
        return await ConsoleAPIErrorHandler.handleResponse(response, context);
        
      } catch (error) {
        lastError = error;
        
        // Don't retry on client errors (4xx)
        if (error instanceof ConsoleAPIError && error.details.status >= 400 && error.details.status < 500) {
          throw error;
        }
        
        // Don't retry on last attempt
        if (attempt === this.maxRetries) {
          break;
        }
        
        // Calculate delay with exponential backoff
        const delay = this.calculateDelay(attempt, error);
        console.log(\`Attempt \${attempt} failed: \${error.message}. Retrying in \${delay}ms...\`);
        
        await this.sleep(delay);
      }
    }
    
    throw new Error(\`Operation failed after \${this.maxRetries} attempts: \${lastError.message}\`);
  }
  
  calculateDelay(attempt, error) {
    // Handle rate limiting
    if (error instanceof ConsoleAPIError && error.details.status === 429) {
      return parseInt(error.details.retryAfter || '60') * 1000;
    }
    
    // Exponential backoff with jitter
    const baseDelay = this.baseDelay * Math.pow(2, attempt - 1);
    const jitter = Math.random() * 0.1 * baseDelay;
    
    return Math.min(baseDelay + jitter, 30000); // Max 30 seconds
  }
  
  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}`}
```

### Role-Based Access Control

```javascript Role-Based Access Control
{`class RoleBasedAccessControl {
  constructor(apiClient) {
    this.apiClient = apiClient;
    this.userPermissions = null;
  }
  
  async loadUserPermissions() {
    const response = await this.apiClient.makeRequest('/auth/user/permissions');
    const data = await response.json();
    this.userPermissions = data.data.permissions;
    return this.userPermissions;
  }
  
  hasPermission(permission) {
    if (!this.userPermissions) {
      throw new Error('User permissions not loaded');
    }
    
    return this.userPermissions.includes(permission);
  }
  
  canManageTeams() {
    return this.hasPermission('manage_teams');
  }
  
  canManagePlatforms() {
    return this.hasPermission('manage_platforms');
  }
  
  canManageUsers() {
    return this.hasPermission('manage_users');
  }
  
  canManageOrganization() {
    return this.hasPermission('manage_organization');
  }
  
  async executeWithPermissionCheck(permission, operation) {
    if (!this.hasPermission(permission)) {
      throw new Error(\`Insufficient permissions: \${permission} required\`);
    }
    
    return await operation();
  }
}`}
```

```javascript Request Monitoring
{`class ConsoleAPIMonitor {
  constructor(apiClient, metrics) {
    this.apiClient = apiClient;
    this.metrics = metrics;
  }
  
  async makeMonitoredRequest(endpoint, options = {}) {
    const startTime = Date.now();
    const requestId = this.generateRequestId();
    
    try {
      const response = await this.apiClient.makeRequest(endpoint, options);
      const duration = Date.now() - startTime;
      
      this.metrics.recordRequest({
        requestId,
        endpoint,
        method: options.method || 'GET',
        status: response.status,
        duration,
        success: response.ok
      });
      
      return response;
      
    } catch (error) {
      const duration = Date.now() - startTime;
      
      this.metrics.recordError({
        requestId,
        endpoint,
        method: options.method || 'GET',
        error: error.message,
        duration
      });
      
      throw error;
    }
  }
  
  generateRequestId() {
    return Math.random().toString(36).substr(2, 9);
  }
}`}
```
