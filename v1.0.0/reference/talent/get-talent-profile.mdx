---
title: "Get Talent Profile"
api: "GET /talent/{talentId}"
description: "Retrieve detailed profile information for a specific talent"
---

Retrieve comprehensive profile information for a specific talent including their skills, experience, portfolio, reviews, and availability. This endpoint provides detailed information for talent evaluation and hiring decisions.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication. It's designed for talent evaluation and hiring management.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

<CodeBlock lang="bash">
Authorization: Bearer <your_jwt_token>
</CodeBlock>

<ParamField path="talentId" type="string" required>

import requests
from typing import Dict, List, Optional

class TalentClient:
    def __init__(self, api_client):
        self.api_client = api_client
    
    def get_talent_profile(self, 
                          talent_id: str,
                          include_portfolio: bool = True,
                          include_reviews: bool = True,
                          include_availability: bool = False,
                          include_contact: bool = False) -> Dict:
        """Get detailed talent profile information"""
        try:
            params = {
                "include_portfolio": include_portfolio,
                "include_reviews": include_reviews,
                "include_availability": include_availability,
                "include_contact": include_contact
            }
            
            response = self.api_client.get(f'/talent/{talent_id}', params=params)
            response.raise_for_status()
            
            return response.json()
            
        except requests.exceptions.RequestException as e:
            print(f"Failed to get talent profile: {e}")
            raise
    
    def evaluate_talent(self, talent_id: str, required_skills: List[str]) -> Dict:
        """Evaluate talent against required skills"""
        result = self.get_talent_profile(talent_id, include_portfolio=True)
        talent = result.get("data", {})
        
        evaluation = {
            "talent_id": talent_id,
            "name": talent.get("name"),
            "match_score": 0,
            "matched_skills": [],
            "missing_skills": [],
            "recommendation": ""
        }
        
        talent_skills = talent.get("skills", [])
        
        for skill in required_skills:
            if skill.lower() in [s.lower() for s in talent_skills]:
                evaluation["matched_skills"].append(skill)
            else:
                evaluation["missing_skills"].append(skill)
        
        evaluation["match_score"] = (len(evaluation["matched_skills"]) / len(required_skills)) * 100
        
        if evaluation["match_score"] >= 80:
            evaluation["recommendation"] = "Highly recommended"
        elif evaluation["match_score"] >= 60:
            evaluation["recommendation"] = "Good match"
        elif evaluation["match_score"] >= 40:
            evaluation["recommendation"] = "Partial match"
        else:
            evaluation["recommendation"] = "Not recommended"
        
        return evaluation
    
    def search_talents_by_skills(self, skills: List[str], limit: int = 10) -> List[Dict]:
        """Search talents by skills"""
        result = self.api_client.get('/talent', params={'skills': ','.join(skills), 'limit': limit})
        result.raise_for_status()
        
        return result.json().get("data", [])
