---
title: "Get Clients"
api: "GET /clients"
description: "Retrieve list of API clients and their configurations"
---

Retrieve a comprehensive list of all API clients associated with your organization, including their configurations, permissions, and usage statistics.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication. It's designed for API client management and monitoring.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

<CodeBlock lang="bash">
Authorization: Bearer <your_jwt_token>
</CodeBlock>

## Query Parameters

<ParamField query="page" type="string" required="false">
  Page number for pagination (default: 1)
  
  **Example**: "1"
</ParamField>

<ParamField query="limit" type="string" required="false">
  Number of clients per page (default: 20, max: 100)
  
  **Example**: "20"
</ParamField>

<ParamField query="status" type="string" required="false">
  Filter clients by status
  
  **Options**: "active", "inactive", "suspended", "pending"
  
  **Example**: "active"
</ParamField>

<ParamField query="type" type="string" required="false">
  Filter clients by type
  
  **Options**: "web", "mobile", "server", "integration"
  
  **Example**: "web"
</ParamField>

<ParamField query="search" type="string" required="false">
  Search clients by name or description
  
  **Example**: "mobile app"
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Indicates if the request was successful
</ResponseField>

<ResponseField name="data" type="array">
  Array of client objects
</ResponseField>

<ResponseField name="data[].client_id" type="string">
  Unique identifier for the client
</ResponseField>

<ResponseField name="data[].name" type="string">
  Name of the client
</ResponseField>

<ResponseField name="data[].description" type="string">
  Description of the client
</ResponseField>

<ResponseField name="data[].type" type="string">
  Type of client (web, mobile, server, integration)
</ResponseField>

<ResponseField name="data[].status" type="string">
  Current status of the client
</ResponseField>

<ResponseField name="data[].permissions" type="array">
  List of permissions granted to the client
</ResponseField>

<ResponseField name="data[].created_at" type="string">
  ISO 8601 timestamp when the client was created
</ResponseField>

<ResponseField name="data[].last_used" type="string">
  ISO 8601 timestamp of last API usage
</ResponseField>

<ResponseField name="data[].usage_stats" type="object">
  Usage statistics for the client
</ResponseField>

<ResponseField name="data[].usage_stats.requests_today" type="integer">
  Number of requests made today
</ResponseField>

<ResponseField name="data[].usage_stats.requests_this_month" type="integer">
  Number of requests made this month
</ResponseField>

<ResponseField name="data[].usage_stats.rate_limit" type="integer">
  Rate limit for the client (requests per hour)
</ResponseField>

<ResponseField name="pagination" type="object">
  Pagination information
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable message about the operation result
</ResponseField>

<ResponseField name="timestamp" type="string">
  ISO 8601 timestamp of the response
</ResponseField>

<CodeGroup>

```bash cURL
  -H "Authorization: Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9..." \
  -H "Content-Type: application/json"
```

```javascript JavaScript
const response = await fetch('https://api.example.com/clients?page=1&limit=10&status=active', {
  method: 'GET',
  headers: {
    'Authorization': 'Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...',
    'Content-Type': 'application/json'
  }
});

const data = await response.json();

```

```python Python
import requests

params = {
    'page': 1,
    'limit': 10,
    'status': 'active'
}

response = requests.get(
    'https://api.example.com/clients',
    headers={
        'Authorization': 'Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...',
        'Content-Type': 'application/json'
    },
    params=params
)

print(response.json())
```
