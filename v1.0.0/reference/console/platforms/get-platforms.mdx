---
title: "Get Platforms"
api: "GET /org/platforms"
description: "Retrieve list of platforms in the organization"
---

Retrieve a comprehensive list of all platforms within your organization, including their details, status, and configurations. This endpoint provides detailed information about platform composition and management.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication. It's designed for platform management and oversight within the Weir AI platform.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Query Parameters

<ParamField query="page" type="string" required="false">
  Page number for pagination
  
  **Example**: "1"
  
  **Default**: "1"
  
  **Constraints**: 
  - Must be a positive integer
  - Minimum value: 1
</ParamField>

<ParamField query="limit" type="string" required="false">
  Number of platforms to return per page
  
  **Example**: "20"
  
  **Default**: "20"
  
  **Constraints**: 
  - Must be between 1 and 100
  - Recommended: 20-50 for optimal performance
</ParamField>

<ParamField query="type" type="string" required="false">
  Filter platforms by type
  
  **Options**: "Forum/Community", "Social Media", "E-commerce", "Content Management", "API Service", "Mobile App", "Web Application", "Other"
  
  **Example**: "API Service"
  
  **Note**: If not specified, returns all platforms regardless of type
</ParamField>

<ParamField query="status" type="string" required="false">
  Filter platforms by status
  
  **Options**: "active", "inactive", "pending", "error"
  
  **Example**: "active"
  
  **Default**: "active"
  
  **Note**: If not specified, returns only active platforms
</ParamField>

<ParamField query="teamId" type="string" required="false">
  Filter platforms by team ID
  
  **Example**: "team_1234567890abcdef"
  
  **Constraints**: 
  - Must be a valid team ID
  - Team must exist in your organization
  - You must have access to the team
</ParamField>

<ParamField query="search" type="string" required="false">
  Search platforms by name or description
  
  **Example**: "api"
  
  **Constraints**: 
  - Minimum length: 2 characters
  - Case-insensitive search
  - Searches both name and description fields
</ParamField>

<ParamField query="sort_by" type="string" required="false">
  Sort platforms by specified field
  
  **Options**: "name", "type", "status", "created_at", "updated_at"
  
  **Example**: "created_at"
  
  **Default**: "name"
</ParamField>

<ParamField query="sort_order" type="string" required="false">
  Sort order for the results
  
  **Options**: "asc", "desc"
  
  **Example**: "desc"
  
  **Default**: "asc"
</ParamField>

<ParamField query="tags" type="string" required="false">
  Filter platforms by tags (comma-separated)
  
  **Example**: "production,api"
  
  **Constraints**: 
  - Multiple tags separated by commas
  - Case-insensitive matching
  - Returns platforms that have any of the specified tags
</ParamField>

          if (options.sortOrder) params.append('sort_order', options.sortOrder);
          if (options.tags) params.append('tags', options.tags);
          
          const response = await this.apiClient.makeRequest(
            `/org/platforms?${params.toString()}`
          );
          
          if (!response.ok) {
            throw new Error('Failed to get platforms');
          }
          
          return await response.json();

          console.error('Error getting platforms:', error);

    import requests
    from typing import Dict, List, Optional
    
    class PlatformListingManager:
        def __init__(self, api_client):
            self.api_client = api_client
        
        def get_platforms(self, 
                         page: int = 1,
                         limit: int = 20,
                         platform_type: Optional[str] = None,
                         status: str = "active",
                         team_id: Optional[str] = None,
                         search: Optional[str] = None,
                         sort_by: str = "name",
                         sort_order: str = "asc",
                         tags: Optional[List[str]] = None) -> Dict:
            """Get platforms with optional filtering and pagination"""
            try:
                params = {
                    "page": page,
                    "limit": limit,
                    "status": status,
                    "sort_by": sort_by,
                    "sort_order": sort_order
                }
                
                if platform_type:
                    params["type"] = platform_type
                if team_id:
                    params["teamId"] = team_id
                if search:
                    params["search"] = search
                if tags:
                    params["tags"] = ",".join(tags)
                
                response = self.api_client.get('/org/platforms', params=params)
                response.raise_for_status()
                
                return response.json()
                
            except requests.exceptions.RequestException as e:
                print(f"Error getting platforms: {e}")
                raise
        
        def get_all_platforms(self) -> List[Dict]:
            """Get all platforms across all pages"""
            all_platforms = []
            page = 1
            has_more = True
            
            while has_more:
                result = self.get_platforms(page=page, limit=50)
                all_platforms.extend(result["data"])
                has_more = result["pagination"]["has_next"]
                page += 1
            
            return all_platforms
        
        def get_active_platforms(self) -> Dict:
            """Get all active platforms"""
            return self.get_platforms(
                status="active",
                sort_by="name",
                sort_order="asc"
            )
        
        def get_api_platforms(self) -> Dict:
            """Get all API service platforms"""
            return self.get_platforms(
                platform_type="API Service",
                status="active"
            )
        
        def get_web_app_platforms(self) -> Dict:
            """Get all web application platforms"""
            return self.get_platforms(
                platform_type="Web Application",
                status="active"
            )
        
        def get_mobile_app_platforms(self) -> Dict:
            """Get all mobile app platforms"""
            return self.get_platforms(
                platform_type="Mobile App",
                status="active"
            )
        
        def get_ecommerce_platforms(self) -> Dict:
            """Get all e-commerce platforms"""
            return self.get_platforms(
                platform_type="E-commerce",
                status="active"
            )
        
        def get_platforms_by_team(self, team_id: str) -> Dict:
            """Get platforms for a specific team"""
            return self.get_platforms(
                team_id=team_id,
                status="active"
            )
        
        def search_platforms(self, search_term: str) -> Dict:
            """Search platforms by name or description"""
            return self.get_platforms(
                search=search_term,
                status="active"
            )
        
        def get_recent_platforms(self, limit: int = 10) -> Dict:
            """Get recently created platforms"""
            return self.get_platforms(
                status="active",
                sort_by="created_at",
                sort_order="desc",
                limit=limit
            )
        
        def get_platforms_by_tags(self, tags: List[str]) -> Dict:
            """Get platforms by tags"""
            return self.get_platforms(
                tags=tags,
                status="active"
            )
        
        def get_platform_statistics(self) -> Dict:
            """Get platform statistics"""
            all_platforms = self.get_all_platforms()
            
            stats = {
                "total_platforms": len(all_platforms),
                "active_platforms": len([p for p in all_platforms if p["status"] == "active"]),
                "inactive_platforms": len([p for p in all_platforms if p["status"] == "inactive"]),
                "pending_platforms": len([p for p in all_platforms if p["status"] == "pending"]),
                "error_platforms": len([p for p in all_platforms if p["status"] == "error"]),
                "verified_platforms": len([p for p in all_platforms if p["verification_status"] == "verified"]),
                "platforms_by_type": {},
                "platforms_by_team": {}
            }
            
            # Count by type
            for platform in all_platforms:
                platform_type = platform["type"]
                if platform_type not in stats["platforms_by_type"]:
                    stats["platforms_by_type"][platform_type] = 0
                stats["platforms_by_type"][platform_type] += 1
            
            # Count by team
            for platform in all_platforms:
                team_name = platform.get("team_name", "Unknown")
                if team_name not in stats["platforms_by_team"]:
                    stats["platforms_by_team"][team_name] = 0
                stats["platforms_by_team"][team_name] += 1
            
            return stats
        
        def get_platform_by_name(self, name: str) -> Optional[Dict]:
            """Find a specific platform by name"""
            result = self.search_platforms(name)
            for platform in result["data"]:
                if platform["name"].lower() == name.lower():
                    return platform
            return None
        
        def get_platforms_by_domain(self, domain: str) -> List[Dict]:
            """Find platforms by domain"""
            all_platforms = self.get_all_platforms()
            return [p for p in all_platforms if p.get("domain") == domain]
