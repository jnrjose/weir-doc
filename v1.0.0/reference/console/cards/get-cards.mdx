---
title: "Get Cards"
api: "GET /org/cards"
description: "Retrieve list of organization cards and payment methods"
---

Retrieve a comprehensive list of all payment cards and payment methods associated with your organization. This endpoint provides access to billing and payment information for managing subscriptions and transactions.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication. It's designed for payment and billing management.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

<CodeBlock lang="bash">
Authorization: Bearer <your_jwt_token>
</CodeBlock>

## Query Parameters

<ParamField query="page" type="string" required="false">
  Page number for pagination (default: 1)
  
  **Example**: "1"
</ParamField>

<ParamField query="limit" type="string" required="false">
  Number of cards per page (default: 20, max: 100)
  
  **Example**: "20"
</ParamField>

<ParamField query="status" type="string" required="false">
  Filter cards by status
  
  **Options**: "active", "inactive", "expired", "pending"
  
  **Example**: "active"
</ParamField>

<ParamField query="card_type" type="string" required="false">
  Filter cards by type
  
  **Options**: "credit", "debit", "prepaid", "corporate"
  
  **Example**: "credit"
</ParamField>

<ParamField query="search" type="string" required="false">
  Search cards by last 4 digits or nickname
  
  **Example**: "1234"
</ParamField>

import React, { useState, useEffect } from 'react';

const PaymentCards = () => {
  const [cards, setCards] = useState([]);
  const [loading, setLoading] = useState(true);
  const [filters, setFilters] = useState({
    status: 'active',
    card_type: ''
  });

  useEffect(() => {
    fetchCards();
  }, [filters]);

  const fetchCards = async () => {
    try {
      setLoading(true);
      const queryParams = new URLSearchParams(filters);
      const response = await fetch(`/api/org/cards?${queryParams}`, {
        headers: {
          'Authorization': `Bearer ${localStorage.getItem('access_token')}`
        }
      });
      
      const data = await response.json();
      setCards(data.data);
    } catch (error) {
      console.error('Error fetching cards:', error);
    } finally {
      setLoading(false);
    }
  };

  const formatCardNumber = (lastFour, brand) => {
    const brandPrefix = {
      'Visa': '4',
      'Mastercard': '5',
      'American Express': '3'
    };
    
    const prefix = brandPrefix[brand] || '•';
    return `${prefix}•••• •••• •••• ${lastFour}`;
  };

  if (loading) return <div>Loading cards...</div>;

  return (
    <div className="payment-cards">
      <div className="filters">
        <select 
          value={filters.status} 
          onChange={(e) => setFilters({...filters, status: e.target.value})}
        >
          <option value="">All Status</option>
          <option value="active">Active</option>
          <option value="inactive">Inactive</option>
          <option value="expired">Expired</option>

      </div>
      
      <div className="cards-grid">

import requests
from typing import List, Dict, Optional

class PaymentManager:
    def __init__(self, api_client):
        self.api_client = api_client
    
    def get_cards(self, 
                  status: Optional[str] = None,
                  card_type: Optional[str] = None,
                  search: Optional[str] = None,
                  page: int = 1,
                  limit: int = 20) -> Dict:
        """Get list of payment cards with optional filtering"""
        try:
            params = {
                "page": page,
                "limit": limit
            }
            
            if status:
                params["status"] = status
            if card_type:
                params["card_type"] = card_type
            if search:
                params["search"] = search
            
            response = self.api_client.get('/org/cards', params=params)
            response.raise_for_status()
            
            return response.json()
            
        except requests.exceptions.RequestException as e:
            print(f"Failed to get cards: {e}")
            raise
    
    def get_active_cards(self) -> List[Dict]:
        """Get all active payment cards"""
        result = self.get_cards(status="active")
        return result.get("data", [])
    
    def get_default_card(self) -> Optional[Dict]:
        """Get the default payment card"""
        cards = self.get_active_cards()
        for card in cards:
            if card.get("is_default", False):
                return card
        return None
    
    def find_card_by_last_four(self, last_four: str) -> Optional[Dict]:
        """Find card by last four digits"""
        cards = self.get_cards()
        for card in cards.get("data", []):
            if card["last_four"] == last_four:
                return card
        return None
