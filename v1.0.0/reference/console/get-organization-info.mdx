---
title: "Get Organization Information"
api: "GET /org/information"
description: "Retrieve organization details and settings"
---

Retrieve comprehensive information about your organization including basic details, contact information, settings, and configuration. This endpoint provides access to all organization-related data and settings.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication. It's designed for organization management and information retrieval.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

import requests
from typing import Dict, Optional

class OrganizationClient:
    def __init__(self, api_client):
        self.api_client = api_client
    
    def get_organization_info(self, 
                             include_settings: bool = True,
                             include_billing: bool = False,
                             include_stats: bool = False) -> Dict:
        """Get organization information with optional data"""
        try:
            params = {
                "include_settings": include_settings,
                "include_billing": include_billing,
                "include_stats": include_stats
            }
            
            response = self.api_client.get('/org/information', params=params)
            response.raise_for_status()
            
            return response.json()
            
        except requests.exceptions.RequestException as e:
            print(f"Failed to get organization info: {e}")
            raise
    
    def get_basic_info(self) -> Dict:
        """Get basic organization information only"""
        result = self.get_organization_info(
            include_settings=False,
            include_billing=False,
            include_stats=False
        )
        return result.get("data", {})
    
    def get_organization_stats(self) -> Dict:
        """Get organization statistics"""
        result = self.get_organization_info(
            include_settings=False,
            include_billing=False,
            include_stats=True
        )
        return result.get("data", {}).get("stats", {})
    
    def get_organization_settings(self) -> Dict:
        """Get organization settings"""
        result = self.get_organization_info(
            include_settings=True,
            include_billing=False,
            include_stats=False
        )
        return result.get("data", {}).get("settings", {})
