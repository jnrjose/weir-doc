---
title: "Get Organization Information"
api: "GET /org/information"
description: "Retrieve organization details and settings"
---

Retrieve comprehensive information about your organization including basic details, contact information, address, and configuration settings.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication. It's designed for organization management and administration.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

<CodeBlock lang="bash">
{`Authorization: Bearer <your_jwt_token>`}
</CodeBlock>

## Request

No request body or parameters required.

## Usage Notes

<AccordionGroup>
  <Accordion title="Data Completeness" icon="database">
    The response includes all available organization information. Some fields may be null or empty if not yet configured:
    
    - Address information is optional
    - Tax information may not be available for all organizations
    - Contact information should be kept up to date
  </Accordion>
  
  <Accordion title="Permissions" icon="shield">
    This endpoint requires organization access permissions:
    
    - Users must be associated with an organization
    - Minimum role: Organization User
    - Admin users see additional internal information
  </Accordion>
  
  <Accordion title="Data Sensitivity" icon="lock">
    Some information may be sensitive:
    
    - Tax information should be handled securely
    - Contact information may be confidential
    - Implement proper data protection measures
  </Accordion>
  
  <Accordion title="Caching" icon="clock">
    Organization information changes infrequently:
    
    - Consider caching the response for performance
    - Refresh cache when organization is updated
    - Monitor for changes using webhooks if available
  </Accordion>
</AccordionGroup>

{`class OrganizationDashboard {
  constructor(apiClient) {
    this.apiClient = apiClient;
    this.orgInfo = null;
  }
  
  async loadOrganizationInfo() {
    try {
      const response = await this.apiClient.makeRequest('/org/information');
      
      if (!response.ok) {
        throw new Error('Failed to load organization information');
      }
      
      const data = await response.json();
      this.orgInfo = data.data;
      
      this.displayOrganizationInfo();
      return this.orgInfo;
      
    } catch (error) {
      console.error('Error loading organization info:', error);
      this.displayError('Failed to load organization information');
      throw error;
    }
  }
  
  displayOrganizationInfo() {
    if (!this.orgInfo) return;
    
    // Update UI elements with organization data
    document.getElementById('org-name').textContent = this.orgInfo.name;
    document.getElementById('org-website').textContent = this.orgInfo.website;
    document.getElementById('org-industry').textContent = this.orgInfo.industry;
    document.getElementById('org-size').textContent = this.orgInfo.company_size;
    
    // Display address information
    if (this.orgInfo.address) {
      const address = this.orgInfo.address;
      const addressText = \`\${address.address1}\${address.address2 ? ', ' + address.address2 : ''}, \${address.city}, \${address.state} \${address.postal_code}, \${address.country}\`;
      document.getElementById('org-address').textContent = addressText;
    }
    
    // Display contact information
    if (this.orgInfo.contact) {
      document.getElementById('org-phone').textContent = this.orgInfo.contact.phone;
      document.getElementById('org-email').textContent = this.orgInfo.contact.email;
    }
  }
  
  displayError(message) {
    // Display error message to user
    const errorElement = document.getElementById('error-message');
    if (errorElement) {
      errorElement.textContent = message;
      errorElement.style.display = 'block';
    }
  }
}

// Usage
const dashboard = new OrganizationDashboard(apiClient);
await dashboard.loadOrganizationInfo();`}

{`class OrganizationValidator {
  constructor(apiClient) {
    this.apiClient = apiClient;
  }
  
  async validateOrganizationSetup() {
    try {
      const response = await this.apiClient.makeRequest('/org/information');
      const data = await response.json();
      const org = data.data;
      
      const validation = {
        isComplete: true,
        missingFields: [],
        warnings: []
      };
      
      // Check required fields
      if (!org.name) {
        validation.missingFields.push('Organization name');
        validation.isComplete = false;
      }
      
      if (!org.website) {
        validation.missingFields.push('Website');
        validation.warnings.push('Website is recommended for organization profile');
      }
      
      if (!org.industry) {
        validation.missingFields.push('Industry');
        validation.warnings.push('Industry classification helps with platform recommendations');
      }
      
      if (!org.company_size) {
        validation.missingFields.push('Company size');
        validation.warnings.push('Company size helps with feature recommendations');
      }
      
      // Check address completeness
      if (!org.address || !org.address.address1) {
        validation.missingFields.push('Address');
        validation.warnings.push('Address is required for billing and compliance');
      }
      
      // Check contact information
      if (!org.contact || !org.contact.email) {
        validation.missingFields.push('Contact email');
        validation.isComplete = false;
      }
      
      return validation;
      
    } catch (error) {
      console.error('Error validating organization:', error);
      return {
        isComplete: false,
        missingFields: ['Unable to validate - API error'],
        warnings: []
      };
    }
  }
  
  async getSetupRecommendations() {
    const validation = await this.validateOrganizationSetup();
    
    const recommendations = [];
    
    if (validation.missingFields.length > 0) {
      recommendations.push({
        type: 'required',
        message: \`Complete these required fields: \${validation.missingFields.join(', ')}\`,
        action: 'update_organization'
      });
    }
    
    if (validation.warnings.length > 0) {
      recommendations.push({
        type: 'recommended',
        message: \`Consider adding: \${validation.warnings.join(', ')}\`,
        action: 'enhance_profile'
      });
    }
    
    if (validation.isComplete) {
      recommendations.push({
        type: 'success',
        message: 'Organization profile is complete and up to date',
        action: 'none'
      });
    }
    
    return recommendations;
  }
}`}
