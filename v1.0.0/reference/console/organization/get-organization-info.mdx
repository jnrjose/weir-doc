---
title: "Get Organization Information"
api: "GET /org/information"
description: "Retrieve organization details and settings"
---

Retrieve comprehensive information about your organization, including basic details, contact information, address, and configuration settings. This endpoint provides a complete overview of your organization's profile and settings.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication. It's designed for organization management and profile viewing within the Weir AI platform.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Request

No request body or query parameters are required for this endpoint.

    import requests
    from typing import Dict, Optional
    
    class OrganizationManager:
        def __init__(self, api_client):
            self.api_client = api_client
            self._cached_info: Optional[Dict] = None
        
        def get_organization_info(self, use_cache: bool = True) -> Dict:
            """Get organization information with optional caching"""
            try:
                if use_cache and self._cached_info:
                    return self._cached_info
                
                response = self.api_client.get('/org/information')
                response.raise_for_status()
                
                result = response.json()
                
                # Cache the result
                if use_cache:
                    self._cached_info = result
                
                print(f"Organization: {result['data']['name']}")
                print(f"Teams: {result['data']['stats']['total_teams']}")
                print(f"Members: {result['data']['stats']['total_members']}")
                
                return result
                
            except requests.exceptions.RequestException as e:
                print(f"Error getting organization information: {e}")
                raise
        
        def get_organization_stats(self) -> Dict:
            """Get organization statistics"""
            info = self.get_organization_info()
            return info["data"]["stats"]
        
        def get_organization_settings(self) -> Dict:
            """Get organization settings"""
            info = self.get_organization_info()
            return info["data"]["settings"]
        
        def get_organization_contact(self) -> Dict:
            """Get organization contact information"""
            info = self.get_organization_info()
            return info["data"]["contact"]
        
        def get_organization_address(self) -> Dict:
            """Get organization address"""
            info = self.get_organization_info()
            return info["data"]["address"]
        
        def is_organization_active(self) -> bool:
            """Check if organization is active"""
            stats = self.get_organization_stats()
            return stats["total_members"] > 0
        
        def get_organization_summary(self) -> Dict:
            """Get organization summary"""
            info = self.get_organization_info()
            data = info["data"]
            return {
                "name": data["name"],
                "industry": data["industry"],
                "size": data["company_size"],
                "teams": data["stats"]["total_teams"],
                "members": data["stats"]["total_members"],
                "projects": data["stats"]["total_projects"]
            }
        
        def get_organization_timezone(self) -> str:
            """Get organization timezone"""
            settings = self.get_organization_settings()
            return settings["timezone"]
        
        def get_organization_currency(self) -> str:
            """Get organization currency"""
            settings = self.get_organization_settings()
            return settings["currency"]
        
        def get_organization_language(self) -> str:
            """Get organization language"""
            settings = self.get_organization_settings()
            return settings["language"]
        
        def are_notifications_enabled(self, notification_type: str) -> bool:
            """Check if specific notification type is enabled"""
            settings = self.get_organization_settings()
            return settings["notifications"].get(notification_type, False)
        
        def clear_cache(self):
            """Clear cached organization information"""
            self._cached_info = None
        
        def refresh_organization_info(self) -> Dict:
            """Force refresh organization information"""
            self.clear_cache()
            return self.get_organization_info(use_cache=False)
