---
title: "Get Teams"
api: "GET /org/teams"
description: "Retrieve list of teams in the organization"
---

Retrieve a comprehensive list of all teams within your organization, including their details, member counts, and settings. This endpoint provides team management and overview capabilities.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication. It's designed for team management and organization overview.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Query Parameters

<ParamField query="page" type="string" required="false">
  Page number for pagination (default: 1)
  
  **Example**: "1"
</ParamField>

<ParamField query="limit" type="string" required="false">
  Number of teams per page (default: 20, max: 100)
  
  **Example**: "20"
</ParamField>

<ParamField query="team_type" type="string" required="false">
  Filter teams by type
  
  **Options**: "development", "design", "marketing", "sales", "support", "management", "other"
  
  **Example**: "development"
</ParamField>

<ParamField query="visibility" type="string" required="false">
  Filter teams by visibility level
  
  **Options**: "public", "private", "restricted"
  
  **Example**: "public"
</ParamField>

<ParamField query="search" type="string" required="false">
  Search teams by name or description
  
  **Example**: "development team"
</ParamField>

<ParamField query="sort_by" type="string" required="false">
  Sort teams by field
  
  **Options**: "name", "created_at", "member_count", "team_type"
  
  **Default**: "created_at"
  
  **Example**: "name"
</ParamField>

<ParamField query="sort_order" type="string" required="false">
  Sort order
  
  **Options**: "asc", "desc"
  
  **Default**: "desc"
  
  **Example**: "asc"
</ParamField>

import React, { useState, useEffect } from 'react';

const TeamList = () => {
  const [teams, setTeams] = useState([]);
  const [loading, setLoading] = useState(true);
  const [filters, setFilters] = useState({
    team_type: '',
    visibility: '',
    search: '',
    sort_by: 'created_at',
    sort_order: 'desc'
  });

  useEffect(() => {
    fetchTeams();
  }, [filters]);

  const fetchTeams = async () => {
    try {
      setLoading(true);
      const queryParams = new URLSearchParams(filters);
      const response = await fetch(`/api/org/teams?${queryParams}`, {
        headers: {
          'Authorization': `Bearer ${localStorage.getItem('access_token')}`
        }
      });
      
      const data = await response.json();
      setTeams(data.data);
    } catch (error) {
      console.error('Error fetching teams:', error);
    } finally {
      setLoading(false);
    }
  };

  const handleFilterChange = (key, value) => {
    setFilters(prev => ({ ...prev, [key]: value }));
  };

  if (loading) return <div>Loading teams...</div>;

  return (
    <div className="team-list">
      <div className="filters">
        <select 
          value={filters.team_type} 
          onChange={(e) => handleFilterChange('team_type', e.target.value)}
        >
          <option value="">All Types</option>
          <option value="development">Development</option>
          <option value="design">Design</option>
          <option value="marketing">Marketing</option>
        </select>
        
        <input
          type="text"
          placeholder="Search teams..."
          value={filters.search}
          onChange={(e) => handleFilterChange('search', e.target.value)}
        />
      </div>
      
      <div className="teams-grid">
        {teams.map(team => (
          <div key={team.team_id} className="team-card">
            <h3>{team.name}</h3>
            <p>{team.description}</p>
            <div className="team-meta">
              <span>Type: {team.team_type}</span>
              <span>Members: {team.member_count}/{team.max_members}</span>
            </div>
          </div>
        ))}
      </div>
    </div>
  );

import requests
from typing import List, Dict, Optional

class TeamManager:
    def __init__(self, api_client):
        self.api_client = api_client
    
    def get_teams(self, 
                  team_type: Optional[str] = None,
                  visibility: Optional[str] = None,
                  search: Optional[str] = None,
                  sort_by: str = "created_at",
                  sort_order: str = "desc",
                  page: int = 1,
                  limit: int = 20) -> Dict:
        """Get list of teams with optional filtering"""
        try:
            params = {
                "page": page,
                "limit": limit,
                "sort_by": sort_by,
                "sort_order": sort_order
            }
            
            if team_type:
                params["team_type"] = team_type
            if visibility:
                params["visibility"] = visibility
            if search:
                params["search"] = search
            
            response = self.api_client.get('/org/teams', params=params)
            response.raise_for_status()
            
            return response.json()
            
        except requests.exceptions.RequestException as e:
            print(f"Failed to get teams: {e}")
            raise
    
    def get_development_teams(self) -> List[Dict]:
        """Get all development teams"""
        result = self.get_teams(team_type="development")
        return result.get("data", [])
    
    def search_teams(self, query: str) -> List[Dict]:
        """Search teams by name or description"""
        result = self.get_teams(search=query)
        return result.get("data", [])
    
    def get_team_by_id(self, team_id: str) -> Optional[Dict]:
        """Get specific team by ID"""
        teams = self.get_teams()
        for team in teams.get("data", []):
            if team["team_id"] == team_id:
                return team
        return None
