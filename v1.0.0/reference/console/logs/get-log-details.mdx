---
title: "Get Log Details"
api: "GET /org/logs/{logId}"
description: "Get detailed information for a specific log entry"
---

Retrieve comprehensive details for a specific log entry, including full metadata, context information, and related events. This endpoint provides in-depth log analysis and debugging information.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication. It's designed for detailed log analysis and debugging within the Weir AI platform.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Path Parameters

<ParamField path="logId" type="string" required>
  Unique identifier of the log entry to retrieve
  
  **Example**: "log_1234567890abcdef"
  
  **Constraints**: 
  - Must be a valid log ID
  - Log must exist in your organization
  - You must have permission to view this log
</ParamField>

```javascript JavaScript
          }
```

    ```javascript
          throw error;

        const logDetails = await this.getLogDetails(logId);
        
        if (logDetails.data.level === 'error' || logDetails.data.level === 'fatal') {
          return {
            logId: logDetails.data.log_id,
            error: logDetails.data.error_message,
            stackTrace: logDetails.data.stack_trace,
            relatedLogs: logDetails.data.related_logs,
            metadata: logDetails.data.metadata
          };
        }
        
        return null;
      }

            riskScore: logDetails.data.metadata.security_context?.risk_score,
            anomalyDetected: logDetails.data.metadata.security_context?.anomaly_detected,

          };
        }
        
        return null;
      }
    ```

    ```python
    import requests
    from typing import Dict, Optional, List
    
    class LogAnalyzer:
        def __init__(self, api_client):
            self.api_client = api_client
        
        def get_log_details(self, log_id: str) -> Dict:
            """Get detailed information for a specific log entry"""
            try:
                response = self.api_client.get(f'/org/logs/{log_id}')
                response.raise_for_status()
                
                result = response.json()
                print(f"Log ID: {result['data']['log_id']}")
                print(f"Level: {result['data']['level']}")
                print(f"Category: {result['data']['category']}")
                print(f"Message: {result['data']['message']}")
                
                return result
                
            except requests.exceptions.RequestException as e:
                print(f"Error getting log details: {e}")
                raise
        
        def analyze_error_log(self, log_id: str) -> Optional[Dict]:
            """Analyze error log for debugging information"""
            log_details = self.get_log_details(log_id)
            
            if log_details['data']['level'] in ['error', 'fatal']:
                return {
                    'log_id': log_details['data']['log_id'],
                    'error': log_details['data']['error_message'],
                    'stack_trace': log_details['data']['stack_trace'],
                    'related_logs': log_details['data']['related_logs'],
                    'metadata': log_details['data']['metadata']
                }
            
            return None
        
        def get_security_context(self, log_id: str) -> Optional[Dict]:
            """Get security context for security-related logs"""
            log_details = self.get_log_details(log_id)
            
            if log_details['data']['category'] == 'security':
                security_context = log_details['data']['metadata'].get('security_context', {})
                return {
                    'threat_level': security_context.get('threat_level'),
                    'risk_score': security_context.get('risk_score'),
                    'anomaly_detected': security_context.get('anomaly_detected'),
                    'suspicious_patterns': security_context.get('suspicious_patterns', [])
                }
            
            return None
        
        def get_performance_metrics(self, log_id: str) -> Optional[Dict]:
            """Get performance metrics from log entry"""
            log_details = self.get_log_details(log_id)
            
            if log_details['data']['duration']:
                return {
                    'duration': log_details['data']['duration'],
                    'status_code': log_details['data']['status_code'],
                    'request_id': log_details['data']['request_id'],
                    'endpoint': log_details['data']['metadata'].get('request_details', {}).get('endpoint')
                }
            
            return None
        
        def get_user_context(self, log_id: str) -> Optional[Dict]:
            """Get user context from log entry"""
            log_details = self.get_log_details(log_id)
            
            if log_details['data']['user_id']:
                return {
                    'user_id': log_details['data']['user_id'],
                    'team_id': log_details['data']['team_id'],
                    'ip_address': log_details['data']['ip_address'],
                    'user_agent': log_details['data']['user_agent'],
                    'session_id': log_details['data']['session_id']
                }
            
            return None
        
        def get_related_events(self, log_id: str) -> List[Dict]:
            """Get related log events"""
            log_details = self.get_log_details(log_id)
            return log_details['data']['related_logs']
        
        def analyze_log_chain(self, log_id: str) -> Dict:
            """Analyze the complete log chain and relationships"""
            log_details = self.get_log_details(log_id)
            related_logs = log_details['data']['related_logs']
            
            chain_analysis = {
                'main_log': {
                    'id': log_details['data']['log_id'],
                    'level': log_details['data']['level'],
                    'message': log_details['data']['message'],
                    'timestamp': log_details['data']['timestamp']
                },
                'related_events': [],
                'chain_length': len(related_logs) + 1,
                'severity_levels': [log_details['data']['level']] + [log['level'] for log in related_logs],
                'time_span': None
            }
            
            # Calculate time span
            if related_logs:
                timestamps = [log_details['data']['timestamp']] + [log['timestamp'] for log in related_logs]
                timestamps.sort()
                chain_analysis['time_span'] = {
                    'start': timestamps[0],
                    'end': timestamps[-1],
                    'duration': len(timestamps)
                }
            
            # Analyze related events
            for log in related_logs:
                chain_analysis['related_events'].append({
                    'id': log['log_id'],
                    'level': log['level'],
                    'message': log['message'],
                    'timestamp': log['timestamp'],
                    'relationship': log['relationship']
                })
            
            return chain_analysis

    print(f"Log Details: {log_details['data']}")

    if error_analysis:

    # Get security context

    if security_context:

    print(f"Log Chain Analysis: {chain_analysis}")
    ```
