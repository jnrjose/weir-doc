---
title: "Get Teams"
api: "GET /org/teams"
description: "Retrieve list of teams in the organization with pagination support"
---

Retrieve a paginated list of all teams within your organization. This endpoint provides comprehensive team information including member counts, creation dates, and status.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication. It's designed for team management and organization administration.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

<CodeBlock lang="bash">
{`Authorization: Bearer <your_jwt_token>`}
</CodeBlock>

## Query Parameters

<ParamField query="page" type="string" required="false">
  Page number for pagination (default: 1)
  
  **Example**: "1"
</ParamField>

<ParamField query="limit" type="string" required="false">
  Number of teams per page (default: 20, max: 100)
  
  **Example**: "20"
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Indicates if the request was successful
</ResponseField>

<ResponseField name="data" type="array">
  Array of team objects
</ResponseField>

<ResponseField name="data[].team_id" type="string">

## Usage Notes

<AccordionGroup>
  <Accordion title="Pagination" icon="list">
    This endpoint supports pagination for large datasets:
    
    - Use `page` parameter to navigate through pages
    - Use `limit` parameter to control page size (max 100)
    - Check `pagination` object for navigation information
    - Use `has_next` and `has_prev` for UI navigation controls
  </Accordion>
  
  <Accordion title="Permissions" icon="shield">
    To view teams, you need:
    
    - Organization access permissions
    - Team viewing permissions
    - Valid authentication token
  </Accordion>
  
  <Accordion title="Data Freshness" icon="clock">
    Team data is updated in real-time:
    
    - Member counts reflect current team membership
    - Status changes are immediately visible
    - Creation and update timestamps are accurate
  </Accordion>
  
  <Accordion title="Performance" icon="tachometer">
    For optimal performance:
    
    - Use appropriate page sizes (10-50 items)
    - Cache results when possible
    - Implement proper error handling
  </Accordion>
</AccordionGroup>

{`class TeamDashboard {
  constructor(apiClient) {
    this.apiClient = apiClient;
    this.teams = [];
    this.currentPage = 1;
    this.pageSize = 20;
  }
  
  async loadTeams(page = 1) {
    try {
      const response = await this.apiClient.makeRequest(\`/org/teams?page=\${page}&limit=\${this.pageSize}\`);
      
      if (!response.ok) {
        throw new Error('Failed to load teams');
      }
      
      const data = await response.json();
      this.teams = data.data;
      this.currentPage = page;
      
      this.displayTeams();
      this.updatePagination(data.pagination);
      
      return data;
      
    } catch (error) {
      console.error('Error loading teams:', error);
      this.displayError('Failed to load teams');
      throw error;
    }
  }
  
  displayTeams() {
    const container = document.getElementById('teams-list');
    container.innerHTML = '';
    
    this.teams.forEach(team => {
      const teamElement = this.createTeamElement(team);
      container.appendChild(teamElement);
    });
  }
  
  createTeamElement(team) {
    const div = document.createElement('div');
    div.className = 'team-card';
    div.innerHTML = \`
      <div class="team-header">
        <h3>\${team.name}</h3>
        <span class="status status-\${team.status}">\${team.status}</span>
      </div>
      <div class="team-details">
        <p><strong>Members:</strong> \${team.member_count}</p>
        <p><strong>Created:</strong> \${new Date(team.created_at).toLocaleDateString()}</p>
        <p><strong>Updated:</strong> \${new Date(team.updated_at).toLocaleDateString()}</p>
        \${team.description ? \`<p><strong>Description:</strong> \${team.description}</p>\` : ''}
      </div>
      <div class="team-actions">
        <button onclick="viewTeam('\${team.team_id}')">View Details</button>
        <button onclick="editTeam('\${team.team_id}')">Edit</button>
        <button onclick="manageTeam('\${team.team_id}')">Manage</button>
      </div>
    \`;
    
    return div;
  }
  
  updatePagination(pagination) {
    const paginationElement = document.getElementById('pagination');
    paginationElement.innerHTML = \`
      <div class="pagination-info">
        Showing \${(pagination.page - 1) * pagination.limit + 1} to \${Math.min(pagination.page * pagination.limit, pagination.total)} of \${pagination.total} teams
      </div>
      <div class="pagination-controls">
        <button \${!pagination.has_prev ? 'disabled' : ''} onclick="loadPage(\${pagination.page - 1})">Previous</button>
        <span>Page \${pagination.page} of \${pagination.total_pages}</span>
        <button \${!pagination.has_next ? 'disabled' : ''} onclick="loadPage(\${pagination.page + 1})">Next</button>
      </div>
    \`;
  }
  
  async loadPage(page) {

}`}

{`class TeamAnalytics {
  constructor(apiClient) {
    this.apiClient = apiClient;
  }
  
  async getAllTeams() {
    const allTeams = [];
    let page = 1;
    let hasMore = true;
    
    while (hasMore) {
      const response = await this.apiClient.makeRequest(\`/org/teams?page=\${page}&limit=100\`);
      const data = await response.json();
      
      allTeams.push(...data.data);
      hasMore = data.pagination.has_next;
      page++;
    }
    
    return allTeams;
  }
  
  async generateTeamReport() {
    const teams = await this.getAllTeams();
    
    const stats = {
      total_teams: teams.length,
      active_teams: teams.filter(t => t.status === 'active').length,
      total_members: teams.reduce((sum, team) => sum + team.member_count, 0),
      average_team_size: teams.length > 0 ? teams.reduce((sum, team) => sum + team.member_count, 0) / teams.length : 0,
      largest_team: teams.reduce((max, team) => team.member_count > max.member_count ? team : max, teams[0]),
      newest_team: teams.reduce((newest, team) => new Date(team.created_at) > new Date(newest.created_at) ? team : newest, teams[0]),
      teams_by_month: this.groupTeamsByMonth(teams)
    };
    
    return stats;
  }
  
  groupTeamsByMonth(teams) {
    const months = {};
    
    teams.forEach(team => {
      const month = new Date(team.created_at).toISOString().substring(0, 7); // YYYY-MM
      months[month] = (months[month] || 0) + 1;
    });
    
    return months;
  }
  
  async getTeamGrowthTrend() {
    const teams = await this.getAllTeams();
    const sortedTeams = teams.sort((a, b) => new Date(a.created_at) - new Date(b.created_at));
    
    const growthData = [];
    let cumulativeCount = 0;
    
    sortedTeams.forEach(team => {
      cumulativeCount++;
      growthData.push({
        date: team.created_at,
        cumulative_teams: cumulativeCount,
        team_name: team.name
      });
    });
    
    return growthData;
  }
}`}
