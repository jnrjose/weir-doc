---
title: "Get Team Members"
api: "GET /org/team/members/{teamId}"
description: "Retrieve list of members in a specific team"
---

Retrieve a comprehensive list of all members in a specific team, including their roles, permissions, and membership details.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication. It's designed for team management and member administration.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Path Parameters

<ParamField path="teamId" type="string" required>
  The unique identifier of the team to get members for
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Indicates if the request was successful
</ResponseField>

<ResponseField name="data" type="array">
  Array of team member objects
</ResponseField>

<ResponseField name="data[].user_id" type="string">
  Unique identifier for the user
</ResponseField>

<ResponseField name="data[].email" type="string">
  Email address of the team member
</ResponseField>

<ResponseField name="data[].firstname" type="string">
  First name of the team member
</ResponseField>

<ResponseField name="data[].lastname" type="string">
  Last name of the team member
</ResponseField>

<ResponseField name="data[].role" type="string">
  Role of the member in the team (Organization_User, Team_Admin, Viewer)
</ResponseField>

<ResponseField name="data[].status" type="string">
  Membership status (active, pending, suspended)
</ResponseField>

<ResponseField name="data[].joined_at" type="string">
  ISO 8601 timestamp when the member joined the team
</ResponseField>

<ResponseField name="data[].last_active" type="string">
  ISO 8601 timestamp of the member's last activity
</ResponseField>

<ResponseField name="data[].permissions" type="array">
  List of specific permissions granted to the member
</ResponseField>

<ResponseField name="data[].invited_by" type="string">
  User ID of the person who invited this member
</ResponseField>

<ResponseField name="team_info" type="object">
  Information about the team
</ResponseField>

<ResponseField name="team_info.team_id" type="string">
  Unique identifier for the team
</ResponseField>

<ResponseField name="team_info.name" type="string">
  Name of the team
</ResponseField>

<ResponseField name="team_info.total_members" type="integer">
  Total number of members in the team
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable message about the operation result
</ResponseField>

<ResponseField name="timestamp" type="string">
  ISO 8601 timestamp of the response
</ResponseField>

## Usage Notes

<AccordionGroup>
  <Accordion title="Member Roles" icon="users">
    Team members can have different roles:
    
    - **Team_Admin**: Full team management permissions
    - **Organization_User**: Standard team member with basic permissions
    - **Viewer**: Read-only access to team resources
  </Accordion>
  
  <Accordion title="Member Status" icon="clock">
    Members can have different statuses:
    
    - **active**: Fully active team member
    - **pending**: Invitation sent but not yet accepted
    - **suspended**: Temporarily suspended from team access
  </Accordion>
  
  <Accordion title="Permissions" icon="shield">
    Each member has specific permissions:
    
    - Permissions are role-based but can be customized
    - Permissions determine what actions members can perform
    - Admin members can manage other members' permissions
  </Accordion>
  
  <Accordion title="Activity Tracking" icon="chart-line">
    The endpoint provides activity information:
    
    - `joined_at`: When the member joined the team
    - `last_active`: Last time the member was active
    - `invited_by`: Who invited the member to the team
  </Accordion>
</AccordionGroup>

{`class TeamMemberAnalytics {
  constructor(apiClient) {
    this.apiClient = apiClient;
  }
  
  async getTeamMemberStats(teamId) {
    const response = await this.apiClient.makeRequest(\`/org/team/members/\${teamId}\`);
    const data = await response.json();
    const members = data.data;
    
    const stats = {
      total_members: members.length,
      active_members: members.filter(m => m.status === 'active').length,
      pending_members: members.filter(m => m.status === 'pending').length,
      suspended_members: members.filter(m => m.status === 'suspended').length,
      role_distribution: this.getRoleDistribution(members),
      recent_joiners: this.getRecentJoiners(members, 30), // Last 30 days
      inactive_members: this.getInactiveMembers(members, 7), // Inactive for 7+ days
      average_activity: this.getAverageActivity(members)
    };
    
    return stats;
  }
  
  getRoleDistribution(members) {
    const distribution = {};
    members.forEach(member => {
      distribution[member.role] = (distribution[member.role] || 0) + 1;
    });
    return distribution;
  }
  
  getRecentJoiners(members, days) {
    const cutoffDate = new Date();
    cutoffDate.setDate(cutoffDate.getDate() - days);
    
    return members.filter(member => 
      new Date(member.joined_at) >= cutoffDate
    );
  }
  
  getInactiveMembers(members, days) {
    const cutoffDate = new Date();
    cutoffDate.setDate(cutoffDate.getDate() - days);
    
    return members.filter(member => 
      new Date(member.last_active) < cutoffDate
    );
  }
  
  getAverageActivity(members) {
    const now = new Date();
    const totalDays = members.reduce((sum, member) => {
      const lastActive = new Date(member.last_active);
      const daysSinceActive = Math.floor((now - lastActive) / (1000 * 60 * 60 * 24));
      return sum + daysSinceActive;
    }, 0);
    
    return members.length > 0 ? totalDays / members.length : 0;
  }
  
  async generateMemberReport(teamId) {
    const stats = await this.getTeamMemberStats(teamId);
    
    const report = {
      generated_at: new Date().toISOString(),
      team_id: teamId,
      summary: {
        total_members: stats.total_members,
        active_members: stats.active_members,
        pending_invitations: stats.pending_members,
        inactive_members: stats.inactive_members.length
      },
      insights: this.generateInsights(stats),
      recommendations: this.generateRecommendations(stats)
    };
    
    return report;
  }
  
  generateInsights(stats) {
    const insights = [];
    
    if (stats.pending_members > 0) {
      insights.push({
        type: 'info',
        message: \`\${stats.pending_members} pending invitations need attention\`
      });
    }
    
    if (stats.inactive_members.length > stats.total_members * 0.2) {
      insights.push({
        type: 'warning',
        message: \`\${stats.inactive_members.length} members have been inactive for 7+ days\`
      });
    }
    
    if (stats.role_distribution.Team_Admin === 0) {
      insights.push({
        type: 'error',
        message: 'No team administrators found - consider promoting a member'
      });
    }
    
    return insights;
  }
  
  generateRecommendations(stats) {
    const recommendations = [];
    
    if (stats.pending_members > 0) {
      recommendations.push('Follow up on pending invitations');
    }
    
    if (stats.inactive_members.length > 0) {
      recommendations.push('Reach out to inactive members or consider removing them');
    }
    
    if (stats.role_distribution.Team_Admin < 2) {
      recommendations.push('Consider having at least 2 team administrators for redundancy');
    }
    
    return recommendations;
  }
}
