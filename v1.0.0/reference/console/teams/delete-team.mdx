---
title: "Delete Team"
api: "DELETE /org/team/{teamId}"
description: "Delete a team from the organization permanently"
---

Permanently delete a team from your organization. This action cannot be undone and will remove all team data, member associations, and related resources.

<Warning>
  **Irreversible Action**
  
  Deleting a team is permanent and cannot be undone. All team data, member associations, and related resources will be permanently removed. Make sure you want to delete the team before proceeding.
</Warning>

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication with elevated permissions for team management.
</Note>

## Authentication

This endpoint requires Bearer Token authentication with team management permissions. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Path Parameters

<ParamField path="teamId" type="string" required>
  The unique identifier of the team to delete
</ParamField>

## Request

No request body required for this endpoint.

## Response

<ResponseField name="success" type="boolean">
  Indicates if the team was deleted successfully
</ResponseField>

<ResponseField name="data" type="object">
  Contains deletion confirmation information
</ResponseField>

<ResponseField name="data.team_id" type="string">
  ID of the deleted team
</ResponseField>

<ResponseField name="data.name" type="string">
  Name of the deleted team
</ResponseField>

<ResponseField name="data.deleted_at" type="string">
  ISO 8601 timestamp when the team was deleted
</ResponseField>

<ResponseField name="data.members_affected" type="integer">
  Number of members who were removed from the team
</ResponseField>

<ResponseField name="data.resources_affected" type="array">
  List of resources that were affected by the deletion
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable message about the operation result
</ResponseField>

<ResponseField name="timestamp" type="string">
  ISO 8601 timestamp of the response
</ResponseField>

## Usage Notes

<AccordionGroup>
  <Accordion title="Deletion Requirements" icon="shield">
    To delete a team, you must have:
    
    - Team management permissions
    - Admin role or team owner permissions
    - No active dependencies (platforms, clients, etc.)
    - All team members must be removed or reassigned
  </Accordion>
  
  <Accordion title="What Gets Deleted" icon="trash">
    When a team is deleted, the following are permanently removed:
    
    - Team configuration and settings
    - Team member associations
    - Team-specific permissions
    - Team history and logs
    - Associated platforms and clients (if not reassigned)
  </Accordion>
  
  <Accordion title="Dependencies Check" icon="link">
    Before deletion, the system checks for:
    
    - Active platforms associated with the team
    - API clients owned by the team
    - Pending invitations
    - Active projects or resources
  </Accordion>
  
  <Accordion title="Recovery" icon="exclamation-triangle">
    **Important**: Team deletion is irreversible:
    
    - No recovery mechanism available
    - All data is permanently removed
    - Consider archiving instead of deleting
    - Backup important data before deletion
  </Accordion>
</AccordionGroup>

    ```javascript
{`class SafeTeamDeletion {
  constructor(apiClient) {
    this.apiClient = apiClient;
  }
  
  async checkTeamDependencies(teamId) {
    try {
      // Check for associated platforms
      const platformsResponse = await this.apiClient.makeRequest(\`/org/platforms?teamId=\${teamId}\`);
      const platforms = await platformsResponse.json();
      
      // Check for associated clients
      const clientsResponse = await this.apiClient.makeRequest(\`/org/clients?teamId=\${teamId}\`);
      const clients = await clientsResponse.json();
      
      // Check team members
      const membersResponse = await this.apiClient.makeRequest(\`/org/team/members/\${teamId}\`);
      const members = await membersResponse.json();
      
      return {
        hasPlatforms: platforms.data.length > 0,
        hasClients: clients.data.length > 0,
        hasMembers: members.data.length > 0,
        platforms: platforms.data,
        clients: clients.data,
        members: members.data
      };
      
    } catch (error) {
      console.error('Error checking dependencies:', error);
      throw error;
    }
  }
  
  async deleteTeamSafely(teamId, options = {}) {
    const dependencies = await this.checkTeamDependencies(teamId);
    
    if (dependencies.hasPlatforms && !options.force) {
      throw new Error(\`Team has \${dependencies.platforms.length} associated platforms. Remove them first or use force option.\`);
    }
    
    if (dependencies.hasClients && !options.force) {
      throw new Error(\`Team has \${dependencies.clients.length} associated clients. Remove them first or use force option.\`);
    }
    
    if (dependencies.hasMembers && !options.force) {
      throw new Error(\`Team has \${dependencies.members.length} members. Remove them first or use force option.\`);
    }
    
    // Proceed with deletion
    const response = await this.apiClient.makeRequest(\`/org/team/\${teamId}\`, {
      method: 'DELETE'
    });
    
    if (!response.ok) {
      const error = await response.json();
      throw new Error(\`Failed to delete team: \${error.error.message}\`);
    }
    
    return await response.json();
  }
  
  async deleteTeamWithConfirmation(teamId, confirmationData) {
    // This would typically be called from a UI with user confirmation
    if (!confirmationData.confirmed) {
      throw new Error('Team deletion not confirmed');
    }
    
    if (!confirmationData.acknowledgeIrreversible) {
      throw new Error('User must acknowledge that deletion is irreversible');
    }
    
    return await this.deleteTeamSafely(teamId, { force: confirmationData.force });
  }
}
```

    ```javascript
      total_teams: teamIds.length,
      can_delete: [],
      has_dependencies: [],
      not_found: [],

    };

        }
      }
    }
    
    return report;
  }
```
