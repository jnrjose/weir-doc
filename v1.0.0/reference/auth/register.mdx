---
title: "Register User"
api: "POST /auth/register"
description: "Register a new user account with email verification"
---

Register a new user account in the Weir AI platform. This endpoint initiates the registration process and sends an OTP to the user's email for verification.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and is used for standard user registration (non-mobile). It requires email verification through OTP.
</Note>

## Authentication

This endpoint does not require authentication as it's used for new user registration.

## Request

<ParamField body="fullname" type="string" required>
  User's full name
  
  **Example**: "John Doe"
  
  **Constraints**: 
  - Minimum length: 2 characters
  - Maximum length: 100 characters
  - Must contain only letters, spaces, hyphens, and apostrophes
</ParamField>

<ParamField body="email" type="string" required>
  User's email address
  
  **Example**: "john.doe@example.com"
  
  **Constraints**: 
  - Must be a valid email format
  - Must be unique (not already registered)
  - Maximum length: 255 characters
</ParamField>

<ParamField body="password" type="string" required>
  User's password
  
  **Constraints**: 
  - Minimum length: 8 characters
  - Must contain at least one uppercase letter
  - Must contain at least one lowercase letter
  - Must contain at least one number
  - Must contain at least one special character
</ParamField>

<ParamField body="confirm_password" type="string" required>
  Password confirmation
  
  **Constraints**: 
  - Must match the password field exactly
</ParamField>

<ParamField body="terms_accepted" type="boolean" required>
  Indicates if user has accepted terms and conditions
  
  **Example**: true
  
  **Constraints**: 
  - Must be true to proceed with registration
</ParamField>

<ParamField body="marketing_consent" type="boolean" required="false">
  Indicates if user consents to marketing communications
  
  **Example**: false
  
  **Default**: false
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Indicates if the registration was initiated successfully
</ResponseField>

<ResponseField name="data" type="object">
  Contains registration information
</ResponseField>

<ResponseField name="data.registration_id" type="string">
  Unique identifier for tracking the registration process
</ResponseField>

<ResponseField name="data.email" type="string">
  Email address where OTP was sent
</ResponseField>

<ResponseField name="data.otp_expires_at" type="string">
  ISO 8601 timestamp when the OTP expires
</ResponseField>

<ResponseField name="data.status" type="string">
  Registration status (pending_verification)
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable message about the operation result
</ResponseField>

<ResponseField name="timestamp" type="string">
  ISO 8601 timestamp of the response
</ResponseField>

```python Python
print(response.json())
```

## Usage Notes

<AccordionGroup>
  <Accordion title="Password Requirements" icon="lock">
    Password must meet these criteria:
    
    - Minimum 8 characters
    - At least one uppercase letter (A-Z)
    - At least one lowercase letter (a-z)
    - At least one number (0-9)
    - At least one special character (!@#$%^&*)
    - Cannot be a common password
  </Accordion>
  
  <Accordion title="Email Verification" icon="mail">
    Registration process:
    
    - OTP is sent to the provided email address
    - OTP expires after 10 minutes
    - User must verify email before account is activated
    - Verification is required to complete registration
  </Accordion>
  
  <Accordion title="Terms and Conditions" icon="file-text">
    Legal requirements:
    
    - Terms and conditions must be accepted
    - Marketing consent is optional
    - User data is processed according to privacy policy
    - Account creation is subject to platform terms
  </Accordion>
</AccordionGroup>

## Integration Examples

<AccordionGroup>
  <Accordion title="Registration Form" icon="form">
    ```javascript
// Registration form handler
async function handleRegistration(formData) {
  try {
    const response = await fetch('/auth/register', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        fullname: formData.fullname,
        email: formData.email,
        password: formData.password,
        confirm_password: formData.confirmPassword,
        terms_accepted: formData.termsAccepted,
        marketing_consent: formData.marketingConsent
      })
    });
    
    const data = await response.json();
    
    if (data.success) {
      // Store registration ID for verification
      localStorage.setItem('registration_id', data.data.registration_id);
      localStorage.setItem('registration_email', data.data.email);
      
      // Redirect to verification page
      window.location.href = '/verify-email';
      
      return data;
    } else {
      // Handle validation errors
      displayValidationErrors(data.error.validation_errors);
      throw new Error(data.error.message);
    }
    
  } catch (error) {
    console.error('Registration failed:', error);
    throw error;
  }
}

function displayValidationErrors(errors) {
  errors.forEach(error => {
    const field = document.querySelector(`[name="${error.field}"]`);
    if (field) {
      field.classList.add('error');
      const errorMsg = document.createElement('div');
      errorMsg.className = 'error-message';
      errorMsg.textContent = error.message;
      field.parentNode.appendChild(errorMsg);
    }
  });
}
```
