---
title: "Get Platform Details"
api: "GET /platforms/{platformId}"
description: "Retrieve detailed information about a specific platform"
---

Retrieve comprehensive details about a specific platform including its capabilities, configuration options, pricing, and integration requirements. This endpoint provides in-depth information for platform evaluation and integration planning.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication. It's designed for platform evaluation and integration planning.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Path Parameters

<ParamField path="platformId" type="string" required>
  Unique identifier for the platform
  
  **Example**: "platform_1234567890abcdef"
</ParamField>

## Query Parameters

<ParamField query="include_pricing" type="boolean" required="false">
  Whether to include pricing information in the response
  
  **Example**: true
  
  **Default**: false
</ParamField>

<ParamField query="include_documentation" type="boolean" required="false">
  Whether to include documentation links and examples
  
  **Example**: true
  
  **Default**: true
</ParamField>

<ParamField query="include_integration" type="boolean" required="false">
  Whether to include integration requirements and setup guides
  
  **Example**: true
  
  **Default**: false
</ParamField>

    ```javascript
class PlatformEvaluator {
  constructor(apiClient) {
    this.apiClient = apiClient;
  }
  
  async evaluatePlatform(platformId, includePricing = true, includeIntegration = true) {
    try {
      const response = await this.apiClient.makeRequest(`/platforms/${platformId}`, {
        method: 'GET',
        params: {
          include_pricing: includePricing,
          include_integration: includeIntegration,
          include_documentation: true
        }
      });
      
      if (!response.ok) {
        throw new Error('Failed to get platform details');
      }
      
      const result = await response.json();
      return this.analyzePlatform(result.data);
      
    } catch (error) {
      console.error('Error evaluating platform:', error);
      throw error;
    }
  }
  
  analyzePlatform(platform) {
    const analysis = {
      id: platform.platform_id,
      name: platform.name,
      category: platform.category,
      status: platform.status,
      capabilities: platform.capabilities,
      integrationOptions: platform.integration_options,
      requirements: platform.requirements,
      pricing: platform.pricing,
      score: 0
    };
    
    // Calculate compatibility score
    analysis.score = this.calculateCompatibilityScore(platform);
    
    return analysis;
  }
  
  calculateCompatibilityScore(platform) {
    let score = 0;
    
    // Check integration options
    if (platform.integration_options.api) score += 25;
    if (platform.integration_options.sdk) score += 25;
    if (platform.integration_options.webhook) score += 15;
    
    // Check capabilities
    score += platform.capabilities.length * 5;
    
    // Check status
    if (platform.status === 'active') score += 20;
    
    return Math.min(score, 100);
  }
}
```

    ```python
import requests
from typing import Dict, List, Optional

class PlatformClient:
    def __init__(self, api_client):
        self.api_client = api_client
    
    def get_platform_details(self, 
                           platform_id: str,
                           include_pricing: bool = False,
                           include_documentation: bool = True,
                           include_integration: bool = False) -> Dict:
        """Get detailed platform information"""
        try:
            params = {
                "include_pricing": include_pricing,
                "include_documentation": include_documentation,
                "include_integration": include_integration
            }
            
            response = self.api_client.get(f'/platforms/{platform_id}', params=params)
            response.raise_for_status()
            
            return response.json()
            
        except requests.exceptions.RequestException as e:
            print(f"Failed to get platform details: {e}")
            raise
    
    def check_platform_compatibility(self, platform_id: str, user_plan: str) -> Dict:
        """Check if platform is compatible with user's plan"""
        result = self.get_platform_details(platform_id, include_pricing=True)
        platform = result.get("data", {})
        
        compatibility = {
            "compatible": False,
            "reason": "",
            "required_plan": platform.get("requirements", {}).get("min_plan", "free")
        }
        
        if user_plan == "enterprise" or user_plan == platform["required_plan"]:
            compatibility["compatible"] = True
        elif user_plan == "professional" and platform["required_plan"] in ["free", "standard", "professional"]:
            compatibility["compatible"] = True
        elif user_plan == "standard" and platform["required_plan"] in ["free", "standard"]:
            compatibility["compatible"] = True
        elif user_plan == "free" and platform["required_plan"] == "free":
            compatibility["compatible"] = True
        else:
            compatibility["reason"] = f"Requires {platform['required_plan']} plan or higher"
        
        return compatibility
    
    def get_platform_integration_guide(self, platform_id: str) -> Dict:
        """Get platform integration guide"""
        result = self.get_platform_details(platform_id, include_integration=True)
        platform = result.get("data", {})
        
        return {
            "setup_steps": platform.get("integration", {}).get("setup_steps", []),
            "configuration": platform.get("integration", {}).get("configuration", {}),
            "testing": platform.get("integration", {}).get("testing", {}),
            "documentation": platform.get("documentation", {})
        }
```
