---
title: "Get Log Details"
api: "GET /logs/{logId}"
description: "Retrieve detailed information for a specific log entry"
---

Retrieve comprehensive details for a specific log entry including metadata, context, and related information. This endpoint provides in-depth analysis capabilities for debugging and monitoring purposes.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for detailed log analysis and debugging.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Path Parameters

<ParamField path="logId" type="string" required>
  Unique identifier for the log entry
  
  **Example**: "log_1234567890abcdef"
</ParamField>

## Query Parameters

<ParamField query="include_context" type="boolean" required="false">
  Whether to include additional context information
  
  **Example**: true
  
  **Default**: true
</ParamField>

<ParamField query="include_related" type="boolean" required="false">
  Whether to include related log entries
  
  **Example**: true
  
  **Default**: false
</ParamField>

<ParamField query="include_stack_trace" type="boolean" required="false">
  Whether to include stack trace information (for error logs)
  
  **Example**: true
  
  **Default**: false
</ParamField>

class LogAnalyzer {
  constructor(apiClient) {
    this.apiClient = apiClient;
  }
  
  async getLogDetails(logId, includeContext = true, includeRelated = true) {
    try {
      const response = await this.apiClient.makeRequest(`/logs/${logId}`, {
        method: 'GET',
        params: {
          include_context: includeContext,
          include_related: includeRelated,
          include_stack_trace: true
        }
      });
      
      if (!response.ok) {
        throw new Error('Failed to get log details');
      }
      
      return await response.json();
      
    } catch (error) {
      console.error('Error getting log details:', error);
      throw error;
    }
  }
  
  async analyzeErrorLog(logId) {
    const result = await this.getLogDetails(logId, true, true);
    const log = result.data;
    
    return {
      logId: log.log_id,
      level: log.level,
      message: log.message,
      severity: log.severity_score,
      anomaly: log.anomaly_score,
      hasStackTrace: log.stack_trace && log.stack_trace.length > 0,
      relatedLogs: log.related_logs?.length || 0,
      isHighSeverity: log.severity_score >= 7,
      isAnomalous: log.anomaly_score >= 0.8
    };
  }
  
  async getErrorChain(logId) {
    const result = await this.getLogDetails(logId, true, true);
    const log = result.data;
    
    if (!log.related_logs) return [log];
    
    // Build error chain by analyzing related logs
    const errorChain = [log];
    const relatedLogs = log.related_logs;
    
    // Sort by timestamp to get chronological order
    relatedLogs.sort((a, b) => new Date(a.timestamp) - new Date(b.timestamp));
    
    // Add related logs to chain
    for (const relatedLog of relatedLogs) {
      if (['error', 'fatal'].includes(relatedLog.level)) {
        errorChain.push(relatedLog);
      }
    }
    
    return errorChain;
  }
}

import requests
from typing import Dict, List, Optional
from datetime import datetime

class LogMonitor:
    def __init__(self, api_client):
        self.api_client = api_client
    
    def get_log_details(self, 
                       log_id: str,
                       include_context: bool = True,
                       include_related: bool = True,
                       include_stack_trace: bool = False) -> Dict:
        """Get detailed log entry information"""
        try:
            params = {
                "include_context": include_context,
                "include_related": include_related,
                "include_stack_trace": include_stack_trace
            }
            
            response = self.api_client.get(f'/logs/{log_id}', params=params)
            response.raise_for_status()
            
            return response.json()
            
        except requests.exceptions.RequestException as e:
            print(f"Failed to get log details: {e}")
            raise
    
    def analyze_log_severity(self, log_id: str) -> Dict:
        """Analyze log severity and provide recommendations"""
        result = self.get_log_details(log_id, True, True, True)
        log = result.get("data", {})
        
        analysis = {
            "log_id": log_id,
            "severity_score": log.get("severity_score", 0),
            "anomaly_score": log.get("anomaly_score", 0),
            "level": log.get("level"),
            "recommendation": "",
            "priority": "low"
        }
        
        severity = log.get("severity_score", 0)
        anomaly = log.get("anomaly_score", 0)
        
        if severity >= 8 or anomaly >= 0.9:
            analysis["priority"] = "critical"
            analysis["recommendation"] = "Immediate attention required"
        elif severity >= 6 or anomaly >= 0.7:
            analysis["priority"] = "high"
            analysis["recommendation"] = "Investigate within 1 hour"
        elif severity >= 4 or anomaly >= 0.5:
            analysis["priority"] = "medium"
            analysis["recommendation"] = "Monitor and investigate if pattern continues"
        else:
            analysis["priority"] = "low"
            analysis["recommendation"] = "No immediate action required"
        
        return analysis
    
    def get_error_context(self, log_id: str) -> Dict:
        """Get comprehensive error context"""
        result = self.get_log_details(log_id, True, True, True)
        log = result.get("data", {})
        
        context = {
            "error_details": {
                "message": log.get("message"),
                "level": log.get("level"),
                "source": log.get("source"),
                "timestamp": log.get("timestamp")
            },
            "user_context": {
                "user_id": log.get("user_id"),
                "ip_address": log.get("ip_address"),
                "session_id": log.get("session_id")
            },
            "request_context": {
                "request_id": log.get("request_id"),
                "endpoint": log.get("metadata", {}).get("endpoint"),
                "method": log.get("metadata", {}).get("method"),
                "status_code": log.get("metadata", {}).get("status_code")
            },
            "technical_context": {
                "environment": log.get("context", {}).get("environment"),
                "version": log.get("context", {}).get("version"),
                "server": log.get("context", {}).get("server"),
                "region": log.get("context", {}).get("region")
            },
            "stack_trace": log.get("stack_trace", []),
            "related_logs": log.get("related_logs", [])
        }
        
        return context
    
    def detect_anomalies(self, log_id: str) -> List[Dict]:
        """Detect anomalies in log entry"""
        result = self.get_log_details(log_id, True, True)
        log = result.get("data", {})
        
        anomalies = []
        
        # Check anomaly score
        if log.get("anomaly_score", 0) >= 0.8:
            anomalies.append({
                "type": "high_anomaly_score",
                "description": "Log entry has high anomaly score",
                "score": log.get("anomaly_score")
            })
        
        # Check severity score
        if log.get("severity_score", 0) >= 8:
            anomalies.append({
                "type": "high_severity",
                "description": "Log entry has high severity score",
                "score": log.get("severity_score")
            })
        
        # Check for related errors
        related_logs = log.get("related_logs", [])
        error_count = sum(1 for rl in related_logs if rl.get("level") in ["error", "fatal"])
        if error_count >= 3:
            anomalies.append({
                "type": "error_chain",
                "description": "Multiple related errors detected",
                "count": error_count
            })
        
        return anomalies
