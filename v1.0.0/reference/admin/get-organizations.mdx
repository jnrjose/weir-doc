---
title: "Get Organizations"
api: "GET /admin/organizations"
description: "Retrieve list of all organizations in the system with pagination support"
---

Retrieve a paginated list of all organizations in the system. This endpoint provides comprehensive organization information including details, member counts, and status for administrative oversight.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for system-wide organization management and monitoring.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

<CodeBlock lang="bash">
Authorization: Bearer <your_jwt_token>
</CodeBlock>

## Query Parameters

<ParamField query="page" type="string" required="false">
  Page number for pagination (default: 1)
  
  **Example**: "1"
</ParamField>

<ParamField query="limit" type="string" required="false">
  Number of organizations per page (default: 20, max: 100)
  
  **Example**: "20"
</ParamField>

<ParamField query="status" type="string" required="false">
  Filter organizations by status
  
  **Options**: "active", "inactive", "suspended", "pending"
  
  **Example**: "active"
</ParamField>

<ParamField query="search" type="string" required="false">
  Search organizations by name or domain
  
  **Example**: "acme"
</ParamField>

<ParamField query="sort" type="string" required="false">
  Sort organizations by field
  
  **Options**: "name", "created_at", "updated_at", "member_count"
  
  **Default**: "created_at"
</ParamField>

<ParamField query="order" type="string" required="false">
  Sort order
  
  **Options**: "asc", "desc"
  
  **Default**: "desc"
</ParamField>

class AdminDashboard {
  constructor(apiClient) {
    this.apiClient = apiClient;
    this.organizations = [];
    this.currentPage = 1;
    this.pageSize = 20;
    this.filters = {
      status: 'active',
      search: '',
      sort: 'created_at',
      order: 'desc'
    };
  }
  
  async loadOrganizations(page = 1) {
    try {
      const params = new URLSearchParams({
        page: page.toString(),
        limit: this.pageSize.toString(),
        ...this.filters
      });
      
      const response = await this.apiClient.makeRequest(`/admin/organizations?${params}`);
      
      if (!response.ok) {
        throw new Error('Failed to load organizations');
      }
      
      const data = await response.json();
      this.organizations = data.data;
      this.currentPage = page;
      
      this.displayOrganizations();
      this.updatePagination(data.pagination);
      
      return data;
      
    } catch (error) {
      console.error('Error loading organizations:', error);
      this.displayError('Failed to load organizations');
      throw error;
    }
  }
  
  displayOrganizations() {
    const container = document.getElementById('organizations-list');
    container.innerHTML = '';
    
    this.organizations.forEach(org => {
      const orgElement = this.createOrganizationElement(org);
      container.appendChild(orgElement);
    });
  }
  
  createOrganizationElement(org) {
    const div = document.createElement('div');
    div.className = 'organization-card';
    div.innerHTML = `
      <div class="org-header">
        <h3>${org.name}</h3>
        <span class="status status-${org.status}">${org.status}</span>
      </div>
      <div class="org-details">
        <p><strong>Domain:</strong> ${org.domain}</p>
        <p><strong>Industry:</strong> ${org.industry}</p>
        <p><strong>Size:</strong> ${org.company_size}</p>
        <p><strong>Members:</strong> ${org.member_count}</p>
        <p><strong>Teams:</strong> ${org.team_count}</p>
        <p><strong>Created:</strong> ${new Date(org.created_at).toLocaleDateString()}</p>
        <p><strong>Last Activity:</strong> ${new Date(org.last_activity).toLocaleDateString()}</p>
      </div>
      <div class="org-subscription">
        <p><strong>Plan:</strong> ${org.subscription.plan}</p>
        <p><strong>Status:</strong> ${org.subscription.status}</p>
        <p><strong>Expires:</strong> ${new Date(org.subscription.expires_at).toLocaleDateString()}</p>
      </div>
      <div class="org-actions">
        <button onclick="viewOrganization('${org.organization_id}')">View Details</button>
        <button onclick="editOrganization('${org.organization_id}')">Edit</button>
        <button onclick="manageOrganization('${org.organization_id}')">Manage</button>
      </div>
    `;
    
    return div;
  }
  
  updatePagination(pagination) {
    const paginationElement = document.getElementById('pagination');
    paginationElement.innerHTML = `
      <div class="pagination-info">
        Showing ${(pagination.page - 1) * pagination.limit + 1} to ${Math.min(pagination.page * pagination.limit, pagination.total)} of ${pagination.total} organizations
      </div>
      <div class="pagination-controls">
        <button ${!pagination.has_prev ? 'disabled' : ''} onclick="loadPage(${pagination.page - 1})">Previous</button>
        <span>Page ${pagination.page} of ${pagination.total_pages}</span>
        <button ${!pagination.has_next ? 'disabled' : ''} onclick="loadPage(${pagination.page + 1})">Next</button>
      </div>
    `;
  }
  
  setFilter(filterName, value) {
    this.filters[filterName] = value;
    this.loadOrganizations(1); // Reset to first page
  }
  
  async loadPage(page) {
    await this.loadOrganizations(page);
  }
}

class OrganizationAnalytics {
  constructor(apiClient) {
    this.apiClient = apiClient;
  }
  
  async getAllOrganizations() {
    const allOrganizations = [];
    let page = 1;
    let hasMore = true;
    
    while (hasMore) {
      const response = await this.apiClient.makeRequest(`/admin/organizations?page=${page}&limit=100`);
      const data = await response.json();
      
      allOrganizations.push(...data.data);
      hasMore = data.pagination.has_next;
      page++;
    }
    
    return allOrganizations;
  }
  
  async generateAnalyticsReport() {
    const organizations = await this.getAllOrganizations();
    
    const stats = {
      total_organizations: organizations.length,
      active_organizations: organizations.filter(org => org.status === 'active').length,
      total_members: organizations.reduce((sum, org) => sum + org.member_count, 0),
      total_teams: organizations.reduce((sum, org) => sum + org.team_count, 0),
      average_org_size: organizations.length > 0 ? organizations.reduce((sum, org) => sum + org.member_count, 0) / organizations.length : 0,
      industry_breakdown: this.groupByIndustry(organizations),
      size_breakdown: this.groupBySize(organizations),
      subscription_breakdown: this.groupBySubscription(organizations),
      growth_by_month: this.groupByMonth(organizations)
    };
    
    return stats;
  }
  
  groupByIndustry(organizations) {
    const industries = {};
    organizations.forEach(org => {
      industries[org.industry] = (industries[org.industry] || 0) + 1;
    });
    return industries;
  }
  
  groupBySize(organizations) {
    const sizes = {};
    organizations.forEach(org => {
      sizes[org.company_size] = (sizes[org.company_size] || 0) + 1;
    });
    return sizes;
  }
  
  groupBySubscription(organizations) {
    const subscriptions = {};
    organizations.forEach(org => {
      const plan = org.subscription.plan;
      subscriptions[plan] = (subscriptions[plan] || 0) + 1;
    });
    return subscriptions;
  }
  
  groupByMonth(organizations) {
    const months = {};
    organizations.forEach(org => {
      const month = new Date(org.created_at).toISOString().substring(0, 7); // YYYY-MM
      months[month] = (months[month] || 0) + 1;
    });
    return months;
  }
  
  async getOrganizationGrowthTrend() {
    const organizations = await this.getAllOrganizations();
    const sortedOrgs = organizations.sort((a, b) => new Date(a.created_at) - new Date(b.created_at));
    
    const growthData = [];
    let cumulativeCount = 0;
    
    sortedOrgs.forEach(org => {
      cumulativeCount++;
      growthData.push({
        date: org.created_at,
        cumulative_organizations: cumulativeCount,
        organization_name: org.name
      });
    });
    
    return growthData;
  }
}
