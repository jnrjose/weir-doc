---
title: "Get Organizations"
api: "GET /admin/organizations"
description: "Retrieve list of all organizations in the system for administrative oversight"
---

Retrieve a comprehensive list of all organizations in the Weir AI platform. This endpoint provides administrative oversight capabilities for system administrators.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires admin-level Bearer Token authentication. It provides system-wide organization management capabilities.
</Note>

## Authentication

This endpoint requires Admin Bearer Token authentication with elevated privileges. Include your admin JWT token in the Authorization header:

<CodeBlock lang="bash">
{`Authorization: Bearer <your_admin_jwt_token>`}
</CodeBlock>

## Request

<ParamField query="page" type="string" required="false">
  Page number for pagination (default: 1)
</ParamField>

<ParamField query="limit" type="string" required="false">
  Number of organizations per page (default: 20, max: 100)
</ParamField>

<ParamField query="search" type="string" required="false">
  Search term to filter organizations by name, email, or domain
</ParamField>

<ParamField query="status" type="string" required="false">
  Filter by organization status (active, inactive, suspended, pending)
</ParamField>

<ParamField query="industry" type="string" required="false">
  Filter by industry type
</ParamField>

<ParamField query="company_size" type="string" required="false">
  Filter by company size (SMALL, MEDIUM, LARGE, ENTERPRISE)
</ParamField>

<ParamField query="created_after" type="string" required="false">
  Filter organizations created after this date (ISO 8601 format)
</ParamField>

<ParamField query="created_before" type="string" required="false">
  Filter organizations created before this date (ISO 8601 format)
</ParamField>

<ParamField query="sort_by" type="string" required="false">
  Sort field (name, created_at, updated_at, member_count, platform_count)
</ParamField>

<ParamField query="sort_order" type="string" required="false">
  Sort order (asc, desc) - default: desc
</ParamField>

    try {
      const queryParams = new URLSearchParams({

{`class OrganizationAnalytics {
  constructor(apiClient) {
    this.apiClient = apiClient;
  }
  
  async getOrganizationStats() {
    try {
      // Get all organizations (with pagination)
      const allOrgs = await this.getAllOrganizations();
      
      const stats = {
        total: allOrgs.length,
        byStatus: this.groupBy(allOrgs, 'status'),
        byIndustry: this.groupBy(allOrgs, 'industry'),
        bySize: this.groupBy(allOrgs, 'company_size'),
        byBillingStatus: this.groupBy(allOrgs, 'billing_status'),
        averageMembers: this.calculateAverage(allOrgs, 'member_count'),
        averageTeams: this.calculateAverage(allOrgs, 'team_count'),
        averagePlatforms: this.calculateAverage(allOrgs, 'platform_count'),
        recentSignups: this.getRecentSignups(allOrgs, 30), // Last 30 days
        topIndustries: this.getTopValues(allOrgs, 'industry', 5)
      };
      
      return stats;
      
    } catch (error) {
      console.error('Error calculating organization stats:', error);
      throw error;
    }
  }
  
  async getAllOrganizations() {
    const allOrgs = [];
    let page = 1;
    let hasMore = true;
    
    while (hasMore) {
      const response = await this.apiClient.makeRequest(\`/admin/organizations?page=\${page}&limit=100\`);
      const data = await response.json();
      
      allOrgs.push(...data.data);
      hasMore = data.pagination.has_next;
      page++;
    }
    
    return allOrgs;
  }
  
  groupBy(array, key) {
    return array.reduce((groups, item) => {
      const value = item[key];
      groups[value] = (groups[value] || 0) + 1;
      return groups;
    }, {});
  }
  
  calculateAverage(array, key) {
    const values = array.map(item => item[key]).filter(val => val !== null && val !== undefined);
    return values.length > 0 ? values.reduce((sum, val) => sum + val, 0) / values.length : 0;
  }
  
  getRecentSignups(organizations, days) {
    const cutoffDate = new Date();
    cutoffDate.setDate(cutoffDate.getDate() - days);
    
    return organizations.filter(org => 
      new Date(org.created_at) >= cutoffDate
    );
  }
  
  getTopValues(array, key, limit) {
    const counts = this.groupBy(array, key);
    return Object.entries(counts)
      .sort(([,a], [,b]) => b - a)
      .slice(0, limit)
      .map(([value, count]) => ({ value, count }));
  }
  
  async generateReport() {
    const stats = await this.getOrganizationStats();
    
    const report = {
      generated_at: new Date().toISOString(),
      summary: {
        total_organizations: stats.total,
        active_organizations: stats.byStatus.active || 0,
        trial_organizations: stats.byBillingStatus.trial || 0,
        average_members_per_org: Math.round(stats.averageMembers),
        recent_signups: stats.recentSignups.length
      },
      breakdown: {
        by_status: stats.byStatus,
        by_industry: stats.byIndustry,
        by_company_size: stats.bySize,
        by_billing_status: stats.byBillingStatus
      },
      insights: this.generateInsights(stats)
    };
    
    return report;
  }
  
  generateInsights(stats) {
    const insights = [];
    
    if (stats.byBillingStatus.trial > stats.byBillingStatus.active * 0.1) {
      insights.push({
        type: 'warning',
        message: \`High trial-to-active ratio: \${stats.byBillingStatus.trial} trials vs \${stats.byBillingStatus.active} active\`
      });
    }
    
    if (stats.recentSignups.length > stats.total * 0.05) {
      insights.push({
        type: 'success',
        message: \`Strong growth: \${stats.recentSignups.length} new organizations in the last 30 days\`
      });
    }
    
    const topIndustry = stats.topIndustries[0];
    if (topIndustry) {
      insights.push({
        type: 'info',
        message: \`Most popular industry: \${topIndustry.value} (\${topIndustry.count} organizations)\`
      });
    }
    
    return insights;
  }
}`}
