---
title: "Get Log Entries"
api: "GET /logs"
description: "Retrieve list of log entries with filtering and pagination"
---

Retrieve a list of log entries with filtering, pagination, and sorting capabilities. This endpoint allows admins to view and analyze log entries for monitoring and troubleshooting purposes.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for log management and monitoring within the Weir AI platform.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Query Parameters

<ParamField query="page" type="string" required="false">
  Page number for pagination
  
  **Example**: "1"
  
  **Default**: "1"
</ParamField>

<ParamField query="limit" type="string" required="false">
  Number of log entries to return per page
  
  **Example**: "50"
  
  **Default**: "20"
</ParamField>

<ParamField query="level" type="string" required="false">
  Filter by log level
  
  **Options**: "debug", "info", "warning", "error", "critical"
  
  **Example**: "error"
</ParamField>

<ParamField query="source" type="string" required="false">
  Filter by source component
  
  **Example**: "auth-service"
</ParamField>

<ParamField query="category" type="string" required="false">
  Filter by log category
  
  **Options**: "authentication", "authorization", "system", "database", "api", "security", "performance", "business", "audit"
  
  **Example**: "authentication"
</ParamField>

<ParamField query="user_id" type="string" required="false">
  Filter by user ID
  
  **Example**: "user_1234567890abcdef"
</ParamField>

<ParamField query="session_id" type="string" required="false">
  Filter by session ID
  
  **Example**: "session_1234567890abcdef"
</ParamField>

<ParamField query="request_id" type="string" required="false">
  Filter by request ID
  
  **Example**: "req_1234567890abcdef"
</ParamField>

<ParamField query="ip_address" type="string" required="false">
  Filter by IP address
  
  **Example**: "192.168.1.100"
</ParamField>

<ParamField query="environment" type="string" required="false">
  Filter by environment
  
  **Options**: "production", "staging", "development", "testing"
  
  **Example**: "production"
</ParamField>

<ParamField query="region" type="string" required="false">
  Filter by region
  
  **Example**: "us-east-1"
</ParamField>

<ParamField query="tag" type="string" required="false">
  Filter by tag
  
  **Example**: "security"
  
  **Note**: Can be used multiple times for multiple tags
</ParamField>

<ParamField query="start_date" type="string" required="false">
  Start date for filtering
  
  **Example**: "2024-01-01T00:00:00Z"
  
  **Format**: ISO 8601 timestamp
</ParamField>

<ParamField query="end_date" type="string" required="false">
  End date for filtering
  
  **Example**: "2024-01-31T23:59:59Z"
  
  **Format**: ISO 8601 timestamp
</ParamField>

<ParamField query="sort_by" type="string" required="false">
  Sort by field
  
  **Options**: "timestamp", "level", "source", "category", "message"
  
  **Example**: "timestamp"
  
  **Default**: "timestamp"
</ParamField>

<ParamField query="sort_order" type="string" required="false">
  Sort order
  
  **Options**: "asc", "desc"
  
  **Example**: "desc"
  
  **Default**: "desc"
</ParamField>

<ParamField query="include_metadata" type="string" required="false">
  Whether to include metadata in response
  
  **Options**: "true", "false"
  
  **Example**: "true"
  
  **Default**: "false"
</ParamField>

<ParamField query="include_tags" type="string" required="false">
  Whether to include tags in response
  
  **Options**: "true", "false"
  
  **Example**: "true"
  
  **Default**: "false"
</ParamField>

<ParamField query="search" type="string" required="false">
  Search term for message content
  
  **Example**: "authentication failed"
  
  **Note**: Searches within log messages
</ParamField>

    class LogRetrievalService {
      constructor(apiClient) {
        this.apiClient = apiClient;
      }
      
      async getLogs(options = {}) {
        try {
          const params = new URLSearchParams();
          
          if (options.page) params.append('page', options.page);
          if (options.limit) params.append('limit', options.limit);
          if (options.level) params.append('level', options.level);
          if (options.source) params.append('source', options.source);
          if (options.category) params.append('category', options.category);
          if (options.userId) params.append('user_id', options.userId);
          if (options.sessionId) params.append('session_id', options.sessionId);
          if (options.requestId) params.append('request_id', options.requestId);
          if (options.ipAddress) params.append('ip_address', options.ipAddress);
          if (options.environment) params.append('environment', options.environment);
          if (options.region) params.append('region', options.region);
          if (options.tags) {
            options.tags.forEach(tag => params.append('tag', tag));
          }
          if (options.startDate) params.append('start_date', options.startDate);
          if (options.endDate) params.append('end_date', options.endDate);
          if (options.sortBy) params.append('sort_by', options.sortBy);
          if (options.sortOrder) params.append('sort_order', options.sortOrder);
          if (options.includeMetadata) params.append('include_metadata', options.includeMetadata);
          if (options.includeTags) params.append('include_tags', options.includeTags);
          if (options.search) params.append('search', options.search);
          
          const response = await this.apiClient.makeRequest(
            `/logs?${params.toString()}`
          );
          
          if (!response.ok) {
            throw new Error('Failed to get logs');
          }
          
          return await response.json();
          
        } catch (error) {
          console.error('Error getting logs:', error);
          throw error;
        }
      }
      
      async getErrors(options = {}) {
        return this.getLogs({
          level: 'error',
          ...options
        });
      }
      
      async getWarnings(options = {}) {
        return this.getLogs({
          level: 'warning',
          ...options
        });
      }
      
      async getCriticalLogs(options = {}) {
        return this.getLogs({
          level: 'critical',
          ...options
        });
      }
      
      async getLogsBySource(source, options = {}) {
        return this.getLogs({
          source: source,
          ...options
        });
      }
      
      async getLogsByCategory(category, options = {}) {
        return this.getLogs({
          category: category,
          ...options
        });
      }
      
      async getLogsByUser(userId, options = {}) {
        return this.getLogs({
          userId: userId,
          ...options
        });
      }
      
      async getLogsByDateRange(startDate, endDate, options = {}) {
        return this.getLogs({
          startDate: startDate,
          endDate: endDate,
          ...options
        });
      }
      
      async getRecentLogs(hours = 24, limit = 100) {
        const endDate = new Date().toISOString();
        const startDate = new Date(Date.now() - hours * 60 * 60 * 1000).toISOString();
        
        return this.getLogs({
          startDate: startDate,
          endDate: endDate,
          limit: limit,
          sortBy: 'timestamp',
          sortOrder: 'desc'
        });
      }
      
      async getLogsByEnvironment(environment, options = {}) {
        return this.getLogs({
          environment: environment,
          ...options
        });
      }
      
      async getLogsByRegion(region, options = {}) {
        return this.getLogs({
          region: region,
          ...options
        });
      }
      
      async getLogsByTags(tags, options = {}) {
        return this.getLogs({
          tags: tags,
          ...options
        });
      }
      
      async searchLogs(searchTerm, options = {}) {
        return this.getLogs({
          search: searchTerm,
          ...options
        });
      }
      
      async getLogsWithMetadata(options = {}) {
        return this.getLogs({
          includeMetadata: true,
          includeTags: true,
          ...options
        });
      }
      
      async getLogsByIP(ipAddress, options = {}) {
        return this.getLogs({
          ipAddress: ipAddress,
          ...options
        });
      }
      
      async getLogsBySession(sessionId, options = {}) {
        return this.getLogs({
          sessionId: sessionId,
          ...options
        });
      }
      
      async getLogsByRequest(requestId, options = {}) {
        return this.getLogs({
          requestId: requestId,
          ...options
        });
      }
      
      async getLogStatistics(options = {}) {
        const result = await this.getLogs({
          limit: 1,
          ...options
        });
        
        return result.summary;
      }
    }
    
    // Usage example

    console.log(`Found ${recentErrors.data.length} recent errors`);

    console.log(`Found ${authLogs.data.length} auth service logs`);

    });
    console.log(`Found ${securityLogs.data.length} security logs`);

      limit: 100
    });
    console.log(`Found ${userLogs.data.length} user logs`);
    
    // Get recent logs
    const recentLogs = await logService.getRecentLogs(24, 200);
    console.log(`Found ${recentLogs.data.length} recent logs`);
    
    // Get logs with metadata
    const detailedLogs = await logService.getLogsWithMetadata({
      level: 'error',
      limit: 50
    });
    console.log(`Found ${detailedLogs.data.length} detailed error logs`);
    
    // Get log statistics
    const stats = await logService.getLogStatistics();
