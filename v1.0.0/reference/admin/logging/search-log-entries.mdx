---
title: "Search Log Entries"
api: "POST /logs/search"
description: "Search log entries with advanced filtering and query capabilities"
---

Search log entries with advanced filtering, query capabilities, and full-text search. This endpoint allows admins to find specific log entries based on various criteria and search terms.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for log analysis and monitoring within the Weir AI platform.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Request

<ParamField body="query" type="string" required="false">
  Full-text search query
  
  **Example**: "authentication failed"
  
  **Note**: Searches across message, source, and metadata fields
</ParamField>

<ParamField body="filters" type="object" required="false">
  Advanced filters for log entries
</ParamField>

<ParamField body="filters.level" type="array" required="false">
  Filter by log levels
  
  **Example**: ["error", "critical"]
  
  **Options**: "debug", "info", "warning", "error", "critical"
</ParamField>

<ParamField body="filters.source" type="array" required="false">
  Filter by source components
  
  **Example**: ["auth-service", "api-gateway"]
</ParamField>

<ParamField body="filters.category" type="array" required="false">
  Filter by log categories
  
  **Example**: ["authentication", "security"]
  
  **Options**: "authentication", "authorization", "system", "database", "api", "security", "performance", "business", "audit"
</ParamField>

<ParamField body="filters.user_id" type="array" required="false">
  Filter by user IDs
  
  **Example**: ["user_1234567890abcdef"]
</ParamField>

<ParamField body="filters.session_id" type="array" required="false">
  Filter by session IDs
  
  **Example**: ["session_1234567890abcdef"]
</ParamField>

<ParamField body="filters.request_id" type="array" required="false">
  Filter by request IDs
  
  **Example**: ["req_1234567890abcdef"]
</ParamField>

<ParamField body="filters.ip_address" type="array" required="false">
  Filter by IP addresses
  
  **Example**: ["192.168.1.100", "10.0.0.1"]
</ParamField>

<ParamField body="filters.environment" type="array" required="false">
  Filter by environments
  
  **Example**: ["production", "staging"]
  
  **Options**: "production", "staging", "development", "testing"
</ParamField>

<ParamField body="filters.region" type="array" required="false">
  Filter by regions
  
  **Example**: ["us-east-1", "us-west-2"]
</ParamField>

<ParamField body="filters.tags" type="array" required="false">
  Filter by tags
  
  **Example**: ["security", "authentication"]
</ParamField>

<ParamField body="filters.metadata" type="object" required="false">
  Filter by metadata fields
  
  **Example**: {"status_code": 200, "method": "GET"}
  
  **Note**: Key-value pairs for metadata filtering
</ParamField>

<ParamField body="date_range" type="object" required="false">
  Date range filter
</ParamField>

<ParamField body="date_range.start" type="string" required="false">
  Start date for search
  
  **Example**: "2024-01-01T00:00:00Z"
  
  **Format**: ISO 8601 timestamp
</ParamField>

<ParamField body="date_range.end" type="string" required="false">
  End date for search
  
  **Example**: "2024-01-31T23:59:59Z"
  
  **Format**: ISO 8601 timestamp
</ParamField>

<ParamField body="pagination" type="object" required="false">
  Pagination settings
</ParamField>

<ParamField body="pagination.page" type="integer" required="false">
  Page number
  
  **Example**: 1
  
  **Default**: 1
  
  **Constraints**: Must be greater than 0
</ParamField>

<ParamField body="pagination.limit" type="integer" required="false">
  Number of results per page
  
  **Example**: 50
  
  **Default**: 20
  
  **Constraints**: Must be between 1 and 1000
</ParamField>

<ParamField body="sort" type="object" required="false">
  Sort settings
</ParamField>

<ParamField body="sort.field" type="string" required="false">
  Field to sort by
  
  **Options**: "timestamp", "level", "source", "category", "message"
  
  **Example**: "timestamp"
  
  **Default**: "timestamp"
</ParamField>

<ParamField body="sort.order" type="string" required="false">
  Sort order
  
  **Options**: "asc", "desc"
  
  **Example**: "desc"
  
  **Default**: "desc"
</ParamField>

<ParamField body="aggregations" type="array" required="false">
  Aggregation settings for analytics
  
  **Example**: ["level", "source", "category"]
  
  **Options**: "level", "source", "category", "environment", "region", "user_id"
</ParamField>

<ParamField body="highlight" type="boolean" required="false">
  Whether to highlight search terms in results
  
  **Example**: true
  
  **Default**: false
</ParamField>

## Code Examples

<AccordionGroup>
  <Accordion title="Log Search Service" icon="cog">
    ```javascript
    class LogSearchService {
      constructor(apiClient) {
        this.apiClient = apiClient;
      }
      
      async searchLogs(searchOptions = {}) {
        try {
          const response = await this.apiClient.makeRequest('/logs/search', {
            method: 'POST',
            body: JSON.stringify(searchOptions)
          });
          
          if (!response.ok) {
            throw new Error('Failed to search logs');
          }
          
          return await response.json();
          
        } catch (error) {
          console.error('Error searching logs:', error);
          throw error;
        }
      }
      
      async searchByLevel(levels, options = {}) {
        return this.searchLogs({
          filters: {
            level: levels,
            ...options.filters
          },
          ...options
        });
      }
      
      async searchBySource(sources, options = {}) {
        return this.searchLogs({
          filters: {
            source: sources,
            ...options.filters
          },
          ...options
        });
      }
      
      async searchByCategory(categories, options = {}) {
        return this.searchLogs({
          filters: {
            category: categories,
            ...options.filters
          },
          ...options
        });
      }
      
      async searchByUser(userIds, options = {}) {
        return this.searchLogs({
          filters: {
            user_id: userIds,
            ...options.filters
          },
          ...options
        });
      }
      
      async searchByDateRange(startDate, endDate, options = {}) {
        return this.searchLogs({
          date_range: {
            start: startDate,
            end: endDate
          },
          ...options
        });
      }
      
      async searchByQuery(query, options = {}) {
        return this.searchLogs({
          query: query,
          highlight: true,
          ...options
        });
      }
      
      async searchErrors(options = {}) {
        return this.searchByLevel(['error', 'critical'], options);
      }
      
      async searchSecurityEvents(options = {}) {
        return this.searchLogs({
          filters: {
            category: ['security', 'authentication'],
            tags: ['security'],
            ...options.filters
          },
          ...options
        });
      }
      
      async searchApiRequests(options = {}) {
        return this.searchLogs({
          filters: {
            category: ['api'],
            source: ['api-gateway'],
            ...options.filters
          },
          ...options
        });
      }
      
      async searchByIP(ipAddresses, options = {}) {
        return this.searchLogs({
          filters: {
            ip_address: ipAddresses,
            ...options.filters
          },
          ...options
        });
      }
      
      async searchWithAggregations(aggregations, options = {}) {
        return this.searchLogs({
          aggregations: aggregations,
          ...options
        });
      }
      
      async getLogStatistics(dateRange, aggregations = ['level', 'source', 'category']) {
        return this.searchLogs({
          date_range: dateRange,
          aggregations: aggregations,
          pagination: {
            page: 1,
            limit: 1
          }
        });
      }
      
      async searchRecentErrors(hours = 24, limit = 100) {
        const endDate = new Date().toISOString();
        const startDate = new Date(Date.now() - hours * 60 * 60 * 1000).toISOString();
        
        return this.searchErrors({
          date_range: {
            start: startDate,
            end: endDate
          },
          pagination: {
            page: 1,
            limit: limit
          },
          sort: {
            field: 'timestamp',
            order: 'desc'
          }
        });
      }
      
      async searchUserActivity(userId, dateRange, options = {}) {
        return this.searchByUser([userId], {
          date_range: dateRange,
          ...options
        });
      }
      
      async searchFailedLogins(dateRange, options = {}) {
        return this.searchLogs({
          query: 'authentication failed',
          filters: {
            category: ['authentication', 'security'],
            tags: ['authentication', 'failure'],
            ...options.filters
          },
          date_range: dateRange,
          ...options
        });
      }
    }
    
    // Usage example
    const logSearchService = new LogSearchService(apiClient);
    
    // Search for errors
    const errors = await logSearchService.searchErrors({
      pagination: { page: 1, limit: 50 },
      sort: { field: 'timestamp', order: 'desc' }
    });
    console.log(`Found ${errors.data.pagination.total} errors`);
    
    // Search security events
    const securityEvents = await logSearchService.searchSecurityEvents({
      date_range: {
        start: '2024-01-01T00:00:00Z',
        end: '2024-01-31T23:59:59Z'
      }
    });
    console.log(`Found ${securityEvents.data.pagination.total} security events`);
    
    // Search by query
    const queryResults = await logSearchService.searchByQuery('database connection failed');
    console.log(`Found ${queryResults.data.pagination.total} matching logs`);
    
    // Get log statistics
    const stats = await logSearchService.getLogStatistics({
      start: '2024-01-01T00:00:00Z',
      end: '2024-01-31T23:59:59Z'
    });
    console.log('Log statistics:', stats.data.aggregations);
    
    // Search recent errors
    const recentErrors = await logSearchService.searchRecentErrors(24, 100);
    console.log(`Found ${recentErrors.data.results.length} recent errors`);
    
    // Search user activity
    const userActivity = await logSearchService.searchUserActivity(
      'user_1234567890abcdef',
      {
        start: '2024-01-01T00:00:00Z',
        end: '2024-01-31T23:59:59Z'
      }
    );
    console.log(`Found ${userActivity.data.pagination.total} user activities`);
    ```
