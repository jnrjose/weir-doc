---
title: "Admin Logout"
api: "POST /auth/logout"
description: "Logout and invalidate the current admin session"
---

Logout from the admin panel and invalidate the current session. This endpoint ensures that the refresh token is properly revoked and the session is terminated securely.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for secure session termination and token revocation.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <admin_jwt_token>
```

## Request

<ParamField body="refreshToken" type="string" required>
  The refresh token to invalidate
  
  **Example**: "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9..."
  
  **Constraints**: 
  - Must be a valid refresh token
  - Must be associated with the current session
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Indicates if the logout was successful
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable message about the logout result
</ResponseField>

<ResponseField name="timestamp" type="string">
  ISO 8601 timestamp of the response
</ResponseField>

## Usage Notes

<AccordionGroup>
  <Accordion title="Session Security" icon="shield">
    - Always call logout when the user session ends
    - Invalidate both access and refresh tokens
    - Clear all stored authentication data
    - Redirect user to login page after logout
  </Accordion>
  
  <Accordion title="Token Revocation" icon="key">
    The logout process:
    
    - Invalidates the refresh token immediately
    - Marks the access token as revoked
    - Prevents further API calls with the same tokens
    - Logs the logout event for security auditing
  </Accordion>
  
  <Accordion title="Client Cleanup" icon="broom">
    After successful logout:
    
    - Clear all stored tokens from client storage
    - Remove authentication headers from future requests
    - Reset application state to unauthenticated
    - Clear any cached user data
  </Accordion>
  
  <Accordion title="Error Handling" icon="exclamation-triangle">
    Handle logout errors gracefully:
    
    - If logout fails, still clear local tokens
    - Log the error for debugging
    - Proceed with local cleanup regardless of API response
    - Consider the user logged out locally
  </Accordion>
</AccordionGroup>

{`class WeirAdminClient {
  constructor(baseURL) {
    this.baseURL = baseURL;
    this.accessToken = null;
    this.refreshToken = null;
    this.tokenExpiry = null;
  }
  
  async logout() {
    try {
      // Attempt to logout on server
      if (this.refreshToken) {
        await this.makeRequest('/auth/logout', {
          method: 'POST',
          body: JSON.stringify({ refreshToken: this.refreshToken })
        });
      }
    } catch (error) {
      console.warn('Server logout failed:', error.message);
      // Continue with local cleanup even if server logout fails
    } finally {
      // Always clear local tokens
      this.clearTokens();
      this.redirectToLogin();
    }
  }
  
  clearTokens() {
    this.accessToken = null;
    this.refreshToken = null;
    this.tokenExpiry = null;
    
    // Clear from localStorage/sessionStorage
    localStorage.removeItem('weir_access_token');
    localStorage.removeItem('weir_refresh_token');
    localStorage.removeItem('weir_token_expiry');
  }
  
  redirectToLogin() {
    // Redirect to login page
    window.location.href = '/login';
  }
  
  async makeRequest(endpoint, options = {}) {
    if (!this.accessToken || Date.now() >= this.tokenExpiry) {
      await this.refreshAccessToken();
    }
    
    return fetch(\`\${this.baseURL}\${endpoint}\`, {
      ...options,
      headers: {
        'Authorization': \`Bearer \${this.accessToken}\`,
        'Content-Type': 'application/json',
        ...options.headers
      }
    });
  }
}

{`class AutoLogoutManager {
  constructor(adminClient, timeoutMinutes = 30) {
    this.adminClient = adminClient;
    this.timeoutMinutes = timeoutMinutes;

    this.setupActivityListeners();

    // Reset timeout on user activity
    const events = ['mousedown', 'mousemove', 'keypress', 'scroll', 'touchstart'];
    
    events.forEach(event => {
      document.addEventListener(event, () => {
        this.resetTimeout();
      }, true);
    });
    
    // Start initial timeout
    this.resetTimeout();
  }
  
  resetTimeout() {
    if (this.timeoutId) {
      clearTimeout(this.timeoutId);
    }
    
    this.timeoutId = setTimeout(() => {
      this.performLogout();
    }, this.timeoutMinutes * 60 * 1000);
  }
  
  async performLogout() {
    console.log('Auto-logout triggered due to inactivity');
    
    // Show warning before logout
    const shouldLogout = confirm(
      \`You have been inactive for \${this.timeoutMinutes} minutes. Do you want to stay logged in?\`
    );
    
    if (!shouldLogout) {
      await this.adminClient.logout();
    } else {
      // Reset timeout if user chooses to stay logged in
      this.resetTimeout();
    }
  }
  
  destroy() {
    if (this.timeoutId) {
      clearTimeout(this.timeoutId);
    }
  }
}
