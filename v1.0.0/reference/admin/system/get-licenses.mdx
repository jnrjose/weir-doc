---
title: "Get Admin Licenses"
api: "GET /admin/licenses"
description: "Retrieve system license information and usage statistics"
---

Retrieve comprehensive information about system licenses, usage statistics, and compliance status. This endpoint provides admin-level visibility into license management and system capacity.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for system administration and license management within the Weir AI platform.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Query Parameters

<ParamField query="include_usage" type="string" required="false">
  Whether to include usage statistics
  
  **Options**: "true", "false"
  
  **Example**: "true"
  
  **Default**: "false"
  
  **Note**: Setting to true will include detailed usage metrics
</ParamField>

<ParamField query="include_compliance" type="string" required="false">
  Whether to include compliance information
  
  **Options**: "true", "false"
  
  **Example**: "true"
  
  **Default**: "false"
  
  **Note**: Setting to true will include compliance status and alerts
</ParamField>

<ParamField query="status" type="string" required="false">
  Filter licenses by status
  
  **Options**: "active", "expired", "expiring", "suspended", "trial"
  
  **Example**: "active"
  
  **Note**: If not specified, returns licenses of all statuses
</ParamField>

<ParamField query="type" type="string" required="false">
  Filter licenses by type
  
  **Options**: "enterprise", "professional", "basic", "trial", "custom"
  
  **Example**: "enterprise"
  
  **Note**: If not specified, returns licenses of all types
</ParamField>

    class LicenseManagementService {
      constructor(apiClient) {

      async getLicenses(options = {}) {
        try {
          const params = new URLSearchParams();
          
          // Add query parameters
          if (options.includeUsage) params.append('include_usage', options.includeUsage);
          if (options.includeCompliance) params.append('include_compliance', options.includeCompliance);
          if (options.status) params.append('status', options.status);
          if (options.type) params.append('type', options.type);
          
          const response = await this.apiClient.makeRequest(
            `/admin/licenses?${params.toString()}`
          );
          
          if (!response.ok) {
            throw new Error('Failed to get licenses');
          }
          
          return await response.json();
          
        } catch (error) {
          console.error('Error getting licenses:', error);
          throw error;
        }
      }
      
      async getActiveLicenses() {
        return this.getLicenses({
          status: 'active',
          includeUsage: true,
          includeCompliance: true
        });

        return this.getLicenses({
          status: 'expiring',
          includeUsage: true
        });
      }

        return this.getLicenses({
          type: 'enterprise',
          includeUsage: true,
          includeCompliance: true
        });
      }
      
      async getLicenseStatistics() {
        const result = await this.getLicenses({
          includeUsage: true,
          includeCompliance: true
        });
        
        return result.data.summary;

      async getComplianceViolations() {
        const result = await this.getLicenses({
          includeCompliance: true
        });
        
        const violations = [];
        for (const license of result.data.licenses) {
          if (license.compliance && license.compliance.violations) {
            violations.push(...license.compliance.violations);
          }
        }
        
        return violations;
      }
      
      async getUsageAlerts() {
        const result = await this.getLicenses({
          includeUsage: true
        });
        
        const alerts = [];
        for (const license of result.data.licenses) {
          if (license.usage && license.limits) {
            const usagePercentage = (license.usage.users / license.limits.users) * 100;
            if (usagePercentage > 90) {
              alerts.push({
                license_id: license.license_id,
                name: license.name,
                type: 'usage_warning',
                message: `License usage at ${usagePercentage.toFixed(1)}%`,
                severity: 'warning'
              });
            }
          }
        }
        
        return alerts;
      }
      
      async getExpirationAlerts() {
        const result = await this.getLicenses();
        
        const alerts = [];
        const thirtyDaysFromNow = new Date();
        thirtyDaysFromNow.setDate(thirtyDaysFromNow.getDate() + 30);

    import requests
    from typing import Dict, List, Optional
    from datetime import datetime, timedelta
    
    class LicenseManager:
        def __init__(self, api_client):
            self.api_client = api_client
        
        def get_licenses(self, 
                        include_usage: bool = False,
                        include_compliance: bool = False,
                        status: Optional[str] = None,
                        license_type: Optional[str] = None) -> Dict:
            """Get system licenses with optional filtering"""
            try:
                params = {
                    "include_usage": str(include_usage).lower(),
                    "include_compliance": str(include_compliance).lower()
                }
                
                if status:
                    params["status"] = status
                if license_type:
                    params["type"] = license_type
                
                response = self.api_client.get('/admin/licenses', params=params)
                response.raise_for_status()
                
                return response.json()
                
            except requests.exceptions.RequestException as e:
                print(f"Error getting licenses: {e}")
                raise
        
        def get_active_licenses(self) -> Dict:
            """Get all active licenses"""
            return self.get_licenses(
                status="active",
                include_usage=True,
                include_compliance=True
            )
        
        def get_expiring_licenses(self) -> Dict:
            """Get licenses expiring within 30 days"""
            return self.get_licenses(
                status="expiring",
                include_usage=True
            )
        
        def get_enterprise_licenses(self) -> Dict:
            """Get all enterprise licenses"""
            return self.get_licenses(
                license_type="enterprise",
                include_usage=True,
                include_compliance=True
            )
        
        def get_license_statistics(self) -> Dict:
            """Get license statistics and summary"""
            result = self.get_licenses(
                include_usage=True,
                include_compliance=True
            )
            return result["data"]["summary"]
        
        def get_compliance_violations(self) -> List[Dict]:
            """Get all compliance violations"""
            result = self.get_licenses(include_compliance=True)
            
            violations = []
            for license in result["data"]["licenses"]:
                if license.get("compliance") and license["compliance"].get("violations"):
                    violations.extend(license["compliance"]["violations"])
            
            return violations
        
        def get_usage_alerts(self) -> List[Dict]:
            """Get usage alerts for licenses approaching limits"""
            result = self.get_licenses(include_usage=True)
            
            alerts = []
            for license in result["data"]["licenses"]:
                if license.get("usage") and license.get("limits"):
                    usage = license["usage"]
                    limits = license["limits"]
                    
                    # Check user usage
                    if limits["users"] > 0:
                        user_percentage = (usage["users"] / limits["users"]) * 100
                        if user_percentage > 90:
                            alerts.append({
                                "license_id": license["license_id"],
                                "name": license["name"],
                                "type": "usage_warning",
                                "message": f"User usage at {user_percentage:.1f}%",
                                "severity": "warning"
                            })
                    
                    # Check API usage
                    if limits["api_calls"] > 0:
                        api_percentage = (usage["api_calls"] / limits["api_calls"]) * 100
                        if api_percentage > 90:
                            alerts.append({
                                "license_id": license["license_id"],
                                "name": license["name"],
                                "type": "api_usage_warning",
                                "message": f"API usage at {api_percentage:.1f}%",
                                "severity": "warning"
                            })
            
            return alerts
        
        def get_expiration_alerts(self) -> List[Dict]:
            """Get expiration alerts for licenses expiring soon"""
            result = self.get_licenses()
            
            alerts = []
            thirty_days_from_now = datetime.now() + timedelta(days=30)
            
            for license in result["data"]["licenses"]:
                expires_at = datetime.fromisoformat(license["expires_at"].replace("Z", "+00:00"))
                if expires_at <= thirty_days_from_now:
                    alerts.append({
                        "license_id": license["license_id"],
                        "name": license["name"],
                        "type": "expiration_warning",
                        "message": f"License expires on {license['expires_at']}",
                        "severity": "warning"
                    })
            
            return alerts
        
        def get_license_by_id(self, license_id: str) -> Optional[Dict]:
            """Get specific license by ID"""
            result = self.get_licenses(include_usage=True, include_compliance=True)
            
            for license in result["data"]["licenses"]:
                if license["license_id"] == license_id:
                    return license
            
            return None
        
        def get_licenses_by_organization(self, organization: str) -> List[Dict]:
            """Get licenses for specific organization"""
            result = self.get_licenses(include_usage=True)
            
            org_licenses = []
            for license in result["data"]["licenses"]:
                if license["issued_to"].lower() == organization.lower():
                    org_licenses.append(license)
            
            return org_licenses
        
        def get_usage_summary(self) -> Dict:
            """Get overall usage summary across all licenses"""
            result = self.get_licenses(include_usage=True)
            
            total_users = 0
            total_organizations = 0
            total_api_calls = 0
            total_storage = 0
            total_bandwidth = 0
            
            for license in result["data"]["licenses"]:
                if license.get("usage"):
                    usage = license["usage"]
                    total_users += usage.get("users", 0)
                    total_organizations += usage.get("organizations", 0)
                    total_api_calls += usage.get("api_calls", 0)
                    total_storage += usage.get("storage", 0)
                    total_bandwidth += usage.get("bandwidth", 0)
            
            return {
                "total_users": total_users,
                "total_organizations": total_organizations,
                "total_api_calls": total_api_calls,
                "total_storage": total_storage,
                "total_bandwidth": total_bandwidth
            }
