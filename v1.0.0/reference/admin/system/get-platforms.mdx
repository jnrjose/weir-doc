---
title: "Get Admin Platforms"
api: "GET /admin/platforms"
description: "Retrieve list of all platforms in the system with detailed information"
---

Retrieve a comprehensive list of all platforms in the system with their configuration, status, and usage statistics. This endpoint provides admin-level visibility into platform management and monitoring.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for system administration and platform monitoring within the Weir AI platform.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Query Parameters

<ParamField query="page" type="string" required="false">
  Page number for pagination
  
  **Example**: "1"
  
  **Default**: "1"
  
  **Constraints**: 
  - Must be a positive integer
  - Minimum value: 1
</ParamField>

<ParamField query="limit" type="string" required="false">
  Number of platforms to return per page
  
  **Example**: "20"
  
  **Default**: "20"
  
  **Constraints**: 
  - Must be between 1 and 100
  - Recommended: 20-50 for optimal performance
</ParamField>

<ParamField query="status" type="string" required="false">
  Filter platforms by status
  
  **Options**: "active", "inactive", "maintenance", "error", "pending"
  
  **Example**: "active"
  
  **Default**: "active"
  
  **Note**: If not specified, returns platforms of all statuses
</ParamField>

<ParamField query="type" type="string" required="false">
  Filter platforms by type
  
  **Options**: "web", "mobile", "api", "desktop", "integration"
  
  **Example**: "web"
  
  **Note**: If not specified, returns platforms of all types
</ParamField>

<ParamField query="organization" type="string" required="false">
  Filter platforms by organization ID
  
  **Example**: "org_1234567890abcdef"
  
  **Note**: If not specified, returns platforms from all organizations
</ParamField>

<ParamField query="search" type="string" required="false">
  Search platforms by name or description
  
  **Example**: "production"
  
  **Constraints**: 
  - Minimum length: 2 characters
  - Case-insensitive search
  - Searches platform name and description
</ParamField>

<ParamField query="sort_by" type="string" required="false">
  Sort platforms by specified field
  
  **Options**: "name", "created_at", "updated_at", "status", "type", "last_activity", "usage"
  
  **Example**: "last_activity"
  
  **Default**: "created_at"
</ParamField>

<ParamField query="sort_order" type="string" required="false">
  Sort order for the results
  
  **Options**: "asc", "desc"
  
  **Example**: "desc"
  
  **Default**: "desc"
</ParamField>

<ParamField query="include_usage" type="string" required="false">
  Whether to include usage statistics
  
  **Options**: "true", "false"
  
  **Example**: "true"
  
  **Default**: "false"
  
  **Note**: Setting to true will include detailed usage metrics
</ParamField>

<ParamField query="include_config" type="string" required="false">
  Whether to include platform configuration
  
  **Options**: "true", "false"
  
  **Example**: "true"
  
  **Default**: "false"
  
  **Note**: Setting to true will include platform configuration details
</ParamField>

<ParamField query="include_health" type="string" required="false">
  Whether to include health status information
  
  **Options**: "true", "false"
  
  **Example**: "true"
  
  **Default**: "false"
  
  **Note**: Setting to true will include health monitoring data
</ParamField>

    class PlatformManagementService {
      constructor(apiClient) {
        this.apiClient = apiClient;
      }
      
      async getPlatforms(options = {}) {
        try {
          const params = new URLSearchParams();
          
          // Add query parameters
          if (options.page) params.append('page', options.page);
          if (options.limit) params.append('limit', options.limit);
          if (options.status) params.append('status', options.status);
          if (options.type) params.append('type', options.type);
          if (options.organization) params.append('organization', options.organization);
          if (options.search) params.append('search', options.search);
          if (options.sortBy) params.append('sort_by', options.sortBy);
          if (options.sortOrder) params.append('sort_order', options.sortOrder);
          if (options.includeUsage) params.append('include_usage', options.includeUsage);
          if (options.includeConfig) params.append('include_config', options.includeConfig);
          if (options.includeHealth) params.append('include_health', options.includeHealth);
          
          const response = await this.apiClient.makeRequest(
            `/admin/platforms?${params.toString()}`
          );
          
          if (!response.ok) {
            throw new Error('Failed to get platforms');
          }
          
          return await response.json();
          
        } catch (error) {
          console.error('Error getting platforms:', error);
          throw error;
        }
      }
      
      async getActivePlatforms() {
        return this.getPlatforms({
          status: 'active',
          includeUsage: true,
          includeHealth: true,
          sortBy: 'last_activity',
          sortOrder: 'desc'
        });
      }
      
      async getPlatformsByType(type) {
        return this.getPlatforms({
          type: type,
          status: 'active',
          includeUsage: true,
          includeConfig: true
        });
      }
      
      async getPlatformsByOrganization(organizationId) {
        return this.getPlatforms({
          organization: organizationId,
          includeUsage: true,
          includeHealth: true
        });
      }
      
      async getWebPlatforms() {
        return this.getPlatformsByType('web');
      }
      
      async getMobilePlatforms() {
        return this.getPlatformsByType('mobile');
      }
      
      async getAPIPlatforms() {
        return this.getPlatformsByType('api');
      }
      
      async getPlatformsWithErrors() {
        return this.getPlatforms({
          status: 'error',
          includeHealth: true
        });
      }
      
      async getPlatformsInMaintenance() {
        return this.getPlatforms({
          status: 'maintenance',
          includeHealth: true
        });
      }
      
      async getHighUsagePlatforms() {
        const result = await this.getPlatforms({
          includeUsage: true,
          sortBy: 'usage',
          sortOrder: 'desc'
        });
        
        return result.data.filter(platform => 
          platform.usage && platform.usage.api_calls > 100000
        );
      }
      
      async getPlatformsWithHealthIssues() {
        const result = await this.getPlatforms({
          includeHealth: true
        });
        
        return result.data.filter(platform => 
          platform.health && platform.health.status !== 'healthy'
        );
      }
      
      async getPlatformStatistics() {
        const result = await this.getPlatforms({
          limit: 1
        });
        
        return result.summary;
      }
      
      async searchPlatforms(searchTerm) {
        return this.getPlatforms({
          search: searchTerm,
          sortBy: 'name',
          sortOrder: 'asc'
        });
      }
    }
    
    // Usage example
    const platformService = new PlatformManagementService(apiClient);

    console.log(`Found ${webPlatforms.data.length} web platforms`);
    
    // Get platforms with errors
    const errorPlatforms = await platformService.getPlatformsWithErrors();
