---
title: "Get Mail Templates"
api: "GET /mail/templates"
description: "Retrieve list of all email templates in the system"
---

Retrieve a comprehensive list of all email templates with their configuration, usage statistics, and status information. This endpoint provides admin-level visibility into email template management.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for email template management within the Weir AI platform.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Query Parameters

<ParamField query="page" type="string" required="false">
  Page number for pagination
  
  **Example**: "1"
  
  **Default**: "1"
</ParamField>

<ParamField query="limit" type="string" required="false">
  Number of templates to return per page
  
  **Example**: "20"
  
  **Default**: "20"
</ParamField>

<ParamField query="type" type="string" required="false">
  Filter templates by type
  
  **Options**: "welcome", "notification", "alert", "invitation", "custom"
  
  **Example**: "welcome"
</ParamField>

<ParamField query="status" type="string" required="false">
  Filter templates by status
  
  **Options**: "active", "inactive", "draft"
  
  **Example**: "active"
  
  **Default**: "active"
</ParamField>

<ParamField query="search" type="string" required="false">
  Search templates by name or content
  
  **Example**: "welcome"
  
  **Constraints**: 
  - Minimum length: 2 characters
  - Case-insensitive search
</ParamField>

<ParamField query="sort_by" type="string" required="false">
  Sort templates by specified field
  
  **Options**: "name", "type", "created_at", "updated_at", "usage_count"
  
  **Example**: "created_at"
  
  **Default**: "created_at"
</ParamField>

<ParamField query="sort_order" type="string" required="false">
  Sort order for the results
  
  **Options**: "asc", "desc"
  
  **Example**: "desc"
  
  **Default**: "desc"
</ParamField>

<ParamField query="include_usage" type="string" required="false">
  Whether to include usage statistics
  
  **Options**: "true", "false"
  
  **Example**: "true"
  
  **Default**: "false"
</ParamField>

## Code Examples

<AccordionGroup>
  <Accordion title="Template Management Service" icon="cog">
    ```javascript
    class TemplateManagementService {
      constructor(apiClient) {
        this.apiClient = apiClient;
      }
      
      async getTemplates(options = {}) {
        try {
          const params = new URLSearchParams();
          
          if (options.page) params.append('page', options.page);
          if (options.limit) params.append('limit', options.limit);
          if (options.type) params.append('type', options.type);
          if (options.status) params.append('status', options.status);
          if (options.search) params.append('search', options.search);
          if (options.sortBy) params.append('sort_by', options.sortBy);
          if (options.sortOrder) params.append('sort_order', options.sortOrder);
          if (options.includeUsage) params.append('include_usage', options.includeUsage);
          
          const response = await this.apiClient.makeRequest(
            `/mail/templates?${params.toString()}`
          );
          
          if (!response.ok) {
            throw new Error('Failed to get templates');
          }
          
          return await response.json();
          
        } catch (error) {
          console.error('Error getting templates:', error);
          throw error;
        }
      }
      
      async getActiveTemplates() {
        return this.getTemplates({
          status: 'active',
          includeUsage: true,
          sortBy: 'name',
          sortOrder: 'asc'
        });
      }
      
      async getTemplatesByType(type) {
        return this.getTemplates({
          type: type,
          status: 'active',
          includeUsage: true
        });
      }
      
      async getWelcomeTemplates() {
        return this.getTemplatesByType('welcome');
      }
      
      async getNotificationTemplates() {
        return this.getTemplatesByType('notification');
      }
      
      async getTemplateStatistics() {
        const result = await this.getTemplates({
          limit: 1
        });
        
        return result.summary;
      }
    }
    
    // Usage example
    const templateService = new TemplateManagementService(apiClient);
    
    // Get active templates
    const activeTemplates = await templateService.getActiveTemplates();
    console.log(`Found ${activeTemplates.data.length} active templates`);
    
    // Get welcome templates
    const welcomeTemplates = await templateService.getWelcomeTemplates();
    console.log(`Found ${welcomeTemplates.data.length} welcome templates`);
    
    // Get template statistics
    const stats = await templateService.getTemplateStatistics();
    console.log('Template Statistics:', stats);
    ```
