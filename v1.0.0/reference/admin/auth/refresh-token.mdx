---
title: "Admin Refresh Token"
api: "POST /auth/refresh/token"
description: "Refresh an expired admin access token using a refresh token"
---

Refresh an expired admin access token using a valid refresh token. This endpoint allows seamless token renewal without requiring re-authentication.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for secure token management within the Weir AI platform.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Request

<ParamField body="refresh_token" type="string" required="false">
  Refresh token to use for token renewal
  
  **Example**: "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9..."
  
  **Constraints**: 
  - Must be a valid refresh token
  - Token must not be expired
  - If not provided, uses the refresh token from the Authorization header
</ParamField>

<ParamField body="extend_session" type="boolean" required="false">
  Whether to extend the session duration
  
  **Example**: true
  
  **Default**: false
  
  **Note**: If true, extends both access and refresh token expiration times
</ParamField>

    ```javascript
    class AdminTokenManager {
      constructor(apiClient) {
        this.apiClient = apiClient;
        this.tokenRefreshTimer = null;
      }
      
      async refreshToken(extendSession = false) {
        try {
          const refreshToken = this.getRefreshToken();
          
          const response = await this.apiClient.makeRequest('/auth/refresh/token', {
            method: 'POST',
            body: JSON.stringify({
              refresh_token: refreshToken,
              extend_session: extendSession
            })
          });
          
          if (!response.ok) {
            throw new Error('Failed to refresh token');
          }
          
          const result = await response.json();
          
          // Store new tokens
          this.storeTokens(result.data);
          
          // Set up automatic refresh
          this.scheduleTokenRefresh(result.data.expires_in);
          
          console.log('Token refreshed successfully');
          return result;
          
        } catch (error) {
          console.error('Error refreshing token:', error);
          this.handleTokenRefreshError(error);
          throw error;
        }
      }
      
      async refreshTokenIfNeeded() {
        const accessToken = this.getAccessToken();
        
        if (!accessToken || this.isTokenExpired(accessToken)) {
          return await this.refreshToken();
        }
        
        return null;
      }
      
      async refreshTokenWithRetry(maxRetries = 3) {
        for (let attempt = 1; attempt <= maxRetries; attempt++) {
          try {
            return await this.refreshToken();
          } catch (error) {
            if (attempt === maxRetries) {
              throw error;
            }
            
            // Wait before retry
            await new Promise(resolve => setTimeout(resolve, 1000 * attempt));
          }
        }
      }
      
      storeTokens(tokenData) {
        // Store access token
        localStorage.setItem('admin_access_token', tokenData.access_token);
        localStorage.setItem('admin_token_expires_at', tokenData.expires_at);
        
        // Store refresh token
        localStorage.setItem('admin_refresh_token', tokenData.refresh_token);
        localStorage.setItem('admin_refresh_expires_at', tokenData.refresh_expires_at);
        
        // Store user info
        localStorage.setItem('admin_user_id', tokenData.user_id);
        localStorage.setItem('admin_roles', JSON.stringify(tokenData.roles));
        localStorage.setItem('admin_permissions', JSON.stringify(tokenData.permissions));
      }
      
      getAccessToken() {
        return localStorage.getItem('admin_access_token');
      }
      
      getRefreshToken() {
        return localStorage.getItem('admin_refresh_token');
      }
      
      isTokenExpired(token) {
        try {
          const payload = JSON.parse(atob(token.split('.')[1]));
          const now = Math.floor(Date.now() / 1000);
          return payload.exp < now;
        } catch (error) {
          return true;
        }
      }
      
      scheduleTokenRefresh(expiresIn) {
        // Clear existing timer
        if (this.tokenRefreshTimer) {
          clearTimeout(this.tokenRefreshTimer);
        }
        
        // Schedule refresh 5 minutes before expiration
        const refreshTime = (expiresIn - 300) * 1000;
        
        this.tokenRefreshTimer = setTimeout(async () => {
          try {
            await this.refreshToken();
          } catch (error) {
            console.error('Automatic token refresh failed:', error);
            this.handleTokenRefreshError(error);
          }
        }, refreshTime);
      }
      
      handleTokenRefreshError(error) {
        // Clear tokens
        this.clearTokens();
        
        // Redirect to login
        window.location.href = '/admin/login';
      }
      
      clearTokens() {
        localStorage.removeItem('admin_access_token');
        localStorage.removeItem('admin_refresh_token');
        localStorage.removeItem('admin_token_expires_at');
        localStorage.removeItem('admin_refresh_expires_at');
        localStorage.removeItem('admin_user_id');
        localStorage.removeItem('admin_roles');
        localStorage.removeItem('admin_permissions');
        
        if (this.tokenRefreshTimer) {
          clearTimeout(this.tokenRefreshTimer);
          this.tokenRefreshTimer = null;
        }
      }
      
      async extendSession() {
        return await this.refreshToken(true);
      }
    }
    
    // Usage example
    const tokenManager = new AdminTokenManager(apiClient);
    
    // Refresh token
    await tokenManager.refreshToken();
    
    // Refresh token if needed
    await tokenManager.refreshTokenIfNeeded();
    
    // Extend session
    await tokenManager.extendSession();
    
    // Refresh with retry
    await tokenManager.refreshTokenWithRetry(3);
    ```

    ```python
    import requests
    import json
    import time
    from typing import Optional, Dict
    from datetime import datetime, timedelta
    
    class AdminTokenClient:
        def __init__(self, api_client):
            self.api_client = api_client
            self.token_file = 'admin_tokens.json'
        
        def refresh_token(self, extend_session: bool = False) -> Dict:
            """Refresh admin access token"""
            try:
                refresh_token = self.get_refresh_token()
                
                payload = {
                    "refresh_token": refresh_token,
                    "extend_session": extend_session
                }
                
                response = self.api_client.post('/auth/refresh/token', json=payload)
                response.raise_for_status()
                
                result = response.json()
                
                # Store new tokens
                self.store_tokens(result['data'])
                
                print("Token refreshed successfully")
                return result
                
            except requests.exceptions.RequestException as e:
                print(f"Error refreshing token: {e}")
                self.handle_token_refresh_error(e)
                raise
        
        def refresh_token_if_needed(self) -> Optional[Dict]:
            """Refresh token if it's expired or about to expire"""
            access_token = self.get_access_token()
            
            if not access_token or self.is_token_expired(access_token):
                return self.refresh_token()
            
            return None
        
        def refresh_token_with_retry(self, max_retries: int = 3) -> Dict:
            """Refresh token with retry logic"""
            for attempt in range(1, max_retries + 1):
                try:
                    return self.refresh_token()
                except Exception as e:
                    if attempt == max_retries:
                        raise e
                    
                    # Wait before retry
                    time.sleep(1 * attempt)
        
        def store_tokens(self, token_data: Dict):
            """Store tokens securely"""
            tokens = {
                'access_token': token_data['access_token'],
                'refresh_token': token_data['refresh_token'],
                'expires_at': token_data['expires_at'],
                'refresh_expires_at': token_data['refresh_expires_at'],
                'user_id': token_data['user_id'],
                'roles': token_data['roles'],
                'permissions': token_data['permissions']
            }
            
            with open(self.token_file, 'w') as f:
                json.dump(tokens, f)
        
        def get_access_token(self) -> Optional[str]:
            """Get stored access token"""
            try:
                with open(self.token_file, 'r') as f:
                    tokens = json.load(f)
                return tokens.get('access_token')
            except FileNotFoundError:
                return None
        
        def get_refresh_token(self) -> Optional[str]:
            """Get stored refresh token"""
            try:
                with open(self.token_file, 'r') as f:
                    tokens = json.load(f)
                return tokens.get('refresh_token')
            except FileNotFoundError:
                return None
        
        def is_token_expired(self, token: str) -> bool:
            """Check if token is expired"""
            try:
                import base64
                payload = json.loads(base64.b64decode(token.split('.')[1]))
                exp = payload.get('exp', 0)
                return exp < time.time()
            except Exception:
                return True
        
        def handle_token_refresh_error(self, error: Exception):
            """Handle token refresh errors"""
            # Clear tokens
            self.clear_tokens()
            
            # Log error
            print(f"Token refresh failed: {error}")
            
            # Redirect to login (in web context)
            # In API context, raise exception
            raise Exception("Authentication failed. Please login again.")
        
        def clear_tokens(self):
            """Clear stored tokens"""
            try:
                import os
                if os.path.exists(self.token_file):
                    os.remove(self.token_file)
            except Exception:
                pass
        
        def extend_session(self) -> Dict:
            """Extend session duration"""
            return self.refresh_token(extend_session=True)
        
        def get_token_info(self) -> Optional[Dict]:
            """Get current token information"""
            try:
                with open(self.token_file, 'r') as f:
                    return json.load(f)
            except FileNotFoundError:
                return None
        
        def is_session_valid(self) -> bool:
            """Check if current session is valid"""
            access_token = self.get_access_token()
            return access_token is not None and not self.is_token_expired(access_token)
    
    # Usage example
    admin_token_client = AdminTokenClient(api_client)
    
    # Refresh token
    result = admin_token_client.refresh_token()
    
    # Refresh token if needed
    result = admin_token_client.refresh_token_if_needed()
    
    # Extend session
    result = admin_token_client.extend_session()
    
    # Refresh with retry
    result = admin_token_client.refresh_token_with_retry(3)
    
    # Check session validity
    is_valid = admin_token_client.is_session_valid()
    print(f"Session valid: {is_valid}")
    ```
