---
title: "Refresh Token"
api: "POST /auth/refresh/token"
description: "Refresh an expired access token using a valid refresh token"
---

Refresh an expired access token using a valid refresh token. This endpoint allows you to obtain a new access token without requiring the user to log in again.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for seamless token management and session continuity.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

<CodeBlock lang="bash">
{`Authorization: Bearer <your_jwt_token>`}
</CodeBlock>

## Request

<ParamField body="refreshToken" type="string" required>
  The valid refresh token to use for obtaining a new access token
  
  **Example**: "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9..."
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Indicates if the token refresh was successful
</ResponseField>

<ResponseField name="data" type="object">
  Contains the new authentication tokens
</ResponseField>

<ResponseField name="data.access_token" type="string">
  New JWT access token for API authentication
</ResponseField>

<ResponseField name="data.refresh_token" type="string">
  New refresh token (may be the same or different from the original)
</ResponseField>

<ResponseField name="data.expires_in" type="integer">
  Access token expiration time in seconds (typically 3600 seconds = 1 hour)
</ResponseField>

<ResponseField name="data.token_type" type="string">
  Token type, always "Bearer"
</ResponseField>

<ResponseField name="data.refresh_expires_in" type="integer">
  Refresh token expiration time in seconds (typically 604800 seconds = 7 days)
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable message about the operation result
</ResponseField>

<ResponseField name="timestamp" type="string">
  ISO 8601 timestamp of the response
</ResponseField>

## Usage Notes

<AccordionGroup>
  <Accordion title="Token Lifecycle" icon="clock">
    Understanding token expiration:
    
    - Access tokens expire after 1 hour (3600 seconds)
    - Refresh tokens expire after 7 days (604800 seconds)
    - Use refresh tokens to get new access tokens before expiry
    - Refresh tokens may be rotated for security
  </Accordion>
  
  <Accordion title="Automatic Refresh" icon="refresh">
    Implement automatic token refresh:
    
    - Monitor access token expiration
    - Refresh tokens before they expire
    - Handle refresh failures gracefully
    - Update stored tokens after successful refresh
  </Accordion>
  
  <Accordion title="Security Considerations" icon="shield">
    Token refresh security:
    
    - Refresh tokens are more sensitive than access tokens
    - Store refresh tokens securely
    - Implement proper error handling
    - Log refresh attempts for security monitoring
  </Accordion>
  
  <Accordion title="Rate Limiting" icon="clock">
    Token refresh is rate limited:
    
    - Maximum 10 refresh attempts per hour
    - Rate limits help prevent token abuse
    - Implement exponential backoff for retries
    - Consider caching tokens to reduce refresh frequency
  </Accordion>
</AccordionGroup>

<AccordionGroup>
  <Accordion title="Automatic Token Refresh" icon="refresh">
    <CodeBlock lang="javascript">
{`class TokenRefreshManager {
  constructor(apiClient) {
    this.apiClient = apiClient;
    this.refreshPromise = null;
    this.isRefreshing = false;
  }
  
  async refreshAccessToken() {
    // Prevent multiple simultaneous refresh attempts
    if (this.isRefreshing) {
      return this.refreshPromise;
    }
    
    this.isRefreshing = true;
    this.refreshPromise = this.performTokenRefresh();
    
    try {
      const result = await this.refreshPromise;
      return result;
    } finally {
      this.isRefreshing = false;
      this.refreshPromise = null;
    }
  }
  
  async performTokenRefresh() {
    const refreshToken = this.getRefreshToken();
    
    if (!refreshToken) {
      throw new Error('No refresh token available');
    }
    
    try {
      const response = await this.apiClient.makeRequest('/auth/refresh/token', {
        method: 'POST',
        body: JSON.stringify({ refreshToken })
      });
      
      if (!response.ok) {
        const error = await response.json();
        throw new Error(\`Token refresh failed: \${error.error.message}\`);
      }
      
      const data = await response.json();
      
      // Update stored tokens
      this.updateStoredTokens(data.data);
      
      return data.data;
      
    } catch (error) {
      console.error('Token refresh error:', error);
      
      // If refresh fails, clear tokens and redirect to login
      this.clearTokens();
      this.redirectToLogin();
      
      throw error;
    }
  }
  
  updateStoredTokens(tokenData) {
    const expiryTime = new Date(Date.now() + tokenData.expires_in * 1000);
    
    localStorage.setItem('access_token', tokenData.access_token);
    localStorage.setItem('refresh_token', tokenData.refresh_token);
    localStorage.setItem('token_expiry', expiryTime.toISOString());
    
    // Update API client with new token
    this.apiClient.setAccessToken(tokenData.access_token);
  }
  
  clearTokens() {
    localStorage.removeItem('access_token');
    localStorage.removeItem('refresh_token');
    localStorage.removeItem('token_expiry');
  }
  
  redirectToLogin() {
    window.location.href = '/login';
  }
  
  getRefreshToken() {
    return localStorage.getItem('refresh_token');
  }
  
  isTokenExpired() {
    const expiryTime = localStorage.getItem('token_expiry');
    
    if (!expiryTime) {
      return true;
    }
    
    return new Date() >= new Date(expiryTime);
  }
  
  async ensureValidToken() {
    if (this.isTokenExpired()) {
      await this.refreshAccessToken();
    }
  }
}`}

{`class AuthenticatedRequestInterceptor {
  constructor(apiClient) {
    this.apiClient = apiClient;
    this.tokenManager = new TokenRefreshManager(apiClient);
  }
  
  async makeAuthenticatedRequest(endpoint, options = {}) {
    // Ensure we have a valid token before making the request
    await this.tokenManager.ensureValidToken();
    
    try {
      const response = await this.apiClient.makeRequest(endpoint, options);
      
      // If we get a 401, try to refresh the token and retry once
      if (response.status === 401) {
        console.log('Access token expired, attempting refresh...');
        
        try {
          await this.tokenManager.refreshAccessToken();
          
          // Retry the original request with the new token
          return await this.apiClient.makeRequest(endpoint, options);
          
        } catch (refreshError) {
          console.error('Token refresh failed:', refreshError);
          throw new Error('Authentication failed - please login again');
        }
      }
      
      return response;
      
    } catch (error) {
      console.error('Request failed:', error);
      throw error;
    }
  }
  
  // Wrapper methods for common HTTP methods
  async get(endpoint, options = {}) {
    return this.makeAuthenticatedRequest(endpoint, { ...options, method: 'GET' });
  }
  
  async post(endpoint, data, options = {}) {
    return this.makeAuthenticatedRequest(endpoint, {
      ...options,
      method: 'POST',
      body: JSON.stringify(data)
    });
  }
  
  async put(endpoint, data, options = {}) {
    return this.makeAuthenticatedRequest(endpoint, {
      ...options,
      method: 'PUT',
      body: JSON.stringify(data)
    });
  }
  
  async delete(endpoint, options = {}) {
    return this.makeAuthenticatedRequest(endpoint, { ...options, method: 'DELETE' });
  }
}

// Usage
const interceptor = new AuthenticatedRequestInterceptor(apiClient);
