---
title: "Admin Refresh Token"
api: "POST /auth/refresh/token"
description: "Refresh expired admin access token"
---

Refresh an expired admin access token using a valid refresh token. This endpoint allows you to obtain a new access token without requiring the admin to log in again.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for seamless token renewal and session management.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <admin_jwt_token>
```

## Request

<ParamField body="refresh_token" type="string" required="false">
  Refresh token for obtaining new access token
  
  **Example**: "rt_admin_1234567890abcdef"
  
  **Note**: If not provided, will use the refresh token from the current session
</ParamField>

<ParamField body="extend_session" type="boolean" required="false">
  Whether to extend the session duration
  
  **Example**: true
  
  **Default**: false
  
  **Note**: If true, extends both access and refresh token expiration
</ParamField>

    ```javascript
class AdminTokenRefreshService {
  constructor(apiClient) {
    this.apiClient = apiClient;
  }
  
  async refreshToken(refreshToken = null, extendSession = false) {
    try {
      const response = await this.apiClient.makeRequest('/auth/refresh/token', {
        method: 'POST',
        body: JSON.stringify({
          refresh_token: refreshToken,
          extend_session: extendSession
        })
      });
      
      if (!response.ok) {
        throw new Error('Failed to refresh admin token');
      }
      
      const result = await response.json();
      
      // Store new tokens
      await this.storeNewTokens(result.data.access_token, result.data.refresh_token);
      
      return result;
      
    } catch (error) {
      console.error('Error refreshing admin token:', error);
      throw error;
    }
  }
  
  async storeNewTokens(accessToken, refreshToken) {
    // Store new tokens in secure storage
    await SecureStore.setItemAsync('admin_access_token', accessToken);
    await SecureStore.setItemAsync('admin_refresh_token', refreshToken);
  }
  
  async autoRefreshToken() {
    const refreshToken = await SecureStore.getItemAsync('admin_refresh_token');
    if (!refreshToken) {
      throw new Error('No refresh token available');
    }
    
    return this.refreshToken(refreshToken, true);
  }
}
```

    ```python
import requests
from typing import Optional
from datetime import datetime, timedelta

class AdminTokenManager:
    def __init__(self, api_client):
        self.api_client = api_client
        self.access_token = None
        self.refresh_token = None
        self.token_expires_at = None
    
    def refresh_token(self, 
                     refresh_token: Optional[str] = None,
                     extend_session: bool = False) -> dict:
        """Refresh admin access token"""
        try:
            payload = {
                "refresh_token": refresh_token or self.refresh_token,
                "extend_session": extend_session
            }
            
            response = self.api_client.post('/auth/refresh/token', json=payload)
            response.raise_for_status()
            
            result = response.json()
            data = result.get("data", {})
            
            # Update stored tokens
            self.access_token = data.get("access_token")
            self.refresh_token = data.get("refresh_token")
            
            # Calculate expiration time
            expires_in = data.get("expires_in", 3600)
            self.token_expires_at = datetime.utcnow() + timedelta(seconds=expires_in)
            
            return result
            
        except requests.exceptions.RequestException as e:
            print(f"Token refresh failed: {e}")
            raise
    
    def is_token_expired(self) -> bool:
        """Check if current access token is expired"""
        if not self.token_expires_at:
            return True
        
        # Refresh token 5 minutes before expiration
        return datetime.utcnow() >= (self.token_expires_at - timedelta(minutes=5))
    
    def auto_refresh_if_needed(self) -> bool:
        """Automatically refresh token if needed"""
        if self.is_token_expired():
            try:
                self.refresh_token(extend_session=True)
                return True
            except Exception as e:
                print(f"Auto refresh failed: {e}")
                return False
        return True
    
    def get_valid_token(self) -> Optional[str]:
        """Get a valid access token, refreshing if necessary"""
        if self.auto_refresh_if_needed():
            return self.access_token
        return None
```
