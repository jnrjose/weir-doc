---
title: "Admin Logout"
api: "POST /auth/logout"
description: "Logout and invalidate the current admin session"
---

Logout from the admin session and invalidate the current authentication tokens. This endpoint ensures secure session termination and token revocation.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for secure session management and logout functionality.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

<CodeBlock lang="bash">
{`Authorization: Bearer <your_jwt_token>`}
</CodeBlock>

## Request

<ParamField body="refreshToken" type="string" required>
  The refresh token to invalidate during logout
  
  **Example**: "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9..."
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Indicates if the logout was successful
</ResponseField>

<ResponseField name="data" type="object">
  Contains logout confirmation information
</ResponseField>

<ResponseField name="data.logged_out_at" type="string">
  ISO 8601 timestamp when the logout occurred
</ResponseField>

<ResponseField name="data.tokens_invalidated" type="array">
  List of token types that were invalidated
</ResponseField>

<ResponseField name="data.session_ended" type="boolean">
  Confirmation that the session has been terminated
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable message about the operation result
</ResponseField>

<ResponseField name="timestamp" type="string">
  ISO 8601 timestamp of the response
</ResponseField>

## Usage Notes

<AccordionGroup>
  <Accordion title="Token Invalidation" icon="key">
    When you logout:
    
    - Access token is immediately invalidated
    - Refresh token is revoked and cannot be used again
    - All active sessions for the user are terminated
    - User must login again to access protected resources
  </Accordion>
  
  <Accordion title="Security Benefits" icon="shield">
    Logout provides important security benefits:
    
    - Prevents unauthorized access if tokens are compromised
    - Ensures clean session termination
    - Helps with audit trails and compliance
    - Protects against session hijacking
  </Accordion>
  
  <Accordion title="Client-Side Cleanup" icon="trash">
    After successful logout, you should:
    
    - Remove tokens from local storage
    - Clear any cached user data
    - Redirect to login page
    - Clear any application state
  </Accordion>
  
  <Accordion title="Error Handling" icon="exclamation-triangle">
    Handle logout errors gracefully:
    
    - If logout fails, tokens may still be valid
    - Consider forcing client-side cleanup anyway
    - Log errors for debugging purposes
    - Provide user feedback about logout status
  </Accordion>
</AccordionGroup>

  <Accordion title="Secure Logout Implementation" icon="shield-check">
    <CodeBlock lang="javascript">
{`class SecureLogoutManager {
  constructor(apiClient) {
    this.apiClient = apiClient;
  }
  
  async logout() {
    try {
      // Get refresh token from secure storage
      const refreshToken = this.getRefreshToken();
      
      if (!refreshToken) {
        console.warn('No refresh token found, performing client-side logout');
        this.performClientSideLogout();
        return;
      }
      
      // Attempt server-side logout
      const response = await this.apiClient.makeRequest('/auth/logout', {
        method: 'POST',
        body: JSON.stringify({ refreshToken })
      });
      
      if (response.ok) {
        const data = await response.json();
        console.log('Server-side logout successful:', data.message);
      } else {
        console.warn('Server-side logout failed, performing client-side logout');
      }
      
      // Always perform client-side cleanup
      this.performClientSideLogout();
      
    } catch (error) {
      console.error('Logout error:', error);
      // Still perform client-side cleanup even if server logout fails
      this.performClientSideLogout();
    }
  }
  
  performClientSideLogout() {
    // Remove tokens from storage
    this.removeTokens();
    
    // Clear user data
    this.clearUserData();
    
    // Clear application state
    this.clearApplicationState();
    
    // Redirect to login
    this.redirectToLogin();
  }
  
  removeTokens() {
    // Remove from localStorage
    localStorage.removeItem('access_token');
    localStorage.removeItem('refresh_token');
    localStorage.removeItem('token_expiry');
    
    // Remove from sessionStorage
    sessionStorage.removeItem('access_token');
    sessionStorage.removeItem('refresh_token');
    
    // Remove from secure storage if using
    if (typeof SecureStorage !== 'undefined') {
      SecureStorage.removeItem('access_token');
      SecureStorage.removeItem('refresh_token');
    }
  }
  
  clearUserData() {
    localStorage.removeItem('user_profile');
    localStorage.removeItem('user_preferences');
    localStorage.removeItem('organization_data');
  }
  
  clearApplicationState() {
    // Clear any application state
    if (window.appState) {
      window.appState.clear();
    }
    
    // Clear any cached data
    if (window.cache) {
      window.cache.clear();
    }
  }
  
  redirectToLogin() {
    // Redirect to login page
    window.location.href = '/login';
  }
  
  getRefreshToken() {
    // Try different storage locations
    return localStorage.getItem('refresh_token') || 
           sessionStorage.getItem('refresh_token') ||
           (typeof SecureStorage !== 'undefined' ? SecureStorage.getItem('refresh_token') : null);
  }
}`}

    <CodeBlock lang="javascript">
{`class AutoLogoutManager {
  constructor(apiClient) {
    this.apiClient = apiClient;
    this.logoutManager = new SecureLogoutManager(apiClient);
    this.checkInterval = null;
    this.warningShown = false;
  }
  
  startTokenMonitoring() {
    // Check token expiry every minute
    this.checkInterval = setInterval(() => {
      this.checkTokenExpiry();
    }, 60000);
  }
  
  stopTokenMonitoring() {
    if (this.checkInterval) {
      clearInterval(this.checkInterval);
      this.checkInterval = null;
    }
  }
  
  checkTokenExpiry() {
    const tokenExpiry = localStorage.getItem('token_expiry');
    
    if (!tokenExpiry) {
      this.logoutManager.logout();
      return;
    }
    
    const expiryTime = new Date(tokenExpiry);
    const now = new Date();
    const timeUntilExpiry = expiryTime - now;
    
    // If token expires in less than 5 minutes, show warning
    if (timeUntilExpiry < 5 * 60 * 1000 && !this.warningShown) {
      this.showExpiryWarning();
    }
    
    // If token has expired, logout immediately
    if (timeUntilExpiry <= 0) {
      this.logoutManager.logout();
    }
  }
  
  showExpiryWarning() {
    this.warningShown = true;
    
    // Show warning to user
    const warning = confirm(
      'Your session will expire in 5 minutes. Would you like to extend your session?'
    );
    
    if (warning) {
      this.extendSession();
    } else {
      this.logoutManager.logout();
    }
  }
  
  async extendSession() {
    try {
      const refreshToken = this.logoutManager.getRefreshToken();
      
      if (!refreshToken) {
        this.logoutManager.logout();
        return;
      }
      
      const response = await this.apiClient.makeRequest('/auth/refresh/token', {
        method: 'POST',
        body: JSON.stringify({ refresh_token: refreshToken })
      });
      
      if (response.ok) {
        const data = await response.json();
        
        // Update tokens
        localStorage.setItem('access_token', data.data.access_token);
        localStorage.setItem('token_expiry', new Date(Date.now() + data.data.expires_in * 1000).toISOString());
        
        this.warningShown = false;
        console.log('Session extended successfully');
      } else {
        this.logoutManager.logout();
      }
      
    } catch (error) {
      console.error('Failed to extend session:', error);
      this.logoutManager.logout();
    }
  }
  
  // Call this when user becomes active
  onUserActivity() {
    this.warningShown = false;
  }
}`}
