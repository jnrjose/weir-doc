---
title: "Admin Logout"
api: "POST /auth/logout"
description: "Logout admin user and invalidate session"
---

Logout an admin user and invalidate their current session. This endpoint terminates the admin session and invalidates the access token, ensuring secure logout functionality.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for secure admin session termination.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

<CodeBlock lang="bash">
Authorization: Bearer <admin_jwt_token>
</CodeBlock>

## Request

<ParamField body="logout_all_sessions" type="boolean" required="false">
  Whether to logout from all admin sessions
  
  **Example**: false
  
  **Default**: false
  
  **Note**: If true, invalidates all sessions for the admin user
</ParamField>

<ParamField body="reason" type="string" required="false">
  Reason for logout (for audit purposes)
  
  **Example**: "User requested logout"
  
  **Constraints**: 
  - Maximum length: 200 characters
</ParamField>

class AdminLogoutService {
  constructor(apiClient) {
    this.apiClient = apiClient;
  }
  
  async logout(logoutAllSessions = false, reason = null) {
    try {
      const response = await this.apiClient.makeRequest('/auth/logout', {
        method: 'POST',
        body: JSON.stringify({
          logout_all_sessions: logoutAllSessions,
          reason: reason
        })
      });
      
      if (!response.ok) {
        throw new Error('Failed to logout admin user');
      }
      
      const result = await response.json();
      
      // Clear stored tokens
      await this.clearStoredTokens();
      
      return result;
      
    } catch (error) {
      console.error('Error during admin logout:', error);
      throw error;
    }
  }
  
  async clearStoredTokens() {
    // Clear admin tokens from secure storage
    await SecureStore.deleteItemAsync('admin_access_token');
    await SecureStore.deleteItemAsync('admin_refresh_token');
  }

    return this.logout(true, reason);
  }
}

import requests
from typing import Optional

class AdminLogoutClient:
    def __init__(self, api_client):
        self.api_client = api_client
    
    def logout(self, 
               logout_all_sessions: bool = False,
               reason: Optional[str] = None) -> dict:
        """Logout admin user"""
        try:
            payload = {
                "logout_all_sessions": logout_all_sessions,
                "reason": reason
            }
            
            response = self.api_client.post('/auth/logout', json=payload)
            response.raise_for_status()
            
            # Clear stored tokens
            self.clear_tokens()
            
            return response.json()
            
        except requests.exceptions.RequestException as e:
            print(f"Admin logout failed: {e}")
            raise
    
    def clear_tokens(self):
        """Clear stored authentication tokens"""
        # Clear tokens from storage
        if hasattr(self.api_client, 'access_token'):
            self.api_client.access_token = None
        if hasattr(self.api_client, 'refresh_token'):
            self.api_client.refresh_token = None
    
    def logout_all_sessions(self, reason: str = "Security policy enforcement") -> dict:
        """Logout from all admin sessions"""
        return self.logout(logout_all_sessions=True, reason=reason)
