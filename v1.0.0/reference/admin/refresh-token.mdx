---
title: "Refresh Token"
api: "POST /auth/refresh/token"
description: "Refresh an expired access token using a refresh token"
---

Refresh an expired access token using a valid refresh token. This endpoint allows you to obtain a new access token without requiring the user to log in again, providing a seamless authentication experience.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for token renewal and maintaining active sessions.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

<CodeBlock lang="bash">
Authorization: Bearer <admin_jwt_token>
</CodeBlock>

## Request

<ParamField body="refreshToken" type="string" required>
  The refresh token to use for obtaining a new access token
  
  **Example**: "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9..."
  
  **Constraints**: 
  - Must be a valid, non-expired refresh token
  - Must be associated with an active session
  - Must not have been revoked
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Indicates if the token refresh was successful
</ResponseField>

<ResponseField name="data" type="object">
  Contains the new authentication tokens
</ResponseField>

<ResponseField name="data.access_token" type="string">
  New JWT access token for API authentication (expires in 1 hour)
</ResponseField>

<ResponseField name="data.refresh_token" type="string">
  New refresh token for future token renewals (expires in 7 days)
</ResponseField>

<ResponseField name="data.expires_in" type="integer">
  Access token expiration time in seconds (3600 = 1 hour)
</ResponseField>

<ResponseField name="data.token_type" type="string">
  Token type, always "Bearer"
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable message about the operation result
</ResponseField>

<ResponseField name="timestamp" type="string">
  ISO 8601 timestamp of the response
</ResponseField>

## Usage Notes

<AccordionGroup>
  <Accordion title="Token Lifecycle" icon="refresh">
    Token refresh process:
    
    - Access tokens expire after 1 hour
    - Refresh tokens expire after 7 days
    - Both tokens are rotated on refresh
    - Old tokens become invalid immediately
  </Accordion>
  
  <Accordion title="Automatic Refresh" icon="cog">
    Implement automatic token refresh:
    
    - Check token expiry before API calls
    - Refresh tokens proactively (5-10 minutes before expiry)
    - Handle refresh failures gracefully
    - Fall back to login if refresh fails
  </Accordion>
  
  <Accordion title="Security Considerations" icon="shield">
    Security best practices:
    
    - Store refresh tokens securely
    - Use HTTPS for all token operations
    - Implement proper token rotation
    - Monitor for suspicious refresh patterns
  </Accordion>
  
  <Accordion title="Error Handling" icon="exclamation-triangle">
    Handle refresh failures:
    
    - If refresh fails, redirect to login
    - Clear all stored tokens
    - Show appropriate error messages
    - Log refresh failures for monitoring
  </Accordion>
</AccordionGroup>

{`class TokenManager {
  constructor() {
    this.accessToken = null;
    this.refreshToken = null;
    this.tokenExpiry = null;
    this.refreshPromise = null;
  }
  
  async refreshAccessToken() {
    // Prevent multiple simultaneous refresh attempts
    if (this.refreshPromise) {
      return this.refreshPromise;
    }
    
    this.refreshPromise = this.performRefresh();
    
    try {
      const result = await this.refreshPromise;
      return result;
    } finally {
      this.refreshPromise = null;
    }
  }
  
  async performRefresh() {
    if (!this.refreshToken) {
      throw new Error('No refresh token available');
    }
    
    try {
      const response = await fetch('https://api.example.com/auth/refresh/token', {
        method: 'POST',
        headers: {
          'Authorization': \`Bearer \${this.accessToken}\`,
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({ refreshToken: this.refreshToken })
      });
      
      if (!response.ok) {
        throw new Error('Token refresh failed');
      }
      
      const data = await response.json();
      
      // Update tokens
      this.accessToken = data.data.access_token;
      this.refreshToken = data.data.refresh_token;
      this.tokenExpiry = Date.now() + (data.data.expires_in * 1000);
      
      // Store in localStorage
      localStorage.setItem('weir_access_token', this.accessToken);
      localStorage.setItem('weir_refresh_token', this.refreshToken);
      localStorage.setItem('weir_token_expiry', this.tokenExpiry.toString());
      
      return this.accessToken;
      
    } catch (error) {
      // Clear tokens on refresh failure
      this.clearTokens();
      throw error;
    }
  }
  
  isTokenExpired() {
    if (!this.tokenExpiry) return true;
    return Date.now() >= this.tokenExpiry - 60000; // 1 minute buffer
  }
  
  async getValidToken() {
    if (this.isTokenExpired()) {
      await this.refreshAccessToken();
    }
    return this.accessToken;
  }
  
  clearTokens() {
    this.accessToken = null;
    this.refreshToken = null;
    this.tokenExpiry = null;
    
    localStorage.removeItem('weir_access_token');
    localStorage.removeItem('weir_refresh_token');
    localStorage.removeItem('weir_token_expiry');
  }
  
  loadFromStorage() {
    this.accessToken = localStorage.getItem('weir_access_token');
    this.refreshToken = localStorage.getItem('weir_refresh_token');
    this.tokenExpiry = parseInt(localStorage.getItem('weir_token_expiry') || '0');
  }
}

{`class WeirAdminClient {
  constructor(baseURL) {
    this.baseURL = baseURL;
    this.tokenManager = new TokenManager();
    this.tokenManager.loadFromStorage();
  }
  
  async makeRequest(endpoint, options = {}) {
    try {
      const token = await this.tokenManager.getValidToken();
      
      const response = await fetch(\`\${this.baseURL}\${endpoint}\`, {
        ...options,
        headers: {
          'Authorization': \`Bearer \${token}\`,
          'Content-Type': 'application/json',
          ...options.headers
        }
      });
      
      // If token is invalid, try to refresh once
      if (response.status === 401) {
        try {
          await this.tokenManager.refreshAccessToken();
          const newToken = await this.tokenManager.getValidToken();
          
          // Retry the request with new token
          return fetch(\`\${this.baseURL}\${endpoint}\`, {
            ...options,
            headers: {
              'Authorization': \`Bearer \${newToken}\`,
              'Content-Type': 'application/json',
              ...options.headers
            }
          });
        } catch (refreshError) {
          // Refresh failed, redirect to login
          this.redirectToLogin();
          throw new Error('Authentication failed');
        }
      }
      
      return response;
      
    } catch (error) {
      console.error('API request failed:', error);
      throw error;
    }
  }
  
  redirectToLogin() {
    this.tokenManager.clearTokens();
    window.location.href = '/login';
  }
  
  async logout() {
    try {
      if (this.tokenManager.refreshToken) {
        await this.makeRequest('/auth/logout', {
          method: 'POST',
          body: JSON.stringify({ refreshToken: this.tokenManager.refreshToken })
        });
      }
    } catch (error) {
      console.warn('Logout request failed:', error.message);
    } finally {
      this.tokenManager.clearTokens();
      this.redirectToLogin();
    }
  }
}

{`class BackgroundTokenRefresh {
  constructor(tokenManager, refreshIntervalMinutes = 50) {
    this.tokenManager = tokenManager;
    this.refreshIntervalMinutes = refreshIntervalMinutes;
    this.intervalId = null;
    this.isRefreshing = false;
  }
  
  start() {
    // Refresh every 50 minutes (tokens expire in 60 minutes)
    this.intervalId = setInterval(async () => {
      if (!this.isRefreshing && this.tokenManager.refreshToken) {
        await this.refreshToken();
      }
    }, this.refreshIntervalMinutes * 60 * 1000);
  }
  
  stop() {
    if (this.intervalId) {
      clearInterval(this.intervalId);
      this.intervalId = null;
    }
  }
  
  async refreshToken() {
    this.isRefreshing = true;
    
    try {
      await this.tokenManager.refreshAccessToken();
      console.log('Token refreshed successfully in background');
    } catch (error) {
      console.error('Background token refresh failed:', error);
      // Could emit an event or show a notification here
    } finally {
      this.isRefreshing = false;
    }
  }
  
  // Check if we need to refresh when the page becomes visible
  onPageVisible() {
    if (this.tokenManager.isTokenExpired() && this.tokenManager.refreshToken) {
      this.refreshToken();
    }
  }
}

const backgroundRefresh = new BackgroundTokenRefresh(tokenManager);

  if (!document.hidden) {
    backgroundRefresh.onPageVisible();
  }
});
