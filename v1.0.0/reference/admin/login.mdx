title: "Admin Login"
api: "POST /auth/login"
description: "Authenticate admin user and receive access token"
---

Authenticate an admin user and receive access and refresh tokens for administrative operations. This endpoint is used by system administrators to gain access to administrative functions and system management capabilities.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and does not require authentication (it's used to obtain authentication). It's designed for system administrators and super users who need access to administrative functions.
</Note>

## Authentication

This endpoint does not require authentication as it's used to obtain authentication tokens. However, you must provide valid admin credentials in the request body.

## Request

<ParamField body="username" type="string" required>
  Admin username or email address
  
  **Example**: "admin@example.com"
  
  **Constraints**: 
  - Must be a valid email address or username
  - Must belong to an admin user account
  - Case sensitive
</ParamField>

<ParamField body="password" type="string" required>
  Admin password
  
  **Example**: "secure_password"
  
  **Constraints**: 
  - Minimum length: 8 characters
  - Must contain at least one uppercase letter, one lowercase letter, and one number
  - Cannot be empty
</ParamField>

<ParamField body="remember_me" type="boolean" required="false">
  Whether to extend the session duration
  
  **Example**: true
  
  **Default**: false
  
  **Note**: If true, extends both access and refresh token expiration times
</ParamField>

<ParamField body="two_factor_code" type="string" required="false">
  Two-factor authentication code (if 2FA is enabled)
  
  **Example**: "123456"
  
  **Constraints**: 
  - 6-digit numeric code
  - Required only if 2FA is enabled for the admin account
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Indicates if the authentication was successful
</ResponseField>

<ResponseField name="data" type="object">
  Contains the authentication tokens and user information
</ResponseField>

<ResponseField name="data.access_token" type="string">
  JWT access token for API authentication (expires in 1 hour)
</ResponseField>

<ResponseField name="data.refresh_token" type="string">
  Refresh token for obtaining new access tokens (expires in 7 days)
</ResponseField>

<ResponseField name="data.expires_in" type="integer">
  Access token expiration time in seconds (3600 = 1 hour)
</ResponseField>

<ResponseField name="data.token_type" type="string">
  Token type, always "Bearer"
</ResponseField>

<ResponseField name="data.user" type="object">
  Admin user information
</ResponseField>

<ResponseField name="data.user.user_id" type="string">
  Unique identifier for the admin user
</ResponseField>

<ResponseField name="data.user.username" type="string">
  Admin username
</ResponseField>

<ResponseField name="data.user.email" type="string">
  Admin email address
</ResponseField>

<ResponseField name="data.user.role" type="string">
  Admin role level (super_admin, admin, moderator)
</ResponseField>

<ResponseField name="data.user.permissions" type="array">
  List of admin permissions
</ResponseField>

<ResponseField name="data.user.last_login" type="string">
  ISO 8601 timestamp of the last login
</ResponseField>

<ResponseField name="data.user.account_status" type="string">
  Account status (active, suspended, pending)
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable message about the operation result
</ResponseField>

<ResponseField name="timestamp" type="string">
  ISO 8601 timestamp of the response
</ResponseField>

```go Go
package main

import (
    "bytes"
    "encoding/json"
    "fmt"
    "net/http"
)

type LoginRequest struct {
    Username      string `json:"username"`
    Password      string `json:"password"`
    RememberMe    bool   `json:"remember_me,omitempty"`
    TwoFactorCode string `json:"two_factor_code,omitempty"`
}

func main() {
    payload := LoginRequest{
        Username:      "admin@example.com",
        Password:      "secure_password",
        RememberMe:    true,
        TwoFactorCode: "123456",
    }
    
    jsonData, _ := json.Marshal(payload)
    
    req, _ := http.NewRequest("POST", "https://api.example.com/auth/login", bytes.NewBuffer(jsonData))
    req.Header.Set("Content-Type", "application/json")
    
    client := &http.Client{}
    resp, _ := client.Do(req)
    defer resp.Body.Close()
    
    var result map[string]interface{}
    json.NewDecoder(resp.Body).Decode(&result)
    fmt.Printf("%+v\n", result)
}
```

  <Accordion title="Admin Authentication Service" icon="cog">
    ```javascript
    class AdminAuthService {
      constructor(baseUrl) {
        this.baseUrl = baseUrl;
        this.accessToken = null;
        this.refreshToken = null;
        this.user = null;
      }
      
      async login(username, password, rememberMe = false, twoFactorCode = null) {
        try {
          const response = await fetch(`${this.baseUrl}/auth/login`, {
            method: 'POST',
            headers: {
              'Content-Type': 'application/json'
            },
            body: JSON.stringify({
              username,
              password,
              remember_me: rememberMe,
              two_factor_code: twoFactorCode
            })
          });
          
          if (!response.ok) {
            const error = await response.json();
            throw new Error(error.error.message);
          }
          
          const data = await response.json();
          
          // Store tokens and user info
          this.accessToken = data.data.access_token;
          this.refreshToken = data.data.refresh_token;
          this.user = data.data.user;
          
          // Store in localStorage for persistence
          localStorage.setItem('admin_access_token', this.accessToken);
          localStorage.setItem('admin_refresh_token', this.refreshToken);
          localStorage.setItem('admin_user', JSON.stringify(this.user));
          
          return data;
          
        } catch (error) {
          console.error('Admin login failed:', error);
          throw error;
        }
      }
      
      async logout() {
        try {
          if (this.accessToken) {
            await fetch(`${this.baseUrl}/auth/logout`, {
              method: 'POST',
              headers: {
                'Authorization': `Bearer ${this.accessToken}`,
                'Content-Type': 'application/json'
              },
              body: JSON.stringify({
                refreshToken: this.refreshToken
              })
            });
          }
        } catch (error) {
          console.error('Logout error:', error);
        } finally {
          // Clear stored data
          this.accessToken = null;
          this.refreshToken = null;
          this.user = null;
          localStorage.removeItem('admin_access_token');
          localStorage.removeItem('admin_refresh_token');
          localStorage.removeItem('admin_user');
        }
      }
      
      async refreshAccessToken() {
        try {
          const response = await fetch(`${this.baseUrl}/auth/refresh/token`, {
            method: 'POST',
            headers: {
              'Authorization': `Bearer ${this.accessToken}`,
              'Content-Type': 'application/json'
            },
            body: JSON.stringify({
              refreshToken: this.refreshToken
            })
          });
          
          if (!response.ok) {
            throw new Error('Token refresh failed');
          }
          
          const data = await response.json();
          this.accessToken = data.data.access_token;
          this.refreshToken = data.data.refresh_token;
          
          // Update stored tokens
          localStorage.setItem('admin_access_token', this.accessToken);
          localStorage.setItem('admin_refresh_token', this.refreshToken);
          
          return data;
          
        } catch (error) {
          console.error('Token refresh failed:', error);
          // If refresh fails, redirect to login
          this.logout();
          throw error;
        }
      }
      
      isAuthenticated() {
        return this.accessToken !== null;
      }
      
      hasPermission(permission) {
        return this.user && this.user.permissions.includes(permission);
      }
      
      hasRole(role) {
        return this.user && this.user.role === role;
      }
    }
    
    // Usage example
    const adminAuth = new AdminAuthService('https://api.example.com');
    
    // Login
    try {
      const result = await adminAuth.login('admin@example.com', 'secure_password', true);
      console.log('Login successful:', result.data.user);
    } catch (error) {
      console.error('Login failed:', error);
    }
    
    // Check permissions
    if (adminAuth.hasPermission('admin:users:write')) {
      console.log('User can manage users');
    }
    ```

    ```python
    import requests
    import json
    from typing import Optional, Dict, List
    
    class AdminAuthClient:
        def __init__(self, base_url: str):
            self.base_url = base_url
            self.access_token: Optional[str] = None
            self.refresh_token: Optional[str] = None
            self.user: Optional[Dict] = None
        
        def login(self, 
                  username: str, 
                  password: str, 
                  remember_me: bool = False,
                  two_factor_code: Optional[str] = None) -> Dict:
            """Authenticate admin user"""
            try:
                payload = {
                    "username": username,
                    "password": password,
                    "remember_me": remember_me
                }
                
                if two_factor_code:
                    payload["two_factor_code"] = two_factor_code
                
                response = requests.post(
                    f"{self.base_url}/auth/login",
                    headers={"Content-Type": "application/json"},
                    json=payload
                )
                response.raise_for_status()
                
                data = response.json()
                
                # Store tokens and user info
                self.access_token = data["data"]["access_token"]
                self.refresh_token = data["data"]["refresh_token"]
                self.user = data["data"]["user"]
                
                print(f"Login successful for {self.user['username']}")
                return data
                
            except requests.exceptions.RequestException as e:
                print(f"Login failed: {e}")
                raise
        
        def logout(self) -> None:
            """Logout admin user"""
            try:
                if self.access_token:
                    requests.post(
                        f"{self.base_url}/auth/logout",
                        headers={
                            "Authorization": f"Bearer {self.access_token}",
                            "Content-Type": "application/json"
                        },
                        json={"refreshToken": self.refresh_token}
                    )
            except requests.exceptions.RequestException as e:
                print(f"Logout error: {e}")
            finally:
                # Clear stored data
                self.access_token = None
                self.refresh_token = None
                self.user = None
        
        def refresh_token(self) -> Dict:
            """Refresh access token"""
            try:
                response = requests.post(
                    f"{self.base_url}/auth/refresh/token",
                    headers={
                        "Authorization": f"Bearer {self.access_token}",
                        "Content-Type": "application/json"
                    },
                    json={"refreshToken": self.refresh_token}
                )
                response.raise_for_status()
                
                data = response.json()
                self.access_token = data["data"]["access_token"]
                self.refresh_token = data["data"]["refresh_token"]
                
                return data
                
            except requests.exceptions.RequestException as e:
                print(f"Token refresh failed: {e}")
                self.logout()
                raise
        
        def is_authenticated(self) -> bool:
            """Check if user is authenticated"""
            return self.access_token is not None
        
        def has_permission(self, permission: str) -> bool:
            """Check if user has specific permission"""
            return self.user and permission in self.user.get("permissions", [])
        
        def has_role(self, role: str) -> bool:
            """Check if user has specific role"""
            return self.user and self.user.get("role") == role
        
        def make_authenticated_request(self, method: str, endpoint: str, **kwargs) -> requests.Response:
            """Make authenticated request to admin API"""
            if not self.is_authenticated():
                raise Exception("Not authenticated")
            
            headers = kwargs.get("headers", {})
            headers["Authorization"] = f"Bearer {self.access_token}"
            headers["Content-Type"] = "application/json"
            kwargs["headers"] = headers
            
            response = requests.request(method, f"{self.base_url}{endpoint}", **kwargs)
            
            # Handle token expiration
            if response.status_code == 401:
                try:
                    self.refresh_token()
                    headers["Authorization"] = f"Bearer {self.access_token}"
                    response = requests.request(method, f"{self.base_url}{endpoint}", **kwargs)
                except:
                    self.logout()
                    raise
            
            return response
    
    # Usage example
    admin_client = AdminAuthClient("https://api.example.com")
    
    # Login
    try:
        result = admin_client.login("admin@example.com", "secure_password", remember_me=True)
        print(f"Logged in as {result['data']['user']['username']}")
    except Exception as e:
        print(f"Login failed: {e}")
    
    # Check permissions
    if admin_client.has_permission("admin:users:write"):
        print("User can manage users")
    
    # Make authenticated requests
    if admin_client.is_authenticated():
        response = admin_client.make_authenticated_request("GET", "/admin/organizations")
        print(response.json())
    ```
