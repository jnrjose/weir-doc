---
title: "Pause Pod"
api: "POST /pax/pause/pod/{podId}"
description: "Pause a specific pod temporarily"
---

Pause a specific pod temporarily while preserving its state and configuration. This endpoint allows admins to temporarily suspend pod operations without losing data or configuration.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for pod management and control within the Weir AI platform.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Path Parameters

<ParamField path="podId" type="string" required>
  Pod ID to pause
  
  **Example**: "pod_1234567890abcdef"
  
  **Constraints**: 
  - Must be a valid pod ID
  - Pod must exist in the system
</ParamField>

## Request

<ParamField body="pause_reason" type="string" required="false">
  Reason for pausing the pod
  
  **Example**: "Scheduled maintenance"
  
  **Constraints**: 
  - Maximum length: 200 characters
  - Optional but recommended for audit purposes
</ParamField>

<ParamField body="preserve_state" type="boolean" required="false">
  Whether to preserve pod state during pause
  
  **Example**: true
  
  **Default**: true
  
  **Note**: If true, preserves memory and processing state
</ParamField>

<ParamField body="graceful_pause" type="boolean" required="false">
  Whether to pause gracefully
  
  **Example**: true
  
  **Default**: true
  
  **Note**: If true, waits for current operations to complete
</ParamField>

<ParamField body="timeout_seconds" type="integer" required="false">
  Timeout for graceful pause
  
  **Example**: 60
  
  **Default**: 30
  
  **Constraints**: 
  - Must be between 5 and 300 seconds
  - Only used if graceful_pause is true
</ParamField>

## Usage Notes

<AccordionGroup>
  <Accordion title="Pod States" icon="info">
    Pod state transitions:
    
    - **running** → **pausing** → **paused**
    - **paused** → **resuming** → **running**
    - **stopped** → Cannot pause (already stopped)
    - **error** → Cannot pause (in error state)
  </Accordion>
  
  <Accordion title="Pause Types" icon="pause">
    Pause options:
    
    - **Graceful Pause**: Wait for operations to complete
    - **Immediate Pause**: Pause immediately without waiting
    - **State Preservation**: Preserve memory and processing state
    - **Connection Handling**: Handle active connections appropriately
  </Accordion>
  
  <Accordion title="State Management" icon="database">
    State features:
    
    - **Memory Preservation**: Keep data in memory during pause
    - **Processing State**: Preserve current processing state
    - **Connection State**: Maintain connection information
    - **Configuration**: Preserve all configuration settings
  </Accordion>
</AccordionGroup>

## Code Examples

<AccordionGroup>
  <Accordion title="Pod Pause Service" icon="cog">
    ```javascript
    class PodPauseService {
      constructor(apiClient) {
        this.apiClient = apiClient;
      }
      
      async pausePod(podId, options = {}) {
        try {
          const response = await this.apiClient.makeRequest(`/pax/pause/pod/${podId}`, {
            method: 'POST',
            body: JSON.stringify({
              pause_reason: options.pauseReason || 'Pod paused',
              preserve_state: options.preserveState !== false,
              graceful_pause: options.gracefulPause !== false,
              timeout_seconds: options.timeoutSeconds || 30
            })
          });
          
          if (!response.ok) {
            throw new Error('Failed to pause pod');
          }
          
          return await response.json();
          
        } catch (error) {
          console.error('Error pausing pod:', error);
          throw error;
        }
      }
      
      async gracefulPausePod(podId, reason, timeoutSeconds = 60) {
        return this.pausePod(podId, {
          pauseReason: reason,
          preserveState: true,
          gracefulPause: true,
          timeoutSeconds: timeoutSeconds
        });
      }
      
      async immediatePausePod(podId, reason) {
        return this.pausePod(podId, {
          pauseReason: reason,
          preserveState: true,
          gracefulPause: false
        });
      }
      
      async pausePodWithoutState(podId, reason) {
        return this.pausePod(podId, {
          pauseReason: reason,
          preserveState: false,
          gracefulPause: true
        });
      }
      
      async pauseMultiplePods(podIds, reason) {
        const results = [];
        
        for (const podId of podIds) {
          try {
            const result = await this.gracefulPausePod(podId, reason);
            results.push({ success: true, podId, data: result.data });
          } catch (error) {
            results.push({ success: false, podId, error: error.message });
          }
        }
        
        return results;
      }
      
      async pausePodsByTag(tagKey, tagValue, reason) {
        // First get running pods with the tag
        const podsResponse = await this.apiClient.makeRequest(`/pax/pods?tag_filter=${tagKey}:${tagValue}&status=running`);
        const pods = await podsResponse.json();
        
        const podIds = pods.data.map(pod => pod.pod_id);
        return this.pauseMultiplePods(podIds, reason);
      }
      
      async pausePodsByEnvironment(environment, reason) {
        const podsResponse = await this.apiClient.makeRequest(`/pax/pods?environment=${environment}&status=running`);
        const pods = await podsResponse.json();
        
        const podIds = pods.data.map(pod => pod.pod_id);
        return this.pauseMultiplePods(podIds, reason);
      }
      
      async pausePodsForMaintenance(reason = 'Scheduled maintenance') {
        const podsResponse = await this.apiClient.makeRequest('/pax/pods?status=running');
        const pods = await podsResponse.json();
        
        const podIds = pods.data.map(pod => pod.pod_id);
        return this.pauseMultiplePods(podIds, reason);
      }
    }
    
    // Usage example
    const podPauseService = new PodPauseService(apiClient);
    
    // Graceful pause with state preservation
    const pausedPod = await podPauseService.gracefulPausePod(
      'pod_1234567890abcdef',
      'Scheduled maintenance',
      60
    );
    console.log('Pod paused:', pausedPod.data.name);
    
    // Immediate pause
    const immediatePaused = await podPauseService.immediatePausePod(
      'pod_0987654321fedcba',
      'Emergency pause required'
    );
    console.log('Pod immediately paused:', immediatePaused.data.name);
    
    // Pause multiple pods
    const multipleResults = await podPauseService.pauseMultiplePods(
      ['pod_1234567890abcdef', 'pod_0987654321fedcba'],
      'Batch maintenance pause'
    );
    console.log('Multiple pods paused:', multipleResults.length);
    
    // Pause pods by tag
    const tagResults = await podPauseService.pausePodsByTag(
      'department',
      'engineering',
      'Engineering maintenance'
    );
    console.log('Engineering pods paused:', tagResults.length);
    
    // Pause all pods for maintenance
    const maintenanceResults = await podPauseService.pausePodsForMaintenance();
    console.log('All pods paused for maintenance:', maintenanceResults.length);
    ```
