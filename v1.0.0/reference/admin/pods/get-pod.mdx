---
title: "Get Pod"
api: "GET /pax/pod/{podId}"
description: "Retrieve detailed information about a specific pod"
---

Retrieve comprehensive information about a specific pod including configuration, status, resource usage, and health metrics. This endpoint provides detailed pod monitoring and management capabilities.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for pod management and monitoring within the Weir AI platform.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Path Parameters

<ParamField path="podId" type="string" required>
  Pod ID to retrieve
  
  **Example**: "pod_1234567890abcdef"
  
  **Constraints**: 
  - Must be a valid pod ID
  - Pod must exist in the system
</ParamField>

## Query Parameters

<ParamField query="include_usage" type="string" required="false">
  Whether to include detailed resource usage statistics
  
  **Options**: "true", "false"
  
  **Example**: "true"
  
  **Default**: "true"
  
  **Note**: Setting to true will include comprehensive usage metrics
</ParamField>

<ParamField query="include_health" type="string" required="false">
  Whether to include detailed health status information
  
  **Options**: "true", "false"
  
  **Example**: "true"
  
  **Default**: "true"
  
  **Note**: Setting to true will include health monitoring data
</ParamField>

<ParamField query="include_logs" type="string" required="false">
  Whether to include recent log entries
  
  **Options**: "true", "false"
  
  **Example**: "true"
  
  **Default**: "false"
  
  **Note**: Setting to true will include recent log entries
</ParamField>

<ParamField query="include_metrics" type="string" required="false">
  Whether to include performance metrics
  
  **Options**: "true", "false"
  
  **Example**: "true"
  
  **Default**: "false"
  
  **Note**: Setting to true will include performance metrics
</ParamField>

<ParamField query="time_range" type="string" required="false">
  Time range for metrics and logs
  
  **Options**: "1h", "6h", "24h", "7d", "30d"
  
  **Example**: "24h"
  
  **Default**: "1h"
  
  **Note**: Time range for historical data
</ParamField>

  <Accordion title="Pod Detail Service" icon="cog">
    ```javascript
    class PodDetailService {
      constructor(apiClient) {
        this.apiClient = apiClient;
      }
      
      async getPodDetails(podId, options = {}) {
        try {
          const params = new URLSearchParams();
          
          if (options.includeUsage !== false) params.append('include_usage', 'true');
          if (options.includeHealth !== false) params.append('include_health', 'true');
          if (options.includeLogs) params.append('include_logs', 'true');
          if (options.includeMetrics) params.append('include_metrics', 'true');
          if (options.timeRange) params.append('time_range', options.timeRange);
          
          const response = await this.apiClient.makeRequest(
            `/pax/pod/${podId}?${params.toString()}`
          );
          
          if (!response.ok) {
            throw new Error('Failed to get pod details');
          }
          
          return await response.json();
          
        } catch (error) {
          console.error('Error getting pod details:', error);
          throw error;
        }
      }
      
      async getPodOverview(podId) {
        return this.getPodDetails(podId, {
          includeUsage: true,
          includeHealth: true,
          includeLogs: false,
          includeMetrics: false
        });
      }
      
      async getPodWithLogs(podId, timeRange = '1h') {
        return this.getPodDetails(podId, {
          includeUsage: true,
          includeHealth: true,
          includeLogs: true,
          includeMetrics: false,
          timeRange: timeRange
        });
      }
      
      async getPodWithMetrics(podId, timeRange = '24h') {
        return this.getPodDetails(podId, {
          includeUsage: true,
          includeHealth: true,
          includeLogs: false,
          includeMetrics: true,
          timeRange: timeRange
        });
      }
      
      async getPodFullDetails(podId, timeRange = '24h') {
        return this.getPodDetails(podId, {
          includeUsage: true,
          includeHealth: true,
          includeLogs: true,
          includeMetrics: true,
          timeRange: timeRange
        });
      }
      
      async getPodHealthStatus(podId) {
        const result = await this.getPodDetails(podId, {
          includeUsage: false,
          includeHealth: true,
          includeLogs: false,
          includeMetrics: false
        });
        
        return result.data.health;
      }
      
      async getPodResourceUsage(podId) {
        const result = await this.getPodDetails(podId, {
          includeUsage: true,
          includeHealth: false,
          includeLogs: false,
          includeMetrics: false
        });
        
        return result.data.usage;
      }
      
      async getPodScalingHistory(podId) {
        const result = await this.getPodDetails(podId, {
          includeUsage: false,
          includeHealth: false,
          includeLogs: false,
          includeMetrics: false
        });
        
        return result.data.auto_scale.scaling_history;
      }
      
      async getPodRecentLogs(podId, timeRange = '1h') {
        const result = await this.getPodWithLogs(podId, timeRange);
        return result.data.logs;
      }
      
      async getPodPerformanceMetrics(podId, timeRange = '24h') {
        const result = await this.getPodWithMetrics(podId, timeRange);
        return result.data.metrics;
      }
      
      async isPodHealthy(podId) {
        const health = await this.getPodHealthStatus(podId);
        return health.status === 'healthy';
      }
      
      async getPodResourceUtilization(podId) {
        const usage = await this.getPodResourceUsage(podId);
        return {
          cpu: usage.cpu_percent,
          memory: usage.memory_percent,
          storage: usage.storage_percent,
          network: usage.network_usage
        };
      }
    }
    
    // Usage example
    const podDetailService = new PodDetailService(apiClient);
    
    // Get pod overview
    const podOverview = await podDetailService.getPodOverview('pod_1234567890abcdef');
    console.log('Pod Overview:', podOverview.data.name);
    
    // Get pod with logs
    const podWithLogs = await podDetailService.getPodWithLogs('pod_1234567890abcdef', '24h');
    console.log('Recent logs:', podWithLogs.data.logs.length);
    
    // Get pod with metrics
    const podWithMetrics = await podDetailService.getPodWithMetrics('pod_1234567890abcdef', '7d');
    console.log('Performance metrics:', podWithMetrics.data.metrics);
    
    // Check pod health
    const isHealthy = await podDetailService.isPodHealthy('pod_1234567890abcdef');
    console.log('Pod is healthy:', isHealthy);
    
    // Get resource utilization
    const utilization = await podDetailService.getPodResourceUtilization('pod_1234567890abcdef');
    console.log('Resource utilization:', utilization);
    
    // Get scaling history
    const scalingHistory = await podDetailService.getPodScalingHistory('pod_1234567890abcdef');
    console.log('Scaling events:', scalingHistory.length);
    ```
