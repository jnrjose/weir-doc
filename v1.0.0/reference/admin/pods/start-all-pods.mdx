---
title: "Start All Pods"
api: "POST /pax/start/pods"
description: "Start all pods in the system"
---

Start all pods in the system that are currently stopped or in maintenance mode. This endpoint allows admins to activate all pods for processing tasks and workloads.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for pod management and control within the Weir AI platform.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Request

<ParamField body="start_reason" type="string" required="false">
  Reason for starting all pods
  
  **Example**: "System maintenance completed"
  
  **Constraints**: 
  - Maximum length: 200 characters
  - Optional but recommended for audit purposes
</ParamField>

<ParamField body="environment" type="string" required="false">
  Filter pods by environment to start
  
  **Options**: "production", "staging", "development", "testing"
  
  **Example**: "production"
  
  **Note**: If not specified, starts pods from all environments
</ParamField>

<ParamField body="region" type="string" required="false">
  Filter pods by region to start
  
  **Example**: "us-east-1"
  
  **Note**: If not specified, starts pods from all regions
</ParamField>

<ParamField body="pod_type" type="string" required="false">
  Filter pods by type to start
  
  **Options**: "processing", "storage", "compute", "analytics", "custom"
  
  **Example**: "processing"
  
  **Note**: If not specified, starts pods of all types
</ParamField>

<ParamField body="tag_filter" type="string" required="false">
  Filter pods by tag key-value pairs to start
  
  **Example**: "department:engineering,priority:high"
  
  **Format**: "key:value,key:value"
</ParamField>

<ParamField body="force_start" type="boolean" required="false">
  Force start even if pods have issues
  
  **Example**: false
  
  **Default**: false
  
  **Note**: If true, starts pods even if health checks fail
</ParamField>

<ParamField body="wait_for_ready" type="boolean" required="false">
  Wait for all pods to be ready before returning
  
  **Example**: true
  
  **Default**: false
  
  **Note**: If true, waits for all pods to be fully operational
</ParamField>

<ParamField body="timeout_seconds" type="integer" required="false">
  Timeout for waiting for all pods to be ready
  
  **Example**: 600
  
  **Default**: 300
  
  **Constraints**: 
  - Must be between 60 and 1800 seconds
  - Only used if wait_for_ready is true
</ParamField>

<ParamField body="parallel_start" type="boolean" required="false">
  Start pods in parallel
  
  **Example**: true
  
  **Default**: true
  
  **Note**: If true, starts multiple pods simultaneously
</ParamField>

<ParamField body="max_parallel" type="integer" required="false">
  Maximum number of pods to start in parallel
  
  **Example**: 10
  
  **Default**: 5
  
  **Constraints**: 
  - Must be between 1 and 20
  - Only used if parallel_start is true
</ParamField>

## Code Examples

<AccordionGroup>
  <Accordion title="Bulk Pod Start Service" icon="cog">
    ```javascript
    class BulkPodStartService {
      constructor(apiClient) {
        this.apiClient = apiClient;
      }
      
      async startAllPods(options = {}) {
        try {
          const response = await this.apiClient.makeRequest('/pax/start/pods', {
            method: 'POST',
            body: JSON.stringify({
              start_reason: options.startReason || 'Bulk pod start',
              environment: options.environment,
              region: options.region,
              pod_type: options.podType,
              tag_filter: options.tagFilter,
              force_start: options.forceStart || false,
              wait_for_ready: options.waitForReady || false,
              timeout_seconds: options.timeoutSeconds || 300,
              parallel_start: options.parallelStart !== false,
              max_parallel: options.maxParallel || 5
            })
          });
          
          if (!response.ok) {
            throw new Error('Failed to start all pods');
          }
          
          return await response.json();
          
        } catch (error) {
          console.error('Error starting all pods:', error);
          throw error;
        }
      }
      
      async startProductionPods(reason = 'Production startup') {
        return this.startAllPods({
          startReason: reason,
          environment: 'production',
          waitForReady: true,
          timeoutSeconds: 600,
          parallelStart: true,
          maxParallel: 10
        });
      }
      
      async startPodsByRegion(region, reason) {
        return this.startAllPods({
          startReason: reason,
          region: region,
          waitForReady: true,
          timeoutSeconds: 300,
          parallelStart: true,
          maxParallel: 8
        });
      }
      
      async startPodsByType(podType, reason) {
        return this.startAllPods({
          startReason: reason,
          podType: podType,
          waitForReady: true,
          timeoutSeconds: 300,
          parallelStart: true,
          maxParallel: 6
        });
      }
      
      async startPodsByTag(tagKey, tagValue, reason) {
        return this.startAllPods({
          startReason: reason,
          tagFilter: `${tagKey}:${tagValue}`,
          waitForReady: true,
          timeoutSeconds: 300,
          parallelStart: true,
          maxParallel: 5
        });
      }
      
      async startAllPodsWithWait(reason, timeoutSeconds = 600) {
        return this.startAllPods({
          startReason: reason,
          waitForReady: true,
          timeoutSeconds: timeoutSeconds,
          parallelStart: true,
          maxParallel: 10
        });
      }
      
      async startAllPodsParallel(reason, maxParallel = 10) {
        return this.startAllPods({
          startReason: reason,
          waitForReady: false,
          parallelStart: true,
          maxParallel: maxParallel
        });
      }
      
      async startPodsForMaintenance(reason = 'Post-maintenance startup') {
        return this.startAllPods({
          startReason: reason,
          waitForReady: true,
          timeoutSeconds: 600,
          parallelStart: true,
          maxParallel: 15
        });
      }
      
      async startPodsByDepartment(department, reason) {
        return this.startPodsByTag('department', department, reason);
      }
      
      async startPodsByProject(project, reason) {
        return this.startPodsByTag('project', project, reason);
      }
      
      async startPodsByPriority(priority, reason) {
        return this.startPodsByTag('priority', priority, reason);
      }
    }
    
    // Usage example
    const bulkStartService = new BulkPodStartService(apiClient);
    
    // Start all production pods
    const productionResult = await bulkStartService.startProductionPods('System maintenance completed');
    console.log(`Started ${productionResult.data.pods_started} production pods`);
    
    // Start pods by region
    const regionResult = await bulkStartService.startPodsByRegion('us-east-1', 'Regional startup');
    console.log(`Started ${regionResult.data.pods_started} pods in us-east-1`);
    
    // Start pods by type
    const typeResult = await bulkStartService.startPodsByType('processing', 'Processing pods startup');
    console.log(`Started ${typeResult.data.pods_started} processing pods`);
    
    // Start pods by department
    const deptResult = await bulkStartService.startPodsByDepartment('engineering', 'Engineering startup');
    console.log(`Started ${deptResult.data.pods_started} engineering pods`);
    
    // Start all pods with wait
    const allResult = await bulkStartService.startAllPodsWithWait('Full system startup', 900);
    console.log(`Started ${allResult.data.pods_started} pods total`);
    
    // Start pods for maintenance
    const maintenanceResult = await bulkStartService.startPodsForMaintenance();
    console.log(`Post-maintenance startup: ${maintenanceResult.data.pods_started} pods started`);
    ```
