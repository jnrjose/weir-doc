---
title: "Create Pod"
api: "POST /pax/pod"
description: "Create a new pod for processing and execution"
---

Create a new pod with specified configuration and resources. This endpoint allows admins to provision new pods for various processing tasks and workloads.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for pod management and resource provisioning within the Weir AI platform.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Request

<ParamField body="name" type="string" required>
  Pod name
  
  **Example**: "processing-pod-001"
  
  **Constraints**: 
  - Maximum length: 50 characters
  - Must be unique
  - Alphanumeric and hyphens only
</ParamField>

<ParamField body="type" type="string" required>
  Pod type
  
  **Options**: "processing", "storage", "compute", "analytics", "custom"
  
  **Example**: "processing"
</ParamField>

<ParamField body="configuration" type="object" required>
  Pod configuration settings
</ParamField>

<ParamField body="configuration.cpu_cores" type="integer" required>
  Number of CPU cores
  
  **Example**: 4
  
  **Constraints**: 
  - Must be between 1 and 32
  - Determines processing power
</ParamField>

<ParamField body="configuration.memory_gb" type="integer" required>
  Memory allocation in GB
  
  **Example**: 8
  
  **Constraints**: 
  - Must be between 1 and 128
  - Determines available memory
</ParamField>

<ParamField body="configuration.storage_gb" type="integer" required>
  Storage allocation in GB
  
  **Example**: 100
  
  **Constraints**: 
  - Must be between 10 and 1000
  - Determines available storage
</ParamField>

<ParamField body="configuration.network_bandwidth" type="integer" required="false">
  Network bandwidth in Mbps
  
  **Example**: 1000
  
  **Default**: 100
  
  **Constraints**: 
  - Must be between 10 and 10000
</ParamField>

<ParamField body="environment" type="string" required="false">
  Environment for the pod
  
  **Options**: "production", "staging", "development", "testing"
  
  **Example**: "production"
  
  **Default**: "production"
</ParamField>

<ParamField body="region" type="string" required="false">
  Deployment region
  
  **Example**: "us-east-1"
  
  **Default**: "us-east-1"
</ParamField>

<ParamField body="auto_scale" type="boolean" required="false">
  Enable auto-scaling
  
  **Example**: true
  
  **Default**: false
</ParamField>

<ParamField body="auto_scale.min_instances" type="integer" required="false">
  Minimum number of instances for auto-scaling
  
  **Example**: 1
  
  **Default**: 1
  
  **Constraints**: 
  - Must be between 1 and 10
  - Only used if auto_scale is true
</ParamField>

<ParamField body="auto_scale.max_instances" type="integer" required="false">
  Maximum number of instances for auto-scaling
  
  **Example**: 5
  
  **Default**: 3
  
  **Constraints**: 
  - Must be between 1 and 20
  - Only used if auto_scale is true
</ParamField>

<ParamField body="tags" type="object" required="false">
  Tags for pod identification and organization
  
  **Example**: {"department": "engineering", "project": "analytics", "priority": "high"}
</ParamField>

<ParamField body="start_immediately" type="boolean" required="false">
  Whether to start the pod immediately after creation
  
  **Example**: true
  
  **Default**: false
</ParamField>

## Code Examples

<AccordionGroup>
  <Accordion title="Pod Management Service" icon="cog">
    ```javascript
    class PodManagementService {
      constructor(apiClient) {
        this.apiClient = apiClient;
      }
      
      async createPod(podData) {
        try {
          const response = await this.apiClient.makeRequest('/pax/pod', {
            method: 'POST',
            body: JSON.stringify(podData)
          });
          
          if (!response.ok) {
            throw new Error('Failed to create pod');
          }
          
          return await response.json();
          
        } catch (error) {
          console.error('Error creating pod:', error);
          throw error;
        }
      }
      
      async createProcessingPod(name, cpuCores, memoryGB, storageGB) {
        return this.createPod({
          name: name,
          type: 'processing',
          configuration: {
            cpu_cores: cpuCores,
            memory_gb: memoryGB,
            storage_gb: storageGB,
            network_bandwidth: 1000
          },
          environment: 'production',
          auto_scale: {
            enabled: true,
            min_instances: 1,
            max_instances: 5
          },
          start_immediately: true
        });
      }
      
      async createStoragePod(name, storageGB) {
        return this.createPod({
          name: name,
          type: 'storage',
          configuration: {
            cpu_cores: 2,
            memory_gb: 4,
            storage_gb: storageGB,
            network_bandwidth: 500
          },
          environment: 'production',
          start_immediately: true
        });
      }
      
      async createAnalyticsPod(name, cpuCores, memoryGB) {
        return this.createPod({
          name: name,
          type: 'analytics',
          configuration: {
            cpu_cores: cpuCores,
            memory_gb: memoryGB,
            storage_gb: 200,
            network_bandwidth: 2000
          },
          environment: 'production',
          auto_scale: {
            enabled: true,
            min_instances: 2,
            max_instances: 10
          },
          tags: {
            department: 'analytics',
            priority: 'high'
          },
          start_immediately: true
        });
      }
      
      async createDevelopmentPod(name) {
        return this.createPod({
          name: name,
          type: 'custom',
          configuration: {
            cpu_cores: 2,
            memory_gb: 4,
            storage_gb: 50,
            network_bandwidth: 100
          },
          environment: 'development',
          tags: {
            department: 'engineering',
            purpose: 'development'
          },
          start_immediately: false
        });
      }
    }
    
    // Usage example
    const podService = new PodManagementService(apiClient);
    
    // Create processing pod
    const processingPod = await podService.createProcessingPod(
      'analytics-processor-001',
      8,
      16,
      500
    );
    console.log('Processing pod created:', processingPod.data.pod_id);
    
    // Create storage pod
    const storagePod = await podService.createStoragePod(
      'data-storage-001',
      1000
    );
    console.log('Storage pod created:', storagePod.data.pod_id);
    
    // Create analytics pod
    const analyticsPod = await podService.createAnalyticsPod(
      'ml-analytics-001',
      16,
      32
    );
    console.log('Analytics pod created:', analyticsPod.data.pod_id);
    ```
