---
title: "Get Pods"
api: "GET /pax/pods"
description: "Retrieve list of all pods in the system"
---

Retrieve a comprehensive list of all pods with their configuration, status, and resource usage information. This endpoint provides admin-level visibility into pod management and monitoring.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for pod management and monitoring within the Weir AI platform.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Query Parameters

<ParamField query="page" type="string" required="false">
  Page number for pagination
  
  **Example**: "1"
  
  **Default**: "1"
</ParamField>

<ParamField query="limit" type="string" required="false">
  Number of pods to return per page
  
  **Example**: "20"
  
  **Default**: "20"
</ParamField>

<ParamField query="status" type="string" required="false">
  Filter pods by status
  
  **Options**: "running", "stopped", "starting", "stopping", "error", "maintenance"
  
  **Example**: "running"
  
  **Default**: "running"
</ParamField>

<ParamField query="type" type="string" required="false">
  Filter pods by type
  
  **Options**: "processing", "storage", "compute", "analytics", "custom"
  
  **Example**: "processing"
</ParamField>

<ParamField query="environment" type="string" required="false">
  Filter pods by environment
  
  **Options**: "production", "staging", "development", "testing"
  
  **Example**: "production"
</ParamField>

<ParamField query="region" type="string" required="false">
  Filter pods by region
  
  **Example**: "us-east-1"
</ParamField>

<ParamField query="search" type="string" required="false">
  Search pods by name or tags
  
  **Example**: "analytics"
  
  **Constraints**: 
  - Minimum length: 2 characters
  - Case-insensitive search
</ParamField>

<ParamField query="sort_by" type="string" required="false">
  Sort pods by specified field
  
  **Options**: "name", "created_at", "updated_at", "status", "type", "cpu_usage", "memory_usage"
  
  **Example**: "created_at"
  
  **Default**: "created_at"
</ParamField>

<ParamField query="sort_order" type="string" required="false">
  Sort order for the results
  
  **Options**: "asc", "desc"
  
  **Example**: "desc"
  
  **Default**: "desc"
</ParamField>

<ParamField query="include_usage" type="string" required="false">
  Whether to include resource usage statistics
  
  **Options**: "true", "false"
  
  **Example**: "true"
  
  **Default**: "false"
</ParamField>

<ParamField query="include_health" type="string" required="false">
  Whether to include health status information
  
  **Options**: "true", "false"
  
  **Example**: "true"
  
  **Default**: "false"
</ParamField>

<ParamField query="tag_filter" type="string" required="false">
  Filter pods by tag key-value pairs
  
  **Example**: "department:engineering,priority:high"
  
  **Format**: "key:value,key:value"
</ParamField>

## Code Examples

<AccordionGroup>
  <Accordion title="Pod Monitoring Service" icon="cog">
    ```javascript
    class PodMonitoringService {
      constructor(apiClient) {
        this.apiClient = apiClient;
      }
      
      async getPods(options = {}) {
        try {
          const params = new URLSearchParams();
          
          if (options.page) params.append('page', options.page);
          if (options.limit) params.append('limit', options.limit);
          if (options.status) params.append('status', options.status);
          if (options.type) params.append('type', options.type);
          if (options.environment) params.append('environment', options.environment);
          if (options.region) params.append('region', options.region);
          if (options.search) params.append('search', options.search);
          if (options.sortBy) params.append('sort_by', options.sortBy);
          if (options.sortOrder) params.append('sort_order', options.sortOrder);
          if (options.includeUsage) params.append('include_usage', options.includeUsage);
          if (options.includeHealth) params.append('include_health', options.includeHealth);
          if (options.tagFilter) params.append('tag_filter', options.tagFilter);
          
          const response = await this.apiClient.makeRequest(
            `/pax/pods?${params.toString()}`
          );
          
          if (!response.ok) {
            throw new Error('Failed to get pods');
          }
          
          return await response.json();
          
        } catch (error) {
          console.error('Error getting pods:', error);
          throw error;
        }
      }
      
      async getRunningPods() {
        return this.getPods({
          status: 'running',
          includeUsage: true,
          includeHealth: true,
          sortBy: 'name',
          sortOrder: 'asc'
        });
      }
      
      async getPodsByType(type) {
        return this.getPods({
          type: type,
          status: 'running',
          includeUsage: true,
          includeHealth: true
        });
      }
      
      async getProcessingPods() {
        return this.getPodsByType('processing');
      }
      
      async getStoragePods() {
        return this.getPodsByType('storage');
      }
      
      async getAnalyticsPods() {
        return this.getPodsByType('analytics');
      }
      
      async getPodsByEnvironment(environment) {
        return this.getPods({
          environment: environment,
          includeUsage: true,
          includeHealth: true
        });
      }
      
      async getProductionPods() {
        return this.getPodsByEnvironment('production');
      }
      
      async getDevelopmentPods() {
        return this.getPodsByEnvironment('development');
      }
      
      async getPodsWithHighUsage() {
        const result = await this.getPods({
          includeUsage: true,
          sortBy: 'cpu_usage',
          sortOrder: 'desc'
        });
        
        return result.data.filter(pod => 
          pod.usage && pod.usage.cpu_percent > 80
        );
      }
      
      async getPodsWithHealthIssues() {
        const result = await this.getPods({
          includeHealth: true
        });
        
        return result.data.filter(pod => 
          pod.health && pod.health.status !== 'healthy'
        );
      }
      
      async getPodsByTag(tagKey, tagValue) {
        return this.getPods({
          tagFilter: `${tagKey}:${tagValue}`,
          includeUsage: true,
          includeHealth: true
        });
      }
      
      async getPodsByDepartment(department) {
        return this.getPodsByTag('department', department);
      }
      
      async getPodsByProject(project) {
        return this.getPodsByTag('project', project);
      }
      
      async getPodStatistics() {
        const result = await this.getPods({
          limit: 1
        });
        
        return result.summary;
      }
      
      async searchPods(searchTerm) {
        return this.getPods({
          search: searchTerm,
          sortBy: 'name',
          sortOrder: 'asc'
        });
      }
    }
    
    // Usage example
    const podService = new PodMonitoringService(apiClient);
    
    // Get running pods
    const runningPods = await podService.getRunningPods();
    console.log(`Found ${runningPods.data.length} running pods`);
    
    // Get processing pods
    const processingPods = await podService.getProcessingPods();
    console.log(`Found ${processingPods.data.length} processing pods`);
    
    // Get pods with high usage
    const highUsagePods = await podService.getPodsWithHighUsage();
    console.log(`Found ${highUsagePods.length} pods with high usage`);
    
    // Get pods with health issues
    const unhealthyPods = await podService.getPodsWithHealthIssues();
    console.log(`Found ${unhealthyPods.length} pods with health issues`);
    
    // Get pods by department
    const engineeringPods = await podService.getPodsByDepartment('engineering');
    console.log(`Found ${engineeringPods.data.length} engineering pods`);
    
    // Get pod statistics
    const stats = await podService.getPodStatistics();
    console.log('Pod Statistics:', stats);
    ```
