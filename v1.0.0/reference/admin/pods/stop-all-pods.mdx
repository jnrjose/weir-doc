---
title: "Stop All Pods"
api: "POST /pax/stop/pods"
description: "Stop all pods in the system completely"
---

Stop all pods in the system completely, terminating all processes and freeing up resources. This endpoint allows admins to shut down all pods permanently or temporarily.

<Note>
  **Admin API Endpoint**
  
  This endpoint is part of the Admin API category and requires Bearer Token authentication. It's designed for pod management and control within the Weir AI platform.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
Authorization: Bearer <your_jwt_token>
```

## Request

<ParamField body="stop_reason" type="string" required="false">
  Reason for stopping all pods
  
  **Example**: "System shutdown"
  
  **Constraints**: 
  - Maximum length: 200 characters
  - Optional but recommended for audit purposes
</ParamField>

<ParamField body="environment" type="string" required="false">
  Filter pods by environment to stop
  
  **Options**: "production", "staging", "development", "testing"
  
  **Example**: "production"
  
  **Note**: If not specified, stops pods from all environments
</ParamField>

<ParamField body="region" type="string" required="false">
  Filter pods by region to stop
  
  **Example**: "us-east-1"
  
  **Note**: If not specified, stops pods from all regions
</ParamField>

<ParamField body="pod_type" type="string" required="false">
  Filter pods by type to stop
  
  **Options**: "processing", "storage", "compute", "analytics", "custom"
  
  **Example**: "processing"
  
  **Note**: If not specified, stops pods of all types
</ParamField>

<ParamField body="tag_filter" type="string" required="false">
  Filter pods by tag key-value pairs to stop
  
  **Example**: "department:engineering,priority:high"
  
  **Format**: "key:value,key:value"
</ParamField>

<ParamField body="graceful_stop" type="boolean" required="false">
  Whether to stop gracefully
  
  **Example**: true
  
  **Default**: true
  
  **Note**: If true, waits for current operations to complete
</ParamField>

<ParamField body="timeout_seconds" type="integer" required="false">
  Timeout for graceful stop
  
  **Example**: 300
  
  **Default**: 60
  
  **Constraints**: 
  - Must be between 30 and 1800 seconds
  - Only used if graceful_stop is true
</ParamField>

<ParamField body="preserve_data" type="boolean" required="false">
  Whether to preserve data during stop
  
  **Example**: true
  
  **Default**: true
  
  **Note**: If true, preserves data for future restart
</ParamField>

<ParamField body="force_stop" type="boolean" required="false">
  Force stop even if operations are running
  
  **Example**: false
  
  **Default**: false
  
  **Note**: If true, stops immediately without waiting
</ParamField>

<ParamField body="parallel_stop" type="boolean" required="false">
  Stop pods in parallel
  
  **Example**: true
  
  **Default**: true
  
  **Note**: If true, stops multiple pods simultaneously
</ParamField>

<ParamField body="max_parallel" type="integer" required="false">
  Maximum number of pods to stop in parallel
  
  **Example**: 10
  
  **Default**: 5
  
  **Constraints**: 
  - Must be between 1 and 20
  - Only used if parallel_stop is true
</ParamField>

<AccordionGroup>
  <Accordion title="Bulk Pod Stop Service" icon="cog">
    ```javascript
    class BulkPodStopService {
      constructor(apiClient) {
        this.apiClient = apiClient;
      }
      
      async stopAllPods(options = {}) {
        try {
          const response = await this.apiClient.makeRequest('/pax/stop/pods', {
            method: 'POST',
            body: JSON.stringify({
              stop_reason: options.stopReason || 'Bulk pod stop',
              environment: options.environment,
              region: options.region,
              pod_type: options.podType,
              tag_filter: options.tagFilter,
              graceful_stop: options.gracefulStop !== false,
              timeout_seconds: options.timeoutSeconds || 60,
              preserve_data: options.preserveData !== false,
              force_stop: options.forceStop || false,
              parallel_stop: options.parallelStop !== false,
              max_parallel: options.maxParallel || 5
            })
          });
          
          if (!response.ok) {
            throw new Error('Failed to stop all pods');
          }
          
          return await response.json();
          
        } catch (error) {
          console.error('Error stopping all pods:', error);
          throw error;
        }
      }
      
      async stopProductionPods(reason = 'Production shutdown') {
        return this.stopAllPods({
          stopReason: reason,
          environment: 'production',
          gracefulStop: true,
          timeoutSeconds: 300,
          preserveData: true,
          parallelStop: true,
          maxParallel: 10
        });
      }
      
      async stopPodsByRegion(region, reason) {
        return this.stopAllPods({
          stopReason: reason,
          region: region,
          gracefulStop: true,
          timeoutSeconds: 180,
          preserveData: true,
          parallelStop: true,
          maxParallel: 8
        });
      }
      
      async stopPodsByType(podType, reason) {
        return this.stopAllPods({
          stopReason: reason,
          podType: podType,
          gracefulStop: true,
          timeoutSeconds: 180,
          preserveData: true,
          parallelStop: true,
          maxParallel: 6
        });
      }
      
      async stopPodsByTag(tagKey, tagValue, reason) {
        return this.stopAllPods({
          stopReason: reason,
          tagFilter: `${tagKey}:${tagValue}`,
          gracefulStop: true,
          timeoutSeconds: 120,
          preserveData: true,
          parallelStop: true,
          maxParallel: 5
        });
      }
      
      async stopAllPodsGraceful(reason, timeoutSeconds = 300) {
        return this.stopAllPods({
          stopReason: reason,
          gracefulStop: true,
          timeoutSeconds: timeoutSeconds,
          preserveData: true,
          parallelStop: true,
          maxParallel: 10
        });
      }
      
      async stopAllPodsForce(reason) {
        return this.stopAllPods({
          stopReason: reason,
          gracefulStop: false,
          forceStop: true,
          preserveData: true,
          parallelStop: true,
          maxParallel: 15
        });
      }
      
      async stopPodsForMaintenance(reason = 'Scheduled maintenance') {
        return this.stopAllPods({
          stopReason: reason,
          gracefulStop: true,
          timeoutSeconds: 300,
          preserveData: true,
          parallelStop: true,
          maxParallel: 15
        });
      }
      
      async emergencyStopAllPods(reason = 'Emergency shutdown') {
        return this.stopAllPods({
          stopReason: reason,
          gracefulStop: false,
          forceStop: true,
          preserveData: false,
          parallelStop: true,
          maxParallel: 20
        });
      }
      
      async stopPodsByDepartment(department, reason) {
        return this.stopPodsByTag('department', department, reason);
      }
      
      async stopPodsByProject(project, reason) {
        return this.stopPodsByTag('project', project, reason);
      }
      
      async stopPodsByPriority(priority, reason) {
        return this.stopPodsByTag('priority', priority, reason);
      }
    }
    
    // Usage example
    const bulkStopService = new BulkPodStopService(apiClient);
    
    // Stop all production pods
    const productionResult = await bulkStopService.stopProductionPods('System shutdown');
    console.log(`Stopped ${productionResult.data.pods_stopped} production pods`);
    
    // Stop pods by region
    const regionResult = await bulkStopService.stopPodsByRegion('us-east-1', 'Regional shutdown');
    console.log(`Stopped ${regionResult.data.pods_stopped} pods in us-east-1`);
    
    // Stop pods by type
    const typeResult = await bulkStopService.stopPodsByType('processing', 'Processing shutdown');
    console.log(`Stopped ${typeResult.data.pods_stopped} processing pods`);
    
    // Stop pods by department
    const deptResult = await bulkStopService.stopPodsByDepartment('engineering', 'Engineering shutdown');
    console.log(`Stopped ${deptResult.data.pods_stopped} engineering pods`);
    
    // Stop all pods gracefully
    const gracefulResult = await bulkStopService.stopAllPodsGraceful('System-wide shutdown', 600);
    console.log(`Stopped ${gracefulResult.data.pods_stopped} pods gracefully`);
    
    // Emergency stop all pods
    const emergencyResult = await bulkStopService.emergencyStopAllPods();
    console.log(`Emergency stop: ${emergencyResult.data.pods_stopped} pods stopped`);
    ```
