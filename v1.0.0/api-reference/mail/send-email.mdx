---
title: "Send Email"
api: "POST /mail/send"
description: "Send email notifications and communications"
---

Send email notifications, invitations, alerts, and other communications through the Weir AI platform. This endpoint supports various email types including team invitations, system notifications, and custom messages.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication. It's designed for email communication and notifications.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

<CodeBlock lang="bash">
Authorization: Bearer <your_jwt_token>
</CodeBlock>

## Request

<ParamField body="to" type="array" required>
  List of recipient email addresses
  
  **Example**: ["user@example.com", "admin@example.com"]
  
  **Constraints**: 
  - Must be valid email addresses
  - Maximum 50 recipients per request
</ParamField>

<ParamField body="subject" type="string" required>
  Email subject line
  
  **Example**: "Welcome to Weir AI Platform"
  
  **Constraints**: 
  - Maximum length: 200 characters
  - Must not be empty
</ParamField>

<ParamField body="template" type="string" required>
  Email template to use
  
  **Options**: "welcome", "invitation", "notification", "alert", "custom"
  
  **Example**: "welcome"
</ParamField>

<ParamField body="variables" type="object" required="false">
  Template variables for dynamic content
  
  **Example**: {"user_name": "John Doe", "team_name": "Development Team"}
</ParamField>

<ParamField body="priority" type="string" required="false">
  Email priority level
  
  **Options**: "low", "normal", "high", "urgent"
  
  **Default**: "normal"
</ParamField>

<ParamField body="schedule_at" type="string" required="false">
  Schedule email for future delivery (ISO 8601 format)
  
  **Example**: "2024-01-20T09:00:00Z"
</ParamField>

<ParamField body="attachments" type="array" required="false">
  Email attachments
  
  **Properties**:
  - `filename` (string): Name of the attachment
  - `content` (string): Base64 encoded file content
  - `mime_type` (string): MIME type of the file
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Indicates if the email was sent successfully
</ResponseField>

<ResponseField name="data" type="object">
  Contains email delivery information
</ResponseField>

<ResponseField name="data.message_id" type="string">
  Unique identifier for the email message
</ResponseField>

<ResponseField name="data.recipients" type="array">
  List of recipient email addresses
</ResponseField>

<ResponseField name="data.subject" type="string">
  Email subject line
</ResponseField>

<ResponseField name="data.template" type="string">
  Email template used
</ResponseField>

<ResponseField name="data.status" type="string">
  Email status (sent, scheduled, failed)
</ResponseField>

<ResponseField name="data.scheduled_at" type="string">
  ISO 8601 timestamp when email is scheduled (if applicable)
</ResponseField>

<ResponseField name="data.sent_at" type="string">
  ISO 8601 timestamp when email was sent
</ResponseField>

<ResponseField name="data.delivery_status" type="object">
  Delivery status for each recipient
</ResponseField>

<ResponseField name="data.delivery_status.delivered" type="array">
  List of successfully delivered emails
</ResponseField>

<ResponseField name="data.delivery_status.failed" type="array">
  List of failed deliveries with error details
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable message about the operation result
</ResponseField>

<ResponseField name="timestamp" type="string">
  ISO 8601 timestamp of the response
</ResponseField>

## Usage Notes

<AccordionGroup>
  <Accordion title="Email Templates" icon="template">
    Available email templates:
    
    - **welcome**: Welcome new users to the platform
    - **invitation**: Team and organization invitations
    - **notification**: System notifications and alerts
    - **alert**: Critical alerts and warnings
    - **custom**: Custom HTML content
  </Accordion>
  
  <Accordion title="Rate Limiting" icon="clock">
    Email sending limits:
    
    - Maximum 100 emails per hour per user
    - Maximum 1000 emails per day per organization
    - Rate limit resets after the specified time
  </Accordion>
  
  <Accordion title="Permissions" icon="shield">
    To send emails, you need:
    
    - Valid authentication token
    - Email sending permissions
    - Appropriate role level
  </Accordion>
  
  <Accordion title="Template Variables" icon="code">
    Dynamic content support:
    
    - Use variables in email templates
    - Support for user data, team info, and custom fields
    - Automatic variable replacement
    - Fallback values for missing variables
  </Accordion>
</AccordionGroup>

## Integration Examples

<AccordionGroup>
  <Accordion title="Email Service Class" icon="code">
    <CodeBlock lang="javascript">
class EmailService {
  constructor(apiClient) {
    this.apiClient = apiClient;
  }
  
  async sendWelcomeEmail(userEmail, userName, teamName) {
    try {
      const response = await this.apiClient.makeRequest('/mail/send', {
        method: 'POST',
        body: JSON.stringify({
          to: [userEmail],
          subject: 'Welcome to Weir AI Platform',
          template: 'welcome',
          variables: {
            user_name: userName,
            team_name: teamName
          },
          priority: 'normal'
        })
      });
      
      if (!response.ok) {
        throw new Error('Failed to send welcome email');
      }
      
      return await response.json();
      
    } catch (error) {
      console.error('Error sending welcome email:', error);
      throw error;
    }
  }
  
  async sendTeamInvitation(emails, teamName, inviterName) {
    try {
      const response = await this.apiClient.makeRequest('/mail/send', {
        method: 'POST',
        body: JSON.stringify({
          to: emails,
          subject: 'Invitation to join ' + teamName,
          template: 'invitation',
          variables: {
            team_name: teamName,
            inviter_name: inviterName
          },
          priority: 'normal'
        })
      });
      
      if (!response.ok) {
        throw new Error('Failed to send team invitation');
      }
      
      return await response.json();
      
    } catch (error) {
      console.error('Error sending team invitation:', error);
      throw error;
    }
  }
  
  async sendNotification(emails, subject, message, priority = 'normal') {
    try {
      const response = await this.apiClient.makeRequest('/mail/send', {
        method: 'POST',
        body: JSON.stringify({
          to: emails,
          subject: subject,
          template: 'notification',
          variables: {
            message: message
          },
          priority: priority
        })
      });
      
      if (!response.ok) {
        throw new Error('Failed to send notification');
      }
      
      return await response.json();
      
    } catch (error) {
      console.error('Error sending notification:', error);
      throw error;
    }
  }
  
  async scheduleEmail(emails, subject, template, variables, scheduleAt) {
    try {
      const response = await this.apiClient.makeRequest('/mail/send', {
        method: 'POST',
        body: JSON.stringify({
          to: emails,
          subject: subject,
          template: template,
          variables: variables,
          schedule_at: scheduleAt,
          priority: 'normal'
        })
      });
      
      if (!response.ok) {
        throw new Error('Failed to schedule email');
      }
      
      return await response.json();
      
    } catch (error) {
      console.error('Error scheduling email:', error);
      throw error;
    }
  }
}
