---
title: "Register (Mobile)"
api: "POST /auth/m/register"
description: "Register a new user account (mobile/web)"
---

Register a new user account for mobile or web applications. This endpoint creates a new user account with the provided information and returns authentication tokens for immediate use.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and does not require authentication. It's designed for user registration in mobile and web applications.
</Note>

## Request

<ParamField body="fullname" type="string" required>
  Full name of the user
  
  **Example**: "John Doe"
  
  **Constraints**: 
  - Minimum length: 2 characters
  - Maximum length: 100 characters
  - Must contain only letters, spaces, hyphens, and apostrophes
</ParamField>

<ParamField body="email" type="string" required>
  Email address for the account
  
  **Example**: "john@example.com"
  
  **Constraints**: 
  - Must be a valid email format
  - Must be unique across the platform
  - Will be used for login and notifications
</ParamField>

<ParamField body="password" type="string" required>
  Password for the account
  
  **Constraints**: 
  - Minimum length: 8 characters
  - Must contain at least one uppercase letter
  - Must contain at least one lowercase letter
  - Must contain at least one number
  - Special characters are recommended
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Indicates if the registration was successful
</ResponseField>

<ResponseField name="data" type="object">
  Contains user information and authentication tokens
</ResponseField>

<ResponseField name="data.user_id" type="string">
  Unique identifier for the new user
</ResponseField>

<ResponseField name="data.fullname" type="string">
  Full name of the registered user
</ResponseField>

<ResponseField name="data.email" type="string">
  Email address of the registered user
</ResponseField>

<ResponseField name="data.role" type="string">
  Default role assigned to the user
</ResponseField>

<ResponseField name="data.status" type="string">
  Account status (active, pending_verification)
</ResponseField>

<ResponseField name="data.access_token" type="string">
  JWT access token for API authentication (expires in 1 hour)
</ResponseField>

<ResponseField name="data.refresh_token" type="string">
  Refresh token for obtaining new access tokens (expires in 7 days)
</ResponseField>

<ResponseField name="data.expires_in" type="integer">
  Access token expiration time in seconds (3600 = 1 hour)
</ResponseField>

<ResponseField name="data.token_type" type="string">
  Token type, always "Bearer"
</ResponseField>

<ResponseField name="data.created_at" type="string">
  ISO 8601 timestamp when the account was created
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable message about the registration result
</ResponseField>

<ResponseField name="timestamp" type="string">
  ISO 8601 timestamp of the response
</ResponseField>

## Usage Notes

<AccordionGroup>
  <Accordion title="Password Requirements" icon="lock">
    Password must meet these criteria:
    
    - Minimum 8 characters
    - At least one uppercase letter (A-Z)
    - At least one lowercase letter (a-z)
    - At least one number (0-9)
    - Special characters recommended (!@#$%^&*)
  </Accordion>
  
  <Accordion title="Email Verification" icon="mail">
    Email verification process:
    
    - Account is created immediately
    - Verification email is sent automatically
    - User can use the account without verification
    - Some features may be limited until verified
  </Accordion>
  
  <Accordion title="Rate Limiting" icon="clock">
    Registration is rate limited to prevent abuse:
    
    - Maximum 5 registrations per IP per hour
    - Maximum 1 registration per email per day
    - Rate limit resets after the specified time
  </Accordion>
  
  <Accordion title="Security Considerations" icon="shield">
    Security best practices:
    
    - Always use HTTPS for registration
    - Validate input on both client and server
    - Implement proper password hashing
    - Monitor for suspicious registration patterns
  </Accordion>
</AccordionGroup>

{`class RegistrationForm {
  constructor(apiClient) {
    this.apiClient = apiClient;
    this.form = document.getElementById('registration-form');
    this.setupEventListeners();
  }
  
  setupEventListeners() {
    this.form.addEventListener('submit', (e) => {
      e.preventDefault();
      this.handleSubmit();
    });
    
    // Real-time validation
    this.form.addEventListener('input', (e) => {
      this.validateField(e.target);
    });
  }
  
  async handleSubmit() {
    try {
      this.showLoading();
      
      const formData = this.getFormData();
      const validationErrors = this.validateForm(formData);
      
      if (validationErrors.length > 0) {
        this.showValidationErrors(validationErrors);
        return;
      }
      
      const response = await this.apiClient.makeRequest('/auth/m/register', {
        method: 'POST',
        body: JSON.stringify(formData)
      });
      
      if (!response.ok) {
        const error = await response.json();
        throw new Error(error.error.message);
      }
      
      const data = await response.json();
      this.handleRegistrationSuccess(data.data);
      
    } catch (error) {
      console.error('Registration error:', error);
      this.showError(error.message);
    } finally {
      this.hideLoading();
    }
  }
  
  getFormData() {
    const formData = new FormData(this.form);
    
    return {
      fullname: formData.get('fullname'),
      email: formData.get('email'),
      password: formData.get('password')
    };
  }
  
  validateForm(data) {
    const errors = [];
    
    // Validate fullname
    if (!data.fullname || data.fullname.length < 2) {
      errors.push({ field: 'fullname', message: 'Full name must be at least 2 characters' });
    }
    
    // Validate email
    if (!data.email || !this.isValidEmail(data.email)) {
      errors.push({ field: 'email', message: 'Invalid email format' });
    }
    
    // Validate password
    const passwordErrors = this.validatePassword(data.password);
    errors.push(...passwordErrors);
    
    return errors;
  }
  
  validatePassword(password) {
    const errors = [];
    
    if (!password || password.length < 8) {
      errors.push({ field: 'password', message: 'Password must be at least 8 characters' });
    }
    
    if (!/[A-Z]/.test(password)) {
      errors.push({ field: 'password', message: 'Password must contain at least one uppercase letter' });
    }
    
    if (!/[a-z]/.test(password)) {
      errors.push({ field: 'password', message: 'Password must contain at least one lowercase letter' });
    }
    
    if (!/[0-9]/.test(password)) {
      errors.push({ field: 'password', message: 'Password must contain at least one number' });
    }
    
    return errors;
  }
  
  validateField(field) {
    const value = field.value;
    const fieldName = field.name;
    
    // Clear previous error
    this.clearFieldError(field);
    
    let isValid = true;
    let errorMessage = '';
    
    switch (fieldName) {
      case 'fullname':
        if (!value || value.length < 2) {
          isValid = false;
          errorMessage = 'Full name must be at least 2 characters';
        }
        break;
      case 'email':
        if (!value || !this.isValidEmail(value)) {
          isValid = false;
          errorMessage = 'Invalid email format';
        }
        break;
      case 'password':
        const passwordErrors = this.validatePassword(value);
        if (passwordErrors.length > 0) {
          isValid = false;
          errorMessage = passwordErrors[0].message;
        }
        break;
    }
    
    if (!isValid) {
      this.showFieldError(field, errorMessage);
    }
  }
  
  isValidEmail(email) {
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    return emailRegex.test(email);
  }
  
  showFieldError(field, message) {
    const errorElement = field.parentNode.querySelector('.field-error');
    if (errorElement) {
      errorElement.textContent = message;
      errorElement.style.display = 'block';
    }
    
    field.classList.add('error');
  }
  
  clearFieldError(field) {
    const errorElement = field.parentNode.querySelector('.field-error');
    if (errorElement) {
      errorElement.style.display = 'none';
    }
    
    field.classList.remove('error');
  }
  
  showValidationErrors(errors) {
    errors.forEach(error => {
      const field = this.form.querySelector(\`[name="\${error.field}"]\`);
      if (field) {
        this.showFieldError(field, error.message);
      }
    });
  }
  
  handleRegistrationSuccess(userData) {
    // Store authentication tokens
    localStorage.setItem('weir_access_token', userData.access_token);
    localStorage.setItem('weir_refresh_token', userData.refresh_token);
    localStorage.setItem('weir_token_expiry', (Date.now() + (userData.expires_in * 1000)).toString());
    
    // Show success message
    this.showSuccess('Registration successful! Welcome to Weir AI.');
    
    // Redirect to dashboard or next step
    setTimeout(() => {
      window.location.href = '/dashboard';
    }, 2000);
  }
  
  showLoading() {
    const submitButton = this.form.querySelector('button[type="submit"]');
    submitButton.disabled = true;
    submitButton.textContent = 'Creating Account...';
  }
  
  hideLoading() {
    const submitButton = this.form.querySelector('button[type="submit"]');
    submitButton.disabled = false;
    submitButton.textContent = 'Create Account';
  }
  
  showSuccess(message) {
    this.showMessage(message, 'success');
  }
  
  showError(message) {
    this.showMessage(message, 'error');
  }
  
  showMessage(message, type) {
    const messageElement = document.getElementById('form-message');
    messageElement.textContent = message;
    messageElement.className = \`message message-\${type}\`;
    messageElement.style.display = 'block';
    
    setTimeout(() => {
      messageElement.style.display = 'none';
    }, 5000);
  }

{`class MobileRegistrationHandler {
  constructor(apiClient) {
    this.apiClient = apiClient;
  }
  
  async registerUser(userData) {
    try {
      // Validate input
      this.validateUserData(userData);
      
      // Make registration request
      const response = await this.apiClient.makeRequest('/auth/m/register', {
        method: 'POST',
        body: JSON.stringify(userData)
      });
      
      if (!response.ok) {
        const error = await response.json();
        throw new Error(error.error.message);
      }
      
      const data = await response.json();
      
      // Store user data and tokens
      await this.storeUserData(data.data);
      
      return data.data;
      
    } catch (error) {
      console.error('Registration failed:', error);
      throw error;
    }
  }
  
  validateUserData(userData) {
    const { fullname, email, password } = userData;
    
    if (!fullname || fullname.length < 2) {
      throw new Error('Full name must be at least 2 characters');
    }
    
    if (!email || !this.isValidEmail(email)) {
      throw new Error('Invalid email format');
    }
    
    if (!password || !this.isValidPassword(password)) {
      throw new Error('Password does not meet requirements');
    }
  }
  
  isValidEmail(email) {
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    return emailRegex.test(email);
  }
  
  isValidPassword(password) {
    return password.length >= 8 &&
           /[A-Z]/.test(password) &&
           /[a-z]/.test(password) &&
           /[0-9]/.test(password);
  }
  
  async storeUserData(userData) {
    // Store in secure storage
    await this.setSecureItem('weir_access_token', userData.access_token);
    await this.setSecureItem('weir_refresh_token', userData.refresh_token);
    await this.setSecureItem('weir_user_id', userData.user_id);
    await this.setSecureItem('weir_user_email', userData.email);
    await this.setSecureItem('weir_user_name', userData.fullname);
    
    // Store token expiry
    const expiry = Date.now() + (userData.expires_in * 1000);
    await this.setSecureItem('weir_token_expiry', expiry.toString());
  }
  
  async setSecureItem(key, value) {
    // Use appropriate secure storage method for your platform
    // This is a placeholder - implement based on your mobile framework
    if (typeof SecureStore !== 'undefined') {
      await SecureStore.setItemAsync(key, value);
    } else {
      localStorage.setItem(key, value);
    }
  }
  
  async getSecureItem(key) {
    // Use appropriate secure storage method for your platform
    if (typeof SecureStore !== 'undefined') {
      return await SecureStore.getItemAsync(key);
    } else {
      return localStorage.getItem(key);
    }
  }
  
  async isUserRegistered() {
    const token = await this.getSecureItem('weir_access_token');
    return !!token;
  }
  
  async getCurrentUser() {
    const userId = await this.getSecureItem('weir_user_id');
    const email = await this.getSecureItem('weir_user_email');
    const name = await this.getSecureItem('weir_user_name');
    
    if (userId && email && name) {
      return {
        user_id: userId,
        email: email,
        fullname: name
      };
    }
    
    return null;
  }
}`}
