---
title: "Get Organization Information"
api: "GET /org/information"
description: "Retrieve organization details and settings"
---

Retrieve comprehensive information about your organization including basic details, contact information, address, and configuration settings. This endpoint provides all the data needed to display and manage organization information.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication. It's designed for organization management and settings display.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

<CodeBlock lang="bash">
{`Authorization: Bearer <your_jwt_token>`}
</CodeBlock>

{`class OrganizationProfile {
  constructor(apiClient) {
    this.apiClient = apiClient;
    this.organization = null;
  }
  
  async loadOrganizationInfo() {
    try {
      const response = await this.apiClient.makeRequest('/org/information');
      
      if (!response.ok) {
        throw new Error('Failed to load organization information');
      }
      
      const data = await response.json();
      this.organization = data.data;
      
      this.displayOrganizationInfo();
      return data.data;
      
    } catch (error) {
      console.error('Error loading organization info:', error);
      this.displayError('Failed to load organization information');
      throw error;
    }
  }
  
  displayOrganizationInfo() {
    if (!this.organization) return;
    
    // Update basic information
    document.getElementById('org-name').textContent = this.organization.name;
    document.getElementById('org-website').textContent = this.organization.website;
    document.getElementById('org-industry').textContent = this.organization.industry;
    document.getElementById('org-size').textContent = this.organization.company_size;
    document.getElementById('org-founded').textContent = this.organization.founded;
    document.getElementById('org-description').textContent = this.organization.description;
    
    // Update address information
    const address = this.organization.address;
    const addressText = \`\${address.address1}\${address.address2 ? ', ' + address.address2 : ''}, \${address.city}, \${address.state} \${address.postal_code}, \${address.country}\`;
    document.getElementById('org-address').textContent = addressText;
    
    // Update contact information
    document.getElementById('org-phone').textContent = this.organization.contact.phone;
    document.getElementById('org-email').textContent = this.organization.contact.email;
    
    // Update settings
    document.getElementById('org-timezone').textContent = this.organization.settings.timezone;
    document.getElementById('org-language').textContent = this.organization.settings.language;
    document.getElementById('org-currency').textContent = this.organization.settings.currency;
    
    // Update timestamps
    document.getElementById('org-created').textContent = new Date(this.organization.created_at).toLocaleDateString();
    document.getElementById('org-updated').textContent = new Date(this.organization.updated_at).toLocaleDateString();
  }
  
  displayError(message) {
    const errorElement = document.getElementById('org-error');
    errorElement.textContent = message;
    errorElement.style.display = 'block';
  }
  
  hideError() {
    const errorElement = document.getElementById('org-error');
    errorElement.style.display = 'none';
  }
  
  async refresh() {
    this.hideError();
    await this.loadOrganizationInfo();
  }
}`}

{`class OrganizationSettingsManager {
  constructor(apiClient) {
    this.apiClient = apiClient;
    this.organization = null;
  }
  
  async loadSettings() {
    try {
      const response = await this.apiClient.makeRequest('/org/information');
      const data = await response.json();
      
      this.organization = data.data;
      this.populateSettingsForm();
      
      return data.data;
      
    } catch (error) {
      console.error('Error loading organization settings:', error);
      throw error;
    }
  }
  
  populateSettingsForm() {
    if (!this.organization) return;
    
    // Basic information
    document.getElementById('org-name-input').value = this.organization.name || '';
    document.getElementById('org-website-input').value = this.organization.website || '';
    document.getElementById('org-industry-select').value = this.organization.industry || '';
    document.getElementById('org-size-select').value = this.organization.company_size || '';
    document.getElementById('org-founded-input').value = this.organization.founded || '';
    document.getElementById('org-description-textarea').value = this.organization.description || '';
    
    // Address information
    const address = this.organization.address || {};
    document.getElementById('address1-input').value = address.address1 || '';
    document.getElementById('address2-input').value = address.address2 || '';
    document.getElementById('city-input').value = address.city || '';
    document.getElementById('state-input').value = address.state || '';
    document.getElementById('postal-code-input').value = address.postal_code || '';
    document.getElementById('country-input').value = address.country || '';
    
    // Contact information
    const contact = this.organization.contact || {};
    document.getElementById('phone-input').value = contact.phone || '';
    document.getElementById('email-input').value = contact.email || '';
    
    // Settings
    const settings = this.organization.settings || {};
    document.getElementById('timezone-select').value = settings.timezone || '';
    document.getElementById('language-select').value = settings.language || '';
    document.getElementById('currency-select').value = settings.currency || '';
  }
  
  getFormData() {
    return {
      basicInfo: {
        website: document.getElementById('org-website-input').value,
        industry: document.getElementById('org-industry-select').value,
        companySize: document.getElementById('org-size-select').value,
        founded: document.getElementById('org-founded-input').value,
        description: document.getElementById('org-description-textarea').value
      },
      address: {
        address1: document.getElementById('address1-input').value,
        address2: document.getElementById('address2-input').value,
        city: document.getElementById('city-input').value,
        state: document.getElementById('state-input').value,
        postalCode: document.getElementById('postal-code-input').value,
        country: document.getElementById('country-input').value
      },
      contact: {
        phone: document.getElementById('phone-input').value,
        email: document.getElementById('email-input').value
      }
    };
  }
  
  validateFormData(data) {
    const errors = [];
    
    // Validate required fields
    if (!data.basicInfo.website) {
      errors.push('Website is required');
    }
    
    if (!data.contact.email) {
      errors.push('Contact email is required');
    }
    
    // Validate email format
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (data.contact.email && !emailRegex.test(data.contact.email)) {
      errors.push('Invalid email format');
    }
    
    // Validate website format
    const websiteRegex = /^https?:\/\/.+/;
    if (data.basicInfo.website && !websiteRegex.test(data.basicInfo.website)) {
      errors.push('Website must start with http:// or https://');
    }
    
    return errors;
  }
  
  async saveSettings() {
    try {
      const formData = this.getFormData();
      const errors = this.validateFormData(formData);
      
      if (errors.length > 0) {
        throw new Error(\`Validation errors: \${errors.join(', ')}\`);
      }
      
      const response = await this.apiClient.makeRequest('/org/information', {
        method: 'PATCH',
        body: JSON.stringify(formData)
      });
      
      if (!response.ok) {
        throw new Error('Failed to save organization settings');
      }
      
      // Reload settings to get updated data
      await this.loadSettings();
      
      return true;
      
    } catch (error) {
      console.error('Error saving organization settings:', error);
      throw error;
    }
  }
}`}
