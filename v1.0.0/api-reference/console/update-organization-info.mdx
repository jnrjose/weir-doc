---
title: "Update Organization Information"
api: "PATCH /org/information"
description: "Update organization details and settings"
---

Update organization information including basic details, contact information, address, and configuration settings. This endpoint allows you to modify all aspects of your organization's profile and settings.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication. It's designed for organization management and settings updates.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

<CodeBlock lang="bash">
{`Authorization: Bearer <your_jwt_token>`}
</CodeBlock>

## Request

<ParamField body="basicInfo" type="object" required="false">
  Basic organization information
  
  **Properties**:
  - `website` (string): Organization website URL
  - `industry` (string): Industry classification
  - `companySize` (string): Company size category
  - `founded` (string): Year the organization was founded
  - `description` (string): Organization description
</ParamField>

<ParamField body="address" type="object" required="false">
  Organization address information
  
  **Properties**:
  - `address1` (string): Primary address line
  - `address2` (string): Secondary address line (optional)
  - `city` (string): City name
  - `state` (string): State or province
  - `postalCode` (string): Postal or ZIP code
  - `country` (string): Country name
</ParamField>

<ParamField body="contact" type="object" required="false">
  Contact information
  
  **Properties**:
  - `phone` (string): Primary phone number
  - `email` (string): Primary contact email
</ParamField>

## Usage Notes

<AccordionGroup>
  <Accordion title="Partial Updates" icon="edit">
    This endpoint supports partial updates:
    
    - Only include the fields you want to update
    - Omitted fields will remain unchanged
    - You can update basic info, address, or contact separately
    - All fields within a section are optional
  </Accordion>
  
  <Accordion title="Validation Rules" icon="check">
    Field validation requirements:
    
    - Email must be valid format
    - Website must start with http:// or https://
    - Phone number should include country code
    - Industry must be from predefined list
    - Company size must be from predefined list
  </Accordion>
  
  <Accordion title="Permissions" icon="shield">
    To update organization information, you need:
    
    - Valid authentication token
    - Organization admin or owner role
    - Appropriate permissions for the data being updated
  </Accordion>
  
  <Accordion title="Data Sanitization" icon="broom">
    Input data is automatically sanitized:
    
    - HTML tags are stripped from text fields
    - URLs are validated and normalized
    - Phone numbers are formatted consistently
    - Special characters are properly escaped
  </Accordion>

{`class BulkOrganizationUpdate {
  constructor(apiClient) {
    this.apiClient = apiClient;
  }
  
  async updateMultipleOrganizations(updates) {
    const results = [];
    const errors = [];
    
    for (const update of updates) {
      try {
        const response = await this.apiClient.makeRequest('/org/information', {
          method: 'PATCH',
          body: JSON.stringify(update.data)
        });
        
        if (response.ok) {
          const result = await response.json();
          results.push({
            organizationId: update.organizationId,
            success: true,
            data: result.data
          });
        } else {
          const error = await response.json();
          errors.push({
            organizationId: update.organizationId,
            success: false,
            error: error.error.message
          });
        }
      } catch (error) {
        errors.push({
          organizationId: update.organizationId,
          success: false,
          error: error.message
        });
      }
    }
    
    return {
      results,
      errors,
      summary: {
        total: updates.length,
        successful: results.length,
        failed: errors.length
      }
    };
  }
  
  async updateFromCSV(csvData) {
    const updates = this.parseCSV(csvData);
    return await this.updateMultipleOrganizations(updates);
  }
  
  parseCSV(csvData) {
    const lines = csvData.split('\n');
    const headers = lines[0].split(',').map(h => h.trim());
    const updates = [];
    
    for (let i = 1; i < lines.length; i++) {
      const values = lines[i].split(',').map(v => v.trim());
      const update = {
        organizationId: values[0],
        data: {}
      };
      
      // Map CSV columns to API fields
      for (let j = 1; j < headers.length; j++) {
        const header = headers[j];
        const value = values[j];
        
        if (value) {
          this.mapCSVFieldToAPI(update.data, header, value);
        }
      }
      
      updates.push(update);
    }
    
    return updates;
  }
  
  mapCSVFieldToAPI(data, header, value) {
    const fieldMap = {
      'website': 'basicInfo.website',
      'industry': 'basicInfo.industry',
      'company_size': 'basicInfo.companySize',
      'founded': 'basicInfo.founded',
      'description': 'basicInfo.description',
      'address1': 'address.address1',
      'address2': 'address.address2',
      'city': 'address.city',
      'state': 'address.state',
      'postal_code': 'address.postalCode',
      'country': 'address.country',
      'phone': 'contact.phone',
      'email': 'contact.email'
    };
    
    const fieldPath = fieldMap[header];
    if (fieldPath) {
      const parts = fieldPath.split('.');
      let current = data;
      
      for (let i = 0; i < parts.length - 1; i++) {
        if (!current[parts[i]]) {
          current[parts[i]] = {};
        }
        current = current[parts[i]];
      }
      
      current[parts[parts.length - 1]] = value;
    }
  }
