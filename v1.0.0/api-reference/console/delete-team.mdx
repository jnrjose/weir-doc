---
title: "Delete Team"
api: "DELETE /org/team/{teamId}"
description: "Delete a team from the organization"
---

Delete a team from your organization. This endpoint permanently removes the team and all associated data. This action cannot be undone, so use with caution.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication. It's designed for team management and organization cleanup.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

<CodeBlock lang="bash">
{`Authorization: Bearer <your_jwt_token>`}
</CodeBlock>

## Path Parameters

<ParamField path="teamId" type="string" required>
  ID of the team to delete
  
  **Example**: "team_1234567890abcdef"
</ParamField>

## Query Parameters

<ParamField query="confirm" type="boolean" required="false">
  Confirmation flag to prevent accidental deletions
  
  **Example**: "true"
  
  **Default**: "false"
  
  **Note**: Must be set to true to proceed with deletion
</ParamField>

<ParamField query="force" type="boolean" required="false">
  Force deletion even if team has active members
  
  **Example**: "true"
  
  **Default**: "false"
  
  **Note**: Use with caution - will remove all team members
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Indicates if the team deletion was successful
</ResponseField>

<ResponseField name="data" type="object">
  Contains deletion confirmation information
</ResponseField>

<ResponseField name="data.team_id" type="string">
  ID of the deleted team
</ResponseField>

<ResponseField name="data.name" type="string">
  Name of the deleted team
</ResponseField>

<ResponseField name="data.deleted_at" type="string">
  ISO 8601 timestamp when the team was deleted
</ResponseField>

<ResponseField name="data.deleted_by" type="string">
  User ID of the person who deleted the team
</ResponseField>

<ResponseField name="data.members_affected" type="integer">
  Number of team members who were affected by the deletion
</ResponseField>

<ResponseField name="data.resources_cleaned" type="integer">
  Number of team resources that were cleaned up
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable message about the operation result
</ResponseField>

<ResponseField name="timestamp" type="string">
  ISO 8601 timestamp of the response
</ResponseField>

## Usage Notes

<AccordionGroup>
  <Accordion title="Deletion Safety" icon="shield">
    Safety measures for team deletion:
    
    - Confirmation parameter is required to prevent accidental deletions
    - Teams with active members cannot be deleted unless force=true
    - All team data is permanently removed
    - This action cannot be undone
  </Accordion>
  
  <Accordion title="Pre-deletion Checklist" icon="check">
    Before deleting a team:
    
    - Remove all team members
    - Transfer or delete team resources
    - Notify team members of the deletion
    - Ensure no critical dependencies exist
  </Accordion>
  
  <Accordion title="Permissions" icon="key">
    To delete a team, you need:
    
    - Valid authentication token
    - Team admin or organization admin role
    - Appropriate permissions for the team
  </Accordion>
  
  <Accordion title="Data Cleanup" icon="broom">
    When a team is deleted:
    
    - All team members are removed
    - Team resources are cleaned up
    - Team settings are deleted
    - Audit logs are maintained
  </Accordion>
</AccordionGroup>

{`class TeamDeletionConfirmation {
  constructor(apiClient, teamId) {
    this.apiClient = apiClient;
    this.teamId = teamId;
    this.modal = document.getElementById('delete-team-modal');
    this.setupEventListeners();
  }
  
  setupEventListeners() {
    const confirmButton = document.getElementById('confirm-delete');
    const cancelButton = document.getElementById('cancel-delete');
    
    confirmButton.addEventListener('click', () => {
      this.handleDeletion();
    });
    
    cancelButton.addEventListener('click', () => {
      this.hideModal();
    });
  }
  
  async handleDeletion() {
    try {
      this.showLoading();
      
      // Check if team has active members
      const teamInfo = await this.getTeamInfo();
      
      if (teamInfo.member_count > 0) {
        const forceDelete = confirm(
          \`This team has \${teamInfo.member_count} active members. Do you want to force delete and remove all members?\`
        );
        
        if (!forceDelete) {
          this.hideLoading();
          return;
        }
      }
      
      // Proceed with deletion
      const response = await this.apiClient.makeRequest(
        \`/org/team/\${this.teamId}?confirm=true\${teamInfo.member_count > 0 ? '&force=true' : ''}\`,
        {
          method: 'DELETE'
        }
      );
      
      if (!response.ok) {
        const error = await response.json();
        throw new Error(error.error.message);
      }
      
      const data = await response.json();
      this.handleDeletionSuccess(data.data);
      
    } catch (error) {
      console.error('Team deletion error:', error);
      this.showError(error.message);
    } finally {
      this.hideLoading();
    }
  }
  
  async getTeamInfo() {
    const response = await this.apiClient.makeRequest(\`/org/team/members/\${this.teamId}\`);
    const data = await response.json();
    
    return {
      member_count: data.data.length,
      team_name: data.team_name
    };
  }
  
  handleDeletionSuccess(deletionData) {
    // Show success message
    this.showSuccess(\`Team "\${deletionData.name}" deleted successfully\`);
    
    // Hide modal
    this.hideModal();
    
    // Remove team from UI
    this.removeTeamFromUI();
    
    // Redirect to teams list
    setTimeout(() => {
      window.location.href = '/teams';
    }, 2000);
  }
  
  removeTeamFromUI() {
    const teamElement = document.querySelector(\`[data-team-id="\${this.teamId}"]\`);
    if (teamElement) {
      teamElement.remove();
    }
  }
  
  showModal() {
    this.modal.style.display = 'block';
  }
  
  hideModal() {
    this.modal.style.display = 'none';
  }
  
  showLoading() {
    const confirmButton = document.getElementById('confirm-delete');
    confirmButton.disabled = true;
    confirmButton.textContent = 'Deleting...';
  }
  
  hideLoading() {
    const confirmButton = document.getElementById('confirm-delete');
    confirmButton.disabled = false;
    confirmButton.textContent = 'Delete Team';
  }
  
  showSuccess(message) {
    this.showMessage(message, 'success');
  }
  
  showError(message) {
    this.showMessage(message, 'error');
  }
  
  showMessage(message, type) {
    const messageElement = document.getElementById('deletion-message');
    messageElement.textContent = message;
    messageElement.className = \`message message-\${type}\`;
    messageElement.style.display = 'block';
    
    setTimeout(() => {
      messageElement.style.display = 'none';
    }, 5000);
  }
}`}

{`class BulkTeamCleanup {
  constructor(apiClient) {
    this.apiClient = apiClient;
  }
  
  async cleanupInactiveTeams() {
    try {
      // Get all teams
      const response = await this.apiClient.makeRequest('/org/teams');
      const data = await response.json();
      
      const teams = data.data;
      const inactiveTeams = teams.filter(team => this.isInactiveTeam(team));
      
      console.log(\`Found \${inactiveTeams.length} inactive teams\`);
      
      // Delete inactive teams
      const results = [];
      for (const team of inactiveTeams) {
        try {
          await this.deleteTeam(team.team_id);
          results.push({ team_id: team.team_id, success: true });
        } catch (error) {
          results.push({ team_id: team.team_id, success: false, error: error.message });
        }
      }
      
      return results;
      
    } catch (error) {
      console.error('Bulk cleanup error:', error);
      throw error;
    }
  }
  
  isInactiveTeam(team) {
    // Define criteria for inactive teams
    const thirtyDaysAgo = new Date();
    thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 30);
    
    return (
      team.member_count === 0 &&
      new Date(team.last_activity) < thirtyDaysAgo
    );
  }
  
  async deleteTeam(teamId) {
    const response = await this.apiClient.makeRequest(
      \`/org/team/\${teamId}?confirm=true&force=true\`,
      {
        method: 'DELETE'
      }
    );
    
    if (!response.ok) {
      const error = await response.json();
      throw new Error(error.error.message);
    }
    
    return await response.json();
  }
  
  async generateCleanupReport(results) {
    const successful = results.filter(r => r.success).length;
    const failed = results.filter(r => !r.success).length;
    
    const report = {
      timestamp: new Date().toISOString(),
      summary: {
        total: results.length,
        successful: successful,
        failed: failed
      },
      successful_deletions: results.filter(r => r.success).map(r => r.team_id),
      failed_deletions: results.filter(r => !r.success).map(r => ({
        team_id: r.team_id,
        error: r.error
      }))
    };
    
    return report;
  }
}`}
