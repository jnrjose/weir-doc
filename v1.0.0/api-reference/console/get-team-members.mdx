---
title: "Get Team Members"
api: "GET /org/team/members/{teamId}"
description: "Retrieve list of members in a specific team"
---

Retrieve a comprehensive list of all members in a specific team, including their roles, status, and membership details. This endpoint provides detailed information about team composition and member permissions.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication. It's designed for team member management and oversight.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

```bash
{`Authorization: Bearer <your_jwt_token>`}
```

## Path Parameters

<ParamField path="teamId" type="string" required>
  ID of the team to get members for
  
  **Example**: "team_1234567890abcdef"
</ParamField>

## Query Parameters

<ParamField query="page" type="string" required="false">
  Page number for pagination (default: 1)
  
  **Example**: "1"
</ParamField>

<ParamField query="limit" type="string" required="false">
  Number of members per page (default: 20, max: 100)
  
  **Example**: "20"
</ParamField>

<ParamField query="role" type="string" required="false">
  Filter members by role
  
  **Options**: "Organization_User", "Team_Admin", "Viewer"
  
  **Example**: "Team_Admin"
</ParamField>

<ParamField query="status" type="string" required="false">
  Filter members by status
  
  **Options**: "active", "inactive", "pending", "suspended"
  
  **Example**: "active"
</ParamField>

<ParamField query="search" type="string" required="false">
  Search members by name or email
  
  **Example**: "john"
</ParamField>

<ParamField query="sort" type="string" required="false">
  Sort members by field
  
  **Options**: "name", "email", "role", "joined_at", "last_active"
  
  **Default**: "joined_at"
</ParamField>

<ParamField query="order" type="string" required="false">
  Sort order
  
  **Options**: "asc", "desc"
  
  **Default**: "desc"
</ParamField>

{`class TeamMemberAnalytics {
  constructor(apiClient, teamId) {
    this.apiClient = apiClient;
    this.teamId = teamId;
  }
  
  async getAllTeamMembers() {
    const allMembers = [];
    let page = 1;
    let hasMore = true;
    
    while (hasMore) {
      const response = await this.apiClient.makeRequest(\`/org/team/members/\${this.teamId}?page=\${page}&limit=100\`);
      const data = await response.json();
      
      allMembers.push(...data.data);
      hasMore = data.pagination.has_next;
      page++;
    }
    
    return allMembers;
  }
  
  async generateMemberReport() {
    const members = await this.getAllTeamMembers();
    
    const stats = {
      total_members: members.length,
      active_members: members.filter(m => m.status === 'active').length,
      inactive_members: members.filter(m => m.status === 'inactive').length,
      pending_members: members.filter(m => m.status === 'pending').length,
      role_breakdown: this.groupByRole(members),
      department_breakdown: this.groupByDepartment(members),
      recent_joiners: this.getRecentJoiners(members),
      inactive_members_list: this.getInactiveMembers(members),
      activity_summary: this.getActivitySummary(members)
    };
    
    return stats;
  }
  
  groupByRole(members) {
    const roles = {};
    members.forEach(member => {
      roles[member.role] = (roles[member.role] || 0) + 1;
    });
    return roles;
  }
  
  groupByDepartment(members) {
    const departments = {};
    members.forEach(member => {
      const dept = member.profile.department || 'Unknown';
      departments[dept] = (departments[dept] || 0) + 1;
    });
    return departments;
  }
  
  getRecentJoiners(members) {
    const thirtyDaysAgo = new Date();
    thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 30);
    
    return members
      .filter(member => new Date(member.joined_at) > thirtyDaysAgo)
      .sort((a, b) => new Date(b.joined_at) - new Date(a.joined_at));
  }
  
  getInactiveMembers(members) {
    const sevenDaysAgo = new Date();
    sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 7);
    
    return members.filter(member => 
      member.status === 'active' && 
      new Date(member.last_active) < sevenDaysAgo
    );
  }
  
  getActivitySummary(members) {
    const now = new Date();
    const oneDayAgo = new Date(now.getTime() - 24 * 60 * 60 * 1000);
    const oneWeekAgo = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000);
    const oneMonthAgo = new Date(now.getTime() - 30 * 24 * 60 * 60 * 1000);
    
    return {
      active_today: members.filter(m => new Date(m.last_active) > oneDayAgo).length,
      active_this_week: members.filter(m => new Date(m.last_active) > oneWeekAgo).length,
      active_this_month: members.filter(m => new Date(m.last_active) > oneMonthAgo).length,
      never_active: members.filter(m => new Date(m.last_active) === new Date(m.joined_at)).length
    };
  }
}`}
