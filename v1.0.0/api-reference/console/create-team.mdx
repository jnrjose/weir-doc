---
title: "Create Team"
api: "POST /org/create/team"
description: "Create a new team within the organization"
---

Create a new team within your organization. This endpoint allows you to set up teams for organizing members and managing access to different resources and projects.

<Note>
  **Console API Endpoint**
  
  This endpoint is part of the Console API category and requires Bearer Token authentication. It's designed for team management and organization structure.
</Note>

## Authentication

This endpoint requires Bearer Token authentication. Include your JWT token in the Authorization header:

<CodeBlock lang="bash">
{`Authorization: Bearer <your_jwt_token>`}
</CodeBlock>

## Request

<ParamField body="name" type="string" required>
  Name of the team
  
  **Example**: "Development Team"
  
  **Constraints**: 
  - Minimum length: 2 characters
  - Maximum length: 100 characters
  - Must be unique within the organization
  - Can contain letters, numbers, spaces, hyphens, and underscores
</ParamField>

<ParamField body="description" type="string" required="false">
  Description of the team's purpose and responsibilities
  
  **Example**: "Core development team responsible for product development"
  
  **Constraints**: 
  - Maximum length: 500 characters
  - Optional field
</ParamField>

<ParamField body="settings" type="object" required="false">
  Team-specific settings and preferences
  
  **Properties**:
  - `visibility` (string): Team visibility level ("public", "private", "restricted")
  - `default_role` (string): Default role for new team members
  - `notifications` (boolean): Enable team notifications
  - `auto_approve` (boolean): Auto-approve team join requests
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Indicates if the team creation was successful
</ResponseField>

<ResponseField name="data" type="object">
  Contains the created team information
</ResponseField>

<ResponseField name="data.team_id" type="string">
  Unique identifier for the created team
</ResponseField>

<ResponseField name="data.name" type="string">
  Name of the created team
</ResponseField>

<ResponseField name="data.description" type="string">
  Description of the team
</ResponseField>

<ResponseField name="data.status" type="string">
  Current status of the team (active, inactive, pending)
</ResponseField>

<ResponseField name="data.visibility" type="string">
  Team visibility level
</ResponseField>

<ResponseField name="data.member_count" type="integer">
  Number of members in the team (initially 0)
</ResponseField>

<ResponseField name="data.created_by" type="string">
  User ID of the person who created the team
</ResponseField>

<ResponseField name="data.created_at" type="string">
  ISO 8601 timestamp when the team was created
</ResponseField>

<ResponseField name="data.updated_at" type="string">
  ISO 8601 timestamp when the team was last updated
</ResponseField>

<ResponseField name="data.settings" type="object">
  Team settings and preferences
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable message about the operation result
</ResponseField>

<ResponseField name="timestamp" type="string">
  ISO 8601 timestamp of the response
</ResponseField>

{`class TeamCreationForm {
  constructor(apiClient) {
    this.apiClient = apiClient;
    this.form = document.getElementById('create-team-form');
    this.setupEventListeners();
  }
  
  setupEventListeners() {
    this.form.addEventListener('submit', (e) => {
      e.preventDefault();
      this.handleSubmit();
    });
    
    // Real-time validation
    this.form.addEventListener('input', (e) => {
      this.validateField(e.target);
    });
  }
  
  async handleSubmit() {
    try {
      this.showLoading();
      
      const formData = this.getFormData();
      const validationErrors = this.validateForm(formData);
      
      if (validationErrors.length > 0) {
        this.showValidationErrors(validationErrors);
        return;
      }
      
      const response = await this.apiClient.makeRequest('/org/create/team', {
        method: 'POST',
        body: JSON.stringify(formData)
      });
      
      if (!response.ok) {
        const error = await response.json();
        throw new Error(error.error.message);
      }
      
      const data = await response.json();
      this.handleTeamCreationSuccess(data.data);
      
    } catch (error) {
      console.error('Team creation error:', error);
      this.showError(error.message);
    } finally {
      this.hideLoading();
    }
  }
  
  getFormData() {
    const formData = new FormData(this.form);
    
    return {
      name: formData.get('name'),
      description: formData.get('description'),
      settings: {
        visibility: formData.get('visibility'),
        default_role: formData.get('default_role'),
        notifications: formData.get('notifications') === 'on',
        auto_approve: formData.get('auto_approve') === 'on'
      }
    };
  }
  
  validateForm(data) {
    const errors = [];
    
    // Validate team name
    if (!data.name || data.name.length < 2) {
      errors.push({ field: 'name', message: 'Team name must be at least 2 characters' });
    }
    
    if (data.name && data.name.length > 100) {
      errors.push({ field: 'name', message: 'Team name must be less than 100 characters' });
    }
    
    // Validate description
    if (data.description && data.description.length > 500) {
      errors.push({ field: 'description', message: 'Description must be less than 500 characters' });
    }
    
    return errors;
  }
  
  validateField(field) {
    const value = field.value;
    const fieldName = field.name;
    
    // Clear previous error
    this.clearFieldError(field);
    
    let isValid = true;
    let errorMessage = '';
    
    switch (fieldName) {
      case 'name':
        if (!value || value.length < 2) {
          isValid = false;
          errorMessage = 'Team name must be at least 2 characters';
        } else if (value.length > 100) {
          isValid = false;
          errorMessage = 'Team name must be less than 100 characters';
        }
        break;
      case 'description':
        if (value && value.length > 500) {
          isValid = false;
          errorMessage = 'Description must be less than 500 characters';
        }
        break;
    }
    
    if (!isValid) {
      this.showFieldError(field, errorMessage);
    }
  }
  
  showFieldError(field, message) {
    const errorElement = field.parentNode.querySelector('.field-error');
    if (errorElement) {
      errorElement.textContent = message;
      errorElement.style.display = 'block';
    }
    
    field.classList.add('error');
  }
  
  clearFieldError(field) {
    const errorElement = field.parentNode.querySelector('.field-error');
    if (errorElement) {
      errorElement.style.display = 'none';
    }
    
    field.classList.remove('error');
  }
  
  showValidationErrors(errors) {
    errors.forEach(error => {
      const field = this.form.querySelector(\`[name="\${error.field}"]\`);
      if (field) {
        this.showFieldError(field, error.message);
      }
    });
  }
  
  handleTeamCreationSuccess(teamData) {
    // Show success message
    this.showSuccess(\`Team "\${teamData.name}" created successfully!\`);
    
    // Reset form
    this.form.reset();
    
    // Redirect to team management or refresh team list
    setTimeout(() => {
      window.location.href = \`/teams/\${teamData.team_id}\`;
    }, 2000);
  }
  
  showLoading() {
    const submitButton = this.form.querySelector('button[type="submit"]');
    submitButton.disabled = true;
    submitButton.textContent = 'Creating Team...';
  }
  
  hideLoading() {
    const submitButton = this.form.querySelector('button[type="submit"]');
    submitButton.disabled = false;
    submitButton.textContent = 'Create Team';
  }
  
  showSuccess(message) {
    this.showMessage(message, 'success');
  }
  
  showError(message) {
    this.showMessage(message, 'error');
  }
  
  showMessage(message, type) {
    const messageElement = document.getElementById('form-message');
    messageElement.textContent = message;
    messageElement.className = \`message message-\${type}\`;
    messageElement.style.display = 'block';
    
    setTimeout(() => {
      messageElement.style.display = 'none';
    }, 5000);
  }
}`}

{`class TeamManagementDashboard {
  constructor(apiClient) {
    this.apiClient = apiClient;
    this.teams = [];

      const response = await this.apiClient.makeRequest('/org/create/team', {
