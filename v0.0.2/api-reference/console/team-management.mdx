## Authentication

All team management endpoints require console authentication:

```bash
curl -X GET "https://api-dev.weir.ai/org/teams" \
  -H "Authorization: Bearer your-console-token" \
  -H "x-source: console"
```

## Rate Limits

Team management endpoints have the following rate limits:
- **100 requests per minute** per console user
- **1,000 requests per hour** per console user

## Common Use Cases

### 1. Setting Up a New Team
```javascript
async function setupNewTeam(teamName, members) {
  try {
    // Create the team
    const teamResponse = await fetch('https://api-dev.weir.ai/org/create/team', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${consoleToken}`,
        'x-source': 'console',
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({ name: teamName })
    });
    
    const team = await teamResponse.json();
    const teamId = team.data.id;
    
    // Invite team members
    for (const member of members) {
      await fetch(`https://api-dev.weir.ai/org/invite/team/member/${teamId}`, {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${consoleToken}`,
          'x-source': 'console',
          'Content-Type': 'application/json'
        },
        body: JSON.stringify(member)
      });
    }
    
    return teamId;
  } catch (error) {
    console.error('Team setup failed:', error);
    throw error;
  }
}
```

### 2. Managing Team Members
```javascript
async function manageTeamMembers(teamId) {
  try {
    // Get current team members
    const membersResponse = await fetch(`https://api-dev.weir.ai/org/team/members/${teamId}`, {
      method: 'GET',
      headers: {
        'Authorization': `Bearer ${consoleToken}`,
        'x-source': 'console'
      }
    });
    
    const members = await membersResponse.json();
    
    // Process team members
    for (const member of members.data) {
      console.log(`Member: ${member.firstname} ${member.lastname} (${member.email})`);
      console.log(`Role: ${member.role}`);
    }
    
    return members.data;
  } catch (error) {
    console.error('Team member management failed:', error);
    throw error;
  }
}
```

### 3. Team Cleanup
```javascript
async function cleanupTeam(teamId) {
  try {
    // Get team members first
    const membersResponse = await fetch(`https://api-dev.weir.ai/org/team/members/${teamId}`, {
      method: 'GET',
      headers: {
        'Authorization': `Bearer ${consoleToken}`,
        'x-source': 'console'
      }
    });
    
    const members = await membersResponse.json();
    
    if (members.data.length > 0) {
      console.log(`Team has ${members.data.length} members. Cannot delete.`);
      return false;
    }
    
    // Delete the team
    const deleteResponse = await fetch(`https://api-dev.weir.ai/org/team/${teamId}`, {
      method: 'DELETE',
      headers: {
        'Authorization': `Bearer ${consoleToken}`,
        'x-source': 'console'
      }
    });
    
    return deleteResponse.ok;
  } catch (error) {
    console.error('Team cleanup failed:', error);
    throw error;
  }
}
```

## Error Handling

Team management endpoints return specific error codes:

| Code | Description | Solution |
|------|-------------|----------|
| `TEAM_NOT_FOUND` | Team with specified ID doesn't exist | Check team ID |
| `MEMBER_ALREADY_EXISTS` | User is already a team member | Check existing members |
| `INVALID_TEAM_NAME` | Team name is invalid or already taken | Use a different name |
| `INSUFFICIENT_PERMISSIONS` | User lacks team management permissions | Check user role |
| `TEAM_HAS_MEMBERS` | Cannot delete team with active members | Remove members first |

## Best Practices

### 1. Validate Team Names
```javascript
function validateTeamName(name) {
  if (!name || name.trim().length === 0) {
    throw new Error('Team name is required');
  }
  
  if (name.length > 100) {
    throw new Error('Team name must be less than 100 characters');
  }
  
  // Check for invalid characters
  if (!/^[a-zA-Z0-9\s\-_]+$/.test(name)) {
    throw new Error('Team name contains invalid characters');
  }
  
  return true;
}
```

### 2. Handle Member Invitations
```javascript
async function inviteTeamMember(teamId, memberData) {
  try {
    // Validate member data
    if (!memberData.email || !memberData.firstname || !memberData.lastname) {
      throw new Error('Member data is incomplete');
    }
    
    // Check if member already exists
    const existingMembers = await getTeamMembers(teamId);
    const existingMember = existingMembers.find(m => m.email === memberData.email);
    
    if (existingMember) {
      throw new Error('Member is already part of this team');
    }
    
    // Send invitation
    const response = await fetch(`https://api-dev.weir.ai/org/invite/team/member/${teamId}`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${consoleToken}`,
        'x-source': 'console',
        'Content-Type': 'application/json'
      },
      body: JSON.stringify(memberData)
    });
    
    if (!response.ok) {
      const error = await response.json();
      throw new Error(error.error.message);
    }
    
    return await response.json();
  } catch (error) {
    console.error('Member invitation failed:', error);
    throw error;
  }
}
```

### 3. Implement Pagination
```javascript
async function getAllTeams(page = 1, limit = 20) {
  try {
    const response = await fetch(`https://api-dev.weir.ai/org/teams?page=${page}&limit=${limit}`, {
      method: 'GET',
      headers: {
        'Authorization': `Bearer ${consoleToken}`,
        'x-source': 'console'
      }
    });
    
    const data = await response.json();
    
    return {
      teams: data.data.items,
      pagination: data.data.pagination
    };
  } catch (error) {
    console.error('Failed to get teams:', error);
    throw error;
  }
}
```
