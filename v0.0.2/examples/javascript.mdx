# JavaScript Examples

Comprehensive JavaScript examples for integrating Weir's API into your applications. These examples cover both browser and Node.js environments with proper error handling and best practices.

## Prerequisites

Before using these examples, ensure you have:

- **Weir API credentials** (Client ID and Secret Key)
- **Node.js** (version 14 or higher) for server-side examples
- **Modern browser** with fetch API support for client-side examples
- **Basic understanding** of JavaScript, Promises, and async/await

## Installation

### Browser Environment
No additional installation required. The examples use the native `fetch` API available in modern browsers.

### Node.js Environment
```bash
npm install node-fetch
# or
yarn add node-fetch
```

## Basic Setup

### API Client Class

```javascript
class WeirAPIClient {
  constructor(clientId, secretKey, baseUrl = 'https://api-dev.weir.ai') {
    this.clientId = clientId;
    this.secretKey = secretKey;
    this.baseUrl = baseUrl;
    this.accessToken = null;
    this.tokenExpiry = null;
  }

  async authenticate() {
    try {
      const response = await fetch(`${this.baseUrl}/auth/token`, {
        method: 'POST',
        headers: {
          'Authorization': 'Basic ' + btoa(`${this.clientId}:${this.secretKey}`)
        }
      });

      if (!response.ok) {
        throw new Error(`Authentication failed: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error.message);
      }

      this.accessToken = data.data.accessToken;
      this.tokenExpiry = Date.now() + (data.data.expiresIn * 1000);
      
      return data.data;
    } catch (error) {
      console.error('Authentication failed:', error);
      throw error;
    }
  }

  async getValidToken() {
    if (!this.accessToken || this.isTokenExpired()) {
      await this.authenticate();
    }
    return this.accessToken;
  }

  isTokenExpired() {
    return !this.tokenExpiry || Date.now() >= this.tokenExpiry;
  }

  async makeRequest(endpoint, options = {}) {
    try {
      const token = await this.getValidToken();
      
      const response = await fetch(`${this.baseUrl}${endpoint}`, {
        ...options,
        headers: {
          'Authorization': `Bearer ${token}`,
          'Content-Type': 'application/json',
          ...options.headers
        }
      });

      if (!response.ok) {
        const error = await response.json();
        throw new Error(error.error?.message || `HTTP error! status: ${response.status}`);
      }

      return await response.json();
    } catch (error) {
      console.error('API request failed:', error);
      throw error;
    }
  }
}
```

### 3. Token Management

```javascript
class TokenManager {
  constructor() {
    this.accessToken = localStorage.getItem('weir_access_token');
    this.refreshToken = localStorage.getItem('weir_refresh_token');
  }

  async refreshAccessToken() {
    if (!this.refreshToken) {
      throw new Error('No refresh token available');
    }

    try {
      const response = await fetch('https://api-dev.weir.ai/auth/refresh/token', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          refreshToken: this.refreshToken
        })
      });

      if (!response.ok) {
        throw new Error('Token refresh failed');
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error.message);
      }

      this.accessToken = data.data.accessToken;
      this.refreshToken = data.data.refreshToken;

      // Update stored tokens
      localStorage.setItem('weir_access_token', this.accessToken);
      localStorage.setItem('weir_refresh_token', this.refreshToken);

      return this.accessToken;
    } catch (error) {
      console.error('Token refresh failed:', error);
      this.clearTokens();
      throw error;
    }
  }

  async getValidToken() {
    if (!this.accessToken || this.isTokenExpired()) {
      await this.refreshAccessToken();
    }
    return this.accessToken;
  }

  isTokenExpired() {
    if (!this.accessToken) return true;
    
    try {
      const payload = JSON.parse(atob(this.accessToken.split('.')[1]));
      return Date.now() >= payload.exp * 1000;
    } catch (error) {
      return true;
    }
  }

  clearTokens() {
    this.accessToken = null;
    this.refreshToken = null;
    localStorage.removeItem('weir_access_token');
    localStorage.removeItem('weir_refresh_token');
    localStorage.removeItem('weir_user');
  }
}
```

```javascript
async function loadAllTeams() {
```

## Error Handling Examples

### 1. Comprehensive Error Handling

```javascript
class WeirAPIError extends Error {
  constructor(message, code, details, status) {
    super(message);
    this.name = 'WeirAPIError';
    this.code = code;
    this.details = details;
    this.status = status;
  }
}

async function handleWeirResponse(response) {
  if (!response.ok) {
    let errorData;
    try {
      errorData = await response.json();
    } catch (e) {
      throw new WeirAPIError(
        `HTTP ${response.status}: ${response.statusText}`,
        'HTTP_ERROR',
        null,
        response.status
      );
    }

    throw new WeirAPIError(
      errorData.error.message,
      errorData.error.code,
      errorData.error.details,
      response.status
    );
  }

  const data = await response.json();
  
  if (!data.success) {
    throw new WeirAPIError(
      data.error.message,
      data.error.code,
      data.error.details,
      response.status
    );
  }

  return data;
}

// Usage
async function makeWeirRequest(url, options = {}) {
  try {
    const response = await fetch(url, options);
    return await handleWeirResponse(response);
  } catch (error) {
    if (error instanceof WeirAPIError) {
      console.error('Weir API Error:', {
        code: error.code,
        message: error.message,
        details: error.details,
        status: error.status
      });
    } else {
      console.error('Network or other error:', error);
    }
    throw error;
  }
}
```

### 2. Retry Logic with Exponential Backoff

```javascript
async function makeRequestWithRetry(url, options = {}, maxRetries = 3) {
  for (let i = 0; i < maxRetries; i++) {
    try {
      const response = await fetch(url, options);
      
      if (response.status === 429) {
        const retryAfter = response.headers.get('Retry-After') || Math.pow(2, i);
        console.log(`Rate limited. Retrying after ${retryAfter} seconds...`);
        await new Promise(resolve => setTimeout(resolve, retryAfter * 1000));
        continue;
      }
      
      if (response.status >= 500) {
        // Server error, retry with exponential backoff
        const delay = Math.pow(2, i) * 1000;
        console.log(`Server error. Retrying after ${delay}ms...`);
        await new Promise(resolve => setTimeout(resolve, delay));
        continue;
      }
      
      return response;
    } catch (error) {
      if (i === maxRetries - 1) throw error;
      
      const delay = Math.pow(2, i) * 1000;
      console.log(`Request failed. Retrying after ${delay}ms...`);
      await new Promise(resolve => setTimeout(resolve, delay));
    }
  }
}
```

## React Integration Examples

### 1. React Hook for Weir API

```jsx
import { useState, useEffect, useCallback } from 'react';

export function useWeirAPI(clientId, secretKey) {
  const [accessToken, setAccessToken] = useState(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);
  const [user, setUser] = useState(null);
  
  const authenticate = useCallback(async () => {
    setLoading(true);
    setError(null);
    
    try {
      const response = await fetch('https://api-dev.weir.ai/auth/token', {
        method: 'POST',
        headers: {
          'Authorization': 'Basic ' + btoa(`${clientId}:${secretKey}`)
        }
      });
      
      if (!response.ok) {
        throw new Error(`Authentication failed: ${response.status}`);
      }
      
      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error.message);
      }
      
      setAccessToken(data.data.accessToken);
      return data.data;
    } catch (err) {
      setError(err.message);
      throw err;
    } finally {
      setLoading(false);
    }
  }, [clientId, secretKey]);
  
  const makeRequest = useCallback(async (endpoint, options = {}) => {
    if (!accessToken) {
      throw new Error('Not authenticated');
    }
    
    try {
      const response = await fetch(`https://api-dev.weir.ai${endpoint}`, {
        ...options,
        headers: {
          'Authorization': `Bearer ${accessToken}`,
          'Content-Type': 'application/json',
          ...options.headers
        }
      });
      
      if (!response.ok) {
        const error = await response.json();
        throw new Error(error.error?.message || `HTTP error! status: ${response.status}`);
      }
      
      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error.message);
      }
      
      return data.data;
    } catch (err) {
      setError(err.message);
      throw err;
    }
  }, [accessToken]);
  
  const login = useCallback(async (username, password) => {
    setLoading(true);
    setError(null);
    
    try {
      const response = await fetch('https://api-dev.weir.ai/auth/login', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({ username, password })
      });
      
      if (!response.ok) {
        throw new Error(`Login failed: ${response.status}`);
      }
      
      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error.message);
      }
      
      setAccessToken(data.data.accessToken);
      setUser(data.data.user);
      
      // Store tokens securely
      localStorage.setItem('weir_access_token', data.data.accessToken);
      localStorage.setItem('weir_refresh_token', data.data.refreshToken);
      localStorage.setItem('weir_user', JSON.stringify(data.data.user));
      
      return data.data;
    } catch (err) {
      setError(err.message);
      throw err;
    } finally {
      setLoading(false);
    }
  }, []);
  
  const logout = useCallback(() => {
    setAccessToken(null);
    setUser(null);
    setError(null);
    
    // Clear stored tokens
    localStorage.removeItem('weir_access_token');
    localStorage.removeItem('weir_refresh_token');
    localStorage.removeItem('weir_user');
  }, []);
  
  useEffect(() => {
    // Check for existing tokens on mount
    const storedToken = localStorage.getItem('weir_access_token');
    const storedUser = localStorage.getItem('weir_user');
    
    if (storedToken && storedUser) {
      setAccessToken(storedToken);
      setUser(JSON.parse(storedUser));
    }
  }, []);
  
  return {
    accessToken,
    user,
    loading,
    error,
    authenticate,
    makeRequest,
    login,
    logout
  };
}
```

### 2. React Component Example

```jsx
import React, { useState, useEffect } from 'react';
import { useWeirAPI } from './useWeirAPI';

function TeamManager({ clientId, secretKey }) {
  const { accessToken, user, loading, error, makeRequest } = useWeirAPI(clientId, secretKey);
  const [teams, setTeams] = useState([]);
  const [newTeamName, setNewTeamName] = useState('');
  
  useEffect(() => {
    if (accessToken) {
      loadTeams();
    }
  }, [accessToken]);
  
  const loadTeams = async () => {
    try {
      const data = await makeRequest('/org/teams');
      setTeams(data.items);
    } catch (err) {
      console.error('Failed to load teams:', err);
    }
  };
  
  const createTeam = async (e) => {
    e.preventDefault();
    
    if (!newTeamName.trim()) return;
    
    try {
      const team = await makeRequest('/org/create/team', {
        method: 'POST',
        headers: {
          'x-source': 'console'
        },
        body: JSON.stringify({ name: newTeamName })
      });
      
      setTeams(prev => [...prev, team]);
      setNewTeamName('');
    } catch (err) {
      console.error('Failed to create team:', err);
    }
  };
  
  if (loading) return <div>Loading...</div>;
  if (error) return <div>Error: {error}</div>;
  if (!accessToken) return <div>Please authenticate first</div>;
  
  return (
    <div>
      <h2>Team Manager</h2>
      <p>Welcome, {user?.firstName} {user?.lastName}!</p>
      
      <form onSubmit={createTeam}>
        <input
          type="text"
          value={newTeamName}
          onChange={(e) => setNewTeamName(e.target.value)}
          placeholder="Team name"
          required
        />
        <button type="submit">Create Team</button>
      </form>
      
      <h3>Teams ({teams.length})</h3>
      <ul>
        {teams.map(team => (
          <li key={team.id}>
            {team.name} ({team.memberCount} members)
          </li>
        ))}
      </ul>
    </div>
  );
}

export default TeamManager;
```

### 3. Caching Strategy

```javascript
class WeirAPICache {
  constructor(ttl = 300000) { // 5 minutes default TTL
    this.cache = new Map();
    this.ttl = ttl;
  }
  
  get(key) {
    const item = this.cache.get(key);
    if (item && Date.now() - item.timestamp < this.ttl) {
      return item.data;
    }
    this.cache.delete(key);
    return null;
  }
  
  set(key, data) {
    this.cache.set(key, {
      data,
      timestamp: Date.now()
    });
  }
  
  clear() {
    this.cache.clear();
  }
}

// Usage
const cache = new WeirAPICache();

async function getCachedOrganizationInfo() {
  const cacheKey = 'org-info';
  const cached = cache.get(cacheKey);
  
  if (cached) {
    return cached;
  }
  
  const data = await client.makeRequest('/org/information');
  cache.set(cacheKey, data);
  
  return data;
}
```
